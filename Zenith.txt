---Create an instance with properties.
local function create(className, properties, parent)
	local instance = Instance.new(className)

	for key, value in properties do
		instance[key] = value
	end

	instance.Parent = parent
	return instance
end

---Play a tween.
local function tween(instance, info, goal)
	local newTween = tweenService:Create(instance, info, goal)
	newTween:Play()
	return newTween
end

---Connect a global input signal that is cleaned up when the GUI closes.
local function connectInput(signal, callback)
	local connection = signal:Connect(callback)
	table.insert(ZenithHub.connections, connection)
	return connection
end

---Check if an input is a primary press.
local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

---Check if an input is pointer movement.
local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

---Add the accent gradient to an object.
local function addGradient(parent, rotation)
	return create("UIGradient", {
		Color = ColorSequence.new(ACCENT_A, ACCENT_B),
		Rotation = rotation,
	}, parent)
end

---Add a border stroke to an object.
local function addStroke(parent, color, transparency, thickness)
	return create("UIStroke", {
		Color = color,
		Transparency = transparency,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

---Create a rounded row card.
local function createCard(className, parent, text, height)
	local card = create(className, {
		Name = text,
		Size = UDim2.new(1, 0, 0, height),
		BackgroundColor3 = ROW_COLOR,
		BackgroundTransparency = 0.65,
		BorderSizePixel = 0,
	}, ZenithHub.sectionBoxes[parent] or parent)

	create("UICorner", { CornerRadius = UDim.new(0, 5) }, card)
	local stroke = addStroke(card, ACCENT_A, 1)
	card.ClipsDescendants = true

	-- Rows of a collapsible group start hidden and follow the group button.
	local group = ZenithHub.currentGroup
	if group then
		table.insert(group.cards, card)
		card:SetAttribute("GroupHidden", group.collapsed)
		card.Visible = not group.collapsed
	end

	-- Click ripple that expands from the cursor.
	card.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		local offset = Vector2.new(input.Position.X, input.Position.Y) - card.AbsolutePosition

		local ripple = create("Frame", {
			Name = "Ripple",
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromOffset(offset.X, offset.Y),
			Size = UDim2.fromOffset(0, 0),
			BackgroundColor3 = Color3.new(1, 1, 1),
			BackgroundTransparency = 0.8,
			BorderSizePixel = 0,
			ZIndex = 0,
		}, card)

		create("UICorner", { CornerRadius = UDim.new(1, 0) }, ripple)

		local diameter = math.max(card.AbsoluteSize.X, card.AbsoluteSize.Y) * 2

		tween(ripple, MEDIUM_TWEEN, {
			Size = UDim2.fromOffset(diameter, diameter),
			BackgroundTransparency = 1,
		}).Completed:Connect(function()
			ripple:Destroy()
		end)
	end)

	card.MouseEnter:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	card.MouseLeave:Connect(function()
		tween(card, FAST_TWEEN, { BackgroundTransparency = 0.65 })
		tween(stroke, FAST_TWEEN, { Transparency = 1 })
	end)

	return card, stroke
end

---Add a toggle row.
local function addToggle(parent, text, keybind, default, identifier)
	local valueKey = identifier or text
	local supported = ZenithHub.supportedToggles[valueKey] == true and game.PlaceId == 93978595733734
	if not supported then default = false end
	local holder, stroke = createCard("TextButton", parent, text, 28)
	holder.Text = ""
	holder.AutoButtonColor = false

	local box = create("Frame", {
		Size = UDim2.fromOffset(16, 16),
		Position = UDim2.new(0, 8, 0.5, -8),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, box)
	local boxStroke = addStroke(box, default and ACCENT_A or DIM_TEXT_COLOR, default and 0 or 0.5)

	local check = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = default and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, box)

	create("UICorner", { CornerRadius = UDim.new(0, 3) }, check)
	addGradient(check, 45)

	local label = create("TextLabel", {
		Size = UDim2.new(1, -40, 1, 0),
		Position = UDim2.fromOffset(32, 0),
		BackgroundTransparency = 1,
		Text = supported and text or ("[N/A] " .. text),
		TextColor3 = supported and (default and TEXT_COLOR or OFF_TEXT_COLOR) or DIM_TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, holder)

	-- Glowing marker on the left edge while enabled.
	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -10),
		Position = UDim2.fromOffset(0, 5),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = default and 0 or 1,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, accentBar)
	addGradient(accentBar, 90)

	local state = default
	ZenithHub.values[valueKey] = state
	ZenithHub.defaults[valueKey] = default

	ZenithHub.setters[valueKey] = function(value)
		if not supported then return end
		state = value == true
		ZenithHub.values[valueKey] = state

		tween(check, POP_TWEEN, { Size = state and UDim2.fromOffset(10, 10) or UDim2.fromOffset(0, 0) })
		tween(boxStroke, FAST_TWEEN, { Color = state and ACCENT_A or DIM_TEXT_COLOR, Transparency = state and 0 or 0.5 })
		tween(label, FAST_TWEEN, { TextColor3 = state and TEXT_COLOR or OFF_TEXT_COLOR })
		tween(accentBar, FAST_TWEEN, { BackgroundTransparency = state and 0 or 1 })
	end

	-- Clickable keybind label: click it, then press a key. Escape clears it.
	if keybind then
		local keybindName = valueKey .. " Keybind"

		local keybindButton = create("TextButton", {
			Size = UDim2.fromOffset(52, 22),
			Position = UDim2.new(1, -58, 0.5, -11),
			BackgroundTransparency = 1,
			Text = "[" .. keybind .. "]",
			TextColor3 = DIM_TEXT_COLOR,
			TextSize = 12,
			Font = TEXT_FONT,
			TextXAlignment = Enum.TextXAlignment.Right,
			AutoButtonColor = false,
		}, holder)

		---Apply a key name to this toggle.
		local function applyKey(keyName)
			local keyCode = nil

			for _, item in Enum.KeyCode:GetEnumItems() do
				if item.Name == keyName then
					keyCode = item
					break
				end
			end

			ZenithHub.keybinds[valueKey] = keyCode
			ZenithHub.values[keybindName] = keyCode and keyName or ""
			keybindButton.Text = keyCode and ("[" .. keyName .. "]") or "[None]"
			keybindButton.Visible = ZenithHub.values["Keybind Hints"] ~= false
		end

		ZenithHub.defaults[keybindName] = keybind
		ZenithHub.setters[keybindName] = applyKey
		ZenithHub.keyLabels[valueKey] = keybindButton
		applyKey(keybind)

		keybindButton.MouseEnter:Connect(function()
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })
		end)

		keybindButton.MouseLeave:Connect(function()
			if not ZenithHub.listening then
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
			end
		end)

		keybindButton.Activated:Connect(function()
			-- Cancel any other keybind that is still waiting for a key.
			if ZenithHub.cancelListening then
				ZenithHub.cancelListening()
			end

			local previousKey = ZenithHub.values[keybindName]
			keybindButton.Text = "[...]"
			tween(keybindButton, FAST_TWEEN, { TextColor3 = ACCENT_A })

			ZenithHub.cancelListening = function()
				ZenithHub.listening = nil
				ZenithHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(previousKey)
			end

			ZenithHub.listening = function(keyCode)
				ZenithHub.listening = nil
				ZenithHub.cancelListening = nil
				tween(keybindButton, FAST_TWEEN, { TextColor3 = DIM_TEXT_COLOR })
				applyKey(keyCode == Enum.KeyCode.Escape and "" or keyCode.Name)
			end
		end)
	end

	holder.MouseButton1Down:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0 })
		tween(stroke, FAST_TWEEN, { Transparency = 0 })
	end)

	holder.MouseButton1Up:Connect(function()
		tween(holder, FAST_TWEEN, { BackgroundTransparency = 0.35 })
		tween(stroke, FAST_TWEEN, { Transparency = 0.55 })
	end)

	holder.Activated:Connect(function()
		if not supported then
			if ZenithHub.notify then ZenithHub.notify(text .. " unavailable", false) end
			return
		end
		ZenithHub.setters[valueKey](not state)

		if ZenithHub.notify then
			ZenithHub.notify(text, state)
		end
	end)

	holder.MouseEnter:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(36, 0) })
	end)

	holder.MouseLeave:Connect(function()
		tween(label, FAST_TWEEN, { Position = UDim2.fromOffset(32, 0) })
	end)

	table.insert(ZenithHub.searchables, holder)
end

---Add a slider row.
local function addSlider(parent, text, minValue, maxValue, default, decimals, onChanged)
	local holder = createCard("Frame", parent, text, 44)
	local startAlpha = (default - minValue) / (maxValue - minValue)

	create("TextLabel", {
		Size = UDim2.new(1, -70, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local valueLabel = create("TextLabel", {
		Size = UDim2.fromOffset(56, 20),
		Position = UDim2.new(1, -66, 0, 3),
		BackgroundTransparency = 1,
		Text = tostring(default),
		TextColor3 = OFF_TEXT_COLOR,
		TextSize = 13,
		Font = BOLD_FONT,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, holder)

	local bar = create("Frame", {
		Size = UDim2.new(1, -20, 0, 6),
		Position = UDim2.fromOffset(10, 29),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	addStroke(bar, Color3.new(1, 1, 1), 0.88)

	local fill = create("Frame", {
		Size = UDim2.fromScale(startAlpha, 1),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, bar)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	addGradient(fill, 0)

	local knob = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(12, 12),
		Position = UDim2.new(startAlpha, 0, 0.5, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, bar)

	create("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	addStroke(knob, ACCENT_A, 0, 2)
	local knobScale = create("UIScale", {}, knob)

	local hitbox = create("Frame", {
		Size = UDim2.new(1, -20, 0, 22),
		Position = UDim2.fromOffset(10, 21),
		BackgroundTransparency = 1,
	}, holder)

	ZenithHub.values[text] = default
	ZenithHub.defaults[text] = default

	local dragging = false

	local function setValue(value)
		local factor = 10 ^ decimals
		value = math.floor(math.clamp(value, minValue, maxValue) * factor + 0.5) / factor
		local snapped = (value - minValue) / (maxValue - minValue)

		tween(fill, SLIDE_TWEEN, { Size = UDim2.fromScale(snapped, 1) })
		tween(knob, SLIDE_TWEEN, { Position = UDim2.new(snapped, 0, 0.5, 0) })

		valueLabel.Text = tostring(value)
		ZenithHub.values[text] = value
		if onChanged then onChanged(value) end
	end

	ZenithHub.setters[text] = setValue
	if onChanged then onChanged(default) end

	local function updateFromX(positionX)
		local alpha = math.clamp((positionX - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
		setValue(minValue + (maxValue - minValue) * alpha)
	end

	hitbox.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragging = true
		tween(knobScale, POP_TWEEN, { Scale = 1.35 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = ACCENT_A })
		updateFromX(input.Position.X)
	end)

	connectInput(userInputService.InputChanged, function(input)
		if dragging and isMove(input) then
			updateFromX(input.Position.X)
		end
	end)

	connectInput(userInputService.InputEnded, function(input)
		if not dragging or not isPress(input) then
			return
		end

		dragging = false
		tween(knobScale, POP_TWEEN, { Scale = 1 })
		tween(valueLabel, FAST_TWEEN, { TextColor3 = OFF_TEXT_COLOR })
	end)

	table.insert(ZenithHub.searchables, holder)
end

---Add a cycling option row.
local function addCycle(parent, text, options)
	local holder = createCard("Frame", parent, text, 56)

	create("TextLabel", {
		Size = UDim2.new(1, -20, 0, 20),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)

	local button = create("TextButton", {
		Size = UDim2.new(1, -20, 0, 24),
		Position = UDim2.fromOffset(10, 26),
		BackgroundColor3 = ELEMENT_COLOR,
		BorderSizePixel = 0,
		Text = options[1],
		TextColor3 = TEXT_COLOR,
		TextSize = 13,
		Font = TEXT_FONT,
		AutoButtonColor = false,
	}, holder)

	create("UICorner", { CornerRadius = UDim.new(0, 4) }, button)
	local buttonStroke = addStroke(button, DIM_TEXT_COLOR, 0.6)

	local index = 1
	ZenithHub.values[text] = options[1]
	ZenithHub.defaults[text] = options[1]

	ZenithHub.setters[text] = function(value)
		local found = table.find(options, value)
		if not found then
			return
		end

		index = found
		ZenithHub.values[text] = options[index]
		button.Text = options[index]
	end

	button.MouseEnter:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = ACCENT_A, Transparency = 0 })
	end)

	button.MouseLeave:Connect(function()
		tween(buttonStroke, FAST_TWEEN, { Color = DIM_TEXT_COLOR, Transparency = 0.6 })
	end)

	button.Activated:Connect(function()
		index = index % #options + 1
		ZenithHub.values[text] = options[index]

		button.Text = options[index]
		button.TextTransparency = 1
		tween(button, FAST_TWEEN, { TextTransparency = 0 })
	end)

	table.insert(ZenithHub.searchables, holder)
	return function(newOptions)
		if #newOptions == 0 then newOptions = { "Select Config..." } end
		options = newOptions
		index = table.find(options, ZenithHub.values[text]) or 1
		ZenithHub.values[text] = options[index]
		button.Text = options[index]
	end
end

---Make a frame draggable by a handle.
local function makeDraggable(handle, target)
	local dragStart, startPosition

	handle.InputBegan:Connect(function(input)
		if not isPress(input) then
			return
		end

		dragStart = input.Position
		startPosition = target.Position
	end)

	connectInput(userInputService.InputChanged, function(input)
		if not dragStart or not isMove(input) then
			return
		end

		local delta = input.Position - dragStart

		target.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end)

	connectInput(userInputService.InputEnded, function(input)
		if isPress(input) then
			dragStart = nil
		end
	end)
end

---Switch the visible tab.
local function switchTab(name)
	for pageName, page in ZenithHub.pages do
		local selected = pageName == name

		page.Visible = selected
		tween(ZenithHub.tabButtons[pageName], FAST_TWEEN, { TextColor3 = selected and TEXT_COLOR or DIM_TEXT_COLOR })

		if selected then
			page.Position = UDim2.fromOffset(0, 8)
			tween(page, MEDIUM_TWEEN, { Position = UDim2.fromOffset(0, 0) })
		end
	end

	ZenithHub.selected = name
	tween(ZenithHub.indicator, MEDIUM_TWEEN, { Position = UDim2.new((ZenithHub.tabIndex[name] - 1) / #TAB_NAMES, 6, 0, 3) })
end

	end
	local runService = game:GetService("RunService")

	local collectionService = game:GetService("CollectionService")
	local lighting = game:GetService("Lighting")
	local values = ZenithHub.values

		flyVelocity = nil, parryAnimator = nil, parryConnection = nil,
		lastAttack = 0, lastEscape = 0, lastParry = 0, scanElapsed = 0,

	local fov = create("Frame", {

		BackgroundTransparency = 1, BorderSizePixel = 0, Visible = false,
	}, screenGui)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, fov)
	addStroke(fov, Color3.new(1, 1, 1), 0.3, 1)

	local function role(player)
		return player and player.Team and string.lower(player.Team.Name) or ""
	end
	local function inRound()
		local team = role(localPlayer)
		return (team == "killer" or team == "survivors") and workspace:FindFirstChild("Map") ~= nil
	end
	local function rootOf(character)
		return character and character:FindFirstChild("HumanoidRootPart")
	end
	local function findRemote(group, name)
		local remotes = replicatedStorage:FindFirstChild("Remotes")
		local folder = remotes and remotes:FindFirstChild(group)
		local result = folder and folder:FindFirstChild(name)
		return result and result:IsA("RemoteEvent") and result or nil
	end
	local function enemyPlayers()
		local team = role(localPlayer)
		local expected = team == "killer" and "survivors" or team == "survivors" and "killer" or nil
		local result = {}
		if not expected then return result end
		for _, player in ipairs(playersService:GetPlayers()) do
			if player ~= localPlayer and role(player) == expected then
				local character = player.Character
				local humanoid = character and character:FindFirstChildOfClass("Humanoid")
				if humanoid and humanoid.Health > 0 and rootOf(character) then table.insert(result, player) end
			end
		end
		return result
	end
	local function restoreCollisions()
		for part, original in pairs(state.collisions) do
			if part.Parent then part.CanCollide = original end
			state.collisions[part] = nil
		end
	end
	local function restoreEffects()
		for effect, original in pairs(state.effects) do
			if effect.Parent then
				if effect:IsA("PostEffect") then effect.Enabled = original else effect.Visible = original end
			end
			state.effects[effect] = nil
		end
	end
	local function restoreMovement()
		local humanoid = state.humanoid
		if humanoid and humanoid.Parent then
			if state.originalSpeed ~= nil then humanoid.WalkSpeed = state.originalSpeed end
			if state.originalJump then
				humanoid.UseJumpPower = state.originalJump.useJumpPower
				humanoid.JumpPower = state.originalJump.jumpPower
				humanoid.JumpHeight = state.originalJump.jumpHeight
			end
		end
		state.originalSpeed, state.originalJump = nil, nil
		restoreCollisions()
		if state.flyVelocity then state.flyVelocity:Destroy() state.flyVelocity = nil end
	end
	local function setCharacter()
		local character = localPlayer.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if character ~= state.character or humanoid ~= state.humanoid then
			restoreMovement()
			state.character, state.humanoid = character, humanoid
		end
		return character, humanoid, rootOf(character)
	end
	local function updateMovement()
		local character, humanoid, root = setCharacter()
		if not character or not humanoid or not root then return end
		if values["Walk Speed"] then
			if state.originalSpeed == nil then state.originalSpeed = humanoid.WalkSpeed end
			humanoid.WalkSpeed = values["Speed Value"] or 32
		elseif state.originalSpeed ~= nil then
			humanoid.WalkSpeed, state.originalSpeed = state.originalSpeed, nil
		end
		if values["Jump Power (+ unlock)"] then
			if not state.originalJump then
				state.originalJump = { useJumpPower = humanoid.UseJumpPower, jumpPower = humanoid.JumpPower, jumpHeight = humanoid.JumpHeight }
			end
			humanoid.UseJumpPower = true
			humanoid.JumpPower = values["Jump Value"] or 60
		elseif state.originalJump then
			humanoid.UseJumpPower = state.originalJump.useJumpPower
			humanoid.JumpPower, humanoid.JumpHeight = state.originalJump.jumpPower, state.originalJump.jumpHeight
			state.originalJump = nil
		end
		if values.Noclip then
			for _, part in ipairs(character:GetDescendants()) do
				if part:IsA("BasePart") then
					if state.collisions[part] == nil then state.collisions[part] = part.CanCollide end
					part.CanCollide = false
				end
			end
		else
			restoreCollisions()
		end
		if values.Fly then
			if not state.flyVelocity or state.flyVelocity.Parent ~= root then
				if state.flyVelocity then state.flyVelocity:Destroy() end
				state.flyVelocity = Instance.new("BodyVelocity")
				state.flyVelocity.Name = "ZenithFly"
				state.flyVelocity.MaxForce = Vector3.new(1e8, 1e8, 1e8)
				state.flyVelocity.Parent = root
			end
			local vertical = (userInputService:IsKeyDown(Enum.KeyCode.Space) and 1 or 0)
				- (userInputService:IsKeyDown(Enum.KeyCode.LeftControl) and 1 or 0)
			local direction = humanoid.MoveDirection + Vector3.new(0, vertical, 0)
			if direction.Magnitude > 1 then direction = direction.Unit end
			state.flyVelocity.Velocity = direction * (values["Fly Speed"] or 60)
		elseif state.flyVelocity then
			state.flyVelocity:Destroy()
			state.flyVelocity = nil
		end
	end
	local function updateBlind()
		if not values["Anti Blind"] then restoreEffects() return end
		for effect in pairs(state.effects) do
			if not effect.Parent then state.effects[effect] = nil end
		end
		local roots = { lighting }
		local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
		if playerGui then table.insert(roots, playerGui) end
		for _, parent in ipairs(roots) do
			for _, effect in ipairs(parent:GetDescendants()) do
				local name = string.lower(effect.Name)
				if name:find("blind", 1, true) or name:find("flash", 1, true) or name:find("whiteout", 1, true) then
					if effect:IsA("PostEffect") then
						if state.effects[effect] == nil then state.effects[effect] = effect.Enabled end
						effect.Enabled = false
					elseif effect:IsA("GuiObject") and not effect:IsDescendantOf(screenGui) then
						if state.effects[effect] == nil then state.effects[effect] = effect.Visible end
						effect.Visible = false
					end
				end
			end
		end
	end
	local function disconnectParry()
		if state.parryConnection then state.parryConnection:Disconnect() end
		state.parryConnection, state.parryAnimator = nil, nil
	end
	local function updateParry()
		local animator
		if values["Auto Parry"] and inRound() and role(localPlayer) == "survivors" then
			for _, player in ipairs(enemyPlayers()) do
				local humanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
				animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
				if animator then break end
			end
		end
		if animator == state.parryAnimator then return end
		disconnectParry()
		state.parryAnimator = animator
		if not animator then return end
		state.parryConnection = animator.AnimationPlayed:Connect(function(track)
			if not state.alive or not values["Auto Parry"] or not inRound() then return end
			local name = string.lower(track.Name .. " " .. (track.Animation and track.Animation.AnimationId or ""))
			for _, ignored in ipairs({ "idle", "walk", "run", "vault", "stun", "carry", "mori", "break" }) do
				if name:find(ignored, 1, true) then return end
			end
			local attack = name:find("attack", 1, true) or name:find("swing", 1, true) or name:find("slash", 1, true) or name:find("lunge", 1, true)
				or track.Priority == Enum.AnimationPriority.Action or track.Priority == Enum.AnimationPriority.Action2
				or track.Priority == Enum.AnimationPriority.Action3 or track.Priority == Enum.AnimationPriority.Action4
			if not attack then return end
			local character, localRoot = localPlayer.Character, rootOf(localPlayer.Character)
			local target = animator:FindFirstAncestorOfClass("Model")
			local targetRoot = rootOf(target)
			if not character or not character:FindFirstChild("Parrying Dagger") or not localRoot or not targetRoot then return end
			if (localRoot.Position - targetRoot.Position).Magnitude > 18 or os.clock() - state.lastParry < 0.3 then return end
			local remotes = replicatedStorage:FindFirstChild("Remotes")
			local items = remotes and remotes:FindFirstChild("Items")
			local dagger = items and items:FindFirstChild("Parrying Dagger")
			local remote = dagger and dagger:FindFirstChild("parry")
			if remote and remote:IsA("RemoteEvent") then
				state.lastParry = os.clock()
				remote:FireServer(character)
			end
		end)
	end
	local function updateActions(root)
		if not inRound() then return end
		local now = os.clock()
		if values["Auto Kill"] and role(localPlayer) == "killer" and root then
			local range = values["Kill Range"] or 12
			for _, player in ipairs(enemyPlayers()) do
				local target = rootOf(player.Character)
				if target and (target.Position - root.Position).Magnitude <= range then
					local remote = findRemote("Attacks", "BasicAttack")
					if remote and now - state.lastAttack >= math.max(0.15, values["Auto Kill Delay"] or 0.15) then
						state.lastAttack = now
						remote:FireServer()
					end
					break
				end
			end
		end
		local character = localPlayer.Character
		if values["Auto Escape"] and role(localPlayer) == "survivors" and character and character:GetAttribute("IsHooked") and now - state.lastEscape >= 1 then
			local remote = findRemote("Carry", "SelfUnHookEvent")
			if remote then state.lastEscape = now remote:FireServer() end
		end
	end

		local record = state.markers[instance]

			local highlight = Instance.new("Highlight")
			highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
			highlight.FillTransparency = 1
			highlight.OutlineTransparency = 0
			highlight.Parent = visualFolder
			local outlineBox = Instance.new("SelectionBox")
			outlineBox.LineThickness = 0.04
			outlineBox.SurfaceTransparency = 1
			outlineBox.Parent = visualFolder
			local billboard = Instance.new("BillboardGui")
			billboard.Size = UDim2.fromOffset(160, 36)
			billboard.StudsOffset = Vector3.new(0, 3, 0)
			billboard.AlwaysOnTop = true
			billboard.Parent = visualFolder
			local textLabel = create("TextLabel", {
				Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1,
				TextColor3 = color, TextStrokeTransparency = 0.35,
				Font = BOLD_FONT, TextSize = 13,
			}, billboard)
			record = { highlight = highlight, box = outlineBox, billboard = billboard, label = textLabel }
			state.markers[instance] = record
		end
		record.highlight.Adornee = instance
		record.highlight.OutlineColor = color
		record.highlight.Enabled = showHighlight
		record.box.Adornee = instance
		record.box.Color3 = color
		record.box.Visible = showBox == true
		record.billboard.Adornee = adornee
		record.billboard.Enabled = showLabel
		record.label.TextColor3 = color

	end
	local function removeMarkers(seen)
		for instance, record in pairs(state.markers) do
			if not seen[instance] or not instance.Parent then
				record.highlight:Destroy()
				record.box:Destroy()
				record.billboard:Destroy()
				state.markers[instance] = nil
			end
		end
	end
	local function updateVisuals(root)
		local seen = {}
		if inRound() and root then
			local maxDistance = values["ESP Max Distance"] or 1000
			for _, player in ipairs(playersService:GetPlayers()) do
				local targetRole, character = role(player), player.Character
				local target = rootOf(character)
				local humanoid = character and character:FindFirstChildOfClass("Humanoid")
				local showHighlight = (targetRole == "killer" and values["Killer ESP"]) or (targetRole == "survivors" and values["Survivor ESP"])
				if player ~= localPlayer and target and humanoid and humanoid.Health > 0 and (targetRole == "killer" or targetRole == "survivors") then
					local distance = (target.Position - root.Position).Magnitude
					if distance <= maxDistance and (showHighlight or values["Name Tags"] or values["Distance Labels"]) then
						local hex = targetRole == "killer" and values["Killer Color"] or values["Survivor Color"]
						local text = values["Name Tags"] and player.Name or ""
						marker(character, target, text, Color3.fromHex(hex), distance, showHighlight == true, values["Name Tags"] or values["Distance Labels"], seen)
					end
				end
			end
			if values["Generator ESP"] then
				local map = workspace:FindFirstChild("Map")
				for _, model in ipairs(map:GetDescendants()) do
					if model:IsA("Model") and string.lower(model.Name) == "generator" then
						local body = model:FindFirstChild("GeneratorBody", true)
						if body and body:IsA("BasePart") then
							local distance = (body.Position - root.Position).Magnitude
							if distance <= maxDistance then marker(model, body, "Generator", Color3.fromRGB(255, 190, 88), distance, true, true, seen) end
						end
					end
				end
			end
			-- SurvivorProximity and KillerProximity identify interaction points by tag.
			local map = workspace:FindFirstChild("Map")
			local categories = {
				{ tag = "HookPoint", label = "Hook", enabled = values["Hook ESP"], nearest = values["Nearest Hook Only"] },
				{ tag = "PalletPoint", label = "Pallet", enabled = values["Pallet ESP"] },
				{ tag = "VaultPoint", label = "Window", enabled = values["Window ESP"] or values["Box ESP (windows)"], window = true },
				{ tag = "ExitPoint", label = "Exit Gate", enabled = values["Exit Gate ESP"] },
			}
			for _, category in ipairs(categories) do
				if category.enabled then
					local candidates = {}
					for _, point in ipairs(collectionService:GetTagged(category.tag)) do
						if point:IsA("BasePart") and point:IsDescendantOf(map) then
							local distance = (point.Position - root.Position).Magnitude
							if distance <= maxDistance then table.insert(candidates, { point = point, distance = distance }) end
						end
					end
					if category.nearest then table.sort(candidates, function(a, b) return a.distance < b.distance end) end
					for index, candidate in ipairs(candidates) do
						if category.nearest and index > 1 then break end
						local point = candidate.point
						local model = point:FindFirstAncestorOfClass("Model")
						local object = model and model ~= map and model:IsDescendantOf(map) and model or point
						local color = category.window and Color3.fromHex(values["Window Color"]) or Color3.fromRGB(255, 190, 88)
						local highlighted = not category.window or values["Window ESP"] == true
						marker(object, point, category.label, color, candidate.distance, highlighted, true, seen, category.window and values["Box ESP (windows)"])
					end
				end
			end
		end
		removeMarkers(seen)
	end
	local function updateAim()
		local enabled = values.Aimbot == true and inRound()
		if not enabled and state.aimActive then
			state.aimActive = false
			aimButton.TextColor3 = DIM_TEXT_COLOR
		end
		aimButton.Visible = enabled
		fov.Visible = enabled and values["Visible FOV"] == true
		local radius = values["Aim FOV"] or 90
		local mouse = userInputService:GetMouseLocation()
		fov.Position = UDim2.fromOffset(mouse.X, mouse.Y)
		fov.Size = UDim2.fromOffset(radius * 2, radius * 2)
		if not enabled or userInputService:GetFocusedTextBox() then return end
		if not (values["Always Active"] or state.aimActive or userInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)) then return end
		local camera = workspace.CurrentCamera
		if not camera then return end
		local best, bestDistance
		for _, player in ipairs(enemyPlayers()) do
			local character = player.Character
			local target = character and (character:FindFirstChild(values["Target Part"] or "Head") or rootOf(character))
			if target and target:IsA("BasePart") then
				local projected, visible = camera:WorldToScreenPoint(target.Position)
				local distance = (Vector2.new(projected.X, projected.Y) - mouse).Magnitude
				if visible and distance <= radius and (not bestDistance or distance < bestDistance) then best, bestDistance = target, distance end
			end
		end
		if best then
			local point = best.Position
			if values["Movement Prediction"] then point = point + best.AssemblyLinearVelocity * 0.12 end
			local origin = camera.CFrame.Position
			if (point - origin).Magnitude > 0.001 then
				local smoothness = math.clamp(values["Aim Smoothness"] or 0.25, 0, 1)
				camera.CFrame = camera.CFrame:Lerp(CFrame.lookAt(origin, point), math.max(0.02, 1 - smoothness))
			end
		end
	end
	connectInput(userInputService.JumpRequest, function()
		if values["Infinite Jump"] and not userInputService:GetFocusedTextBox() then
			local humanoid = localPlayer.Character and localPlayer.Character:FindFirstChildOfClass("Humanoid")
			if humanoid then humanoid:ChangeState(Enum.HumanoidStateType.Jumping) end
		end
	end)
	connectInput(userInputService.InputBegan, function(input, processed)
		if processed or ZenithHub.listening or input.UserInputType ~= Enum.UserInputType.Keyboard or userInputService:GetFocusedTextBox() then return end
		for bind, feature in pairs({ ["Walk Speed Bind"] = "Walk Speed", ["Noclip Bind"] = "Noclip", ["Fly Bind"] = "Fly" }) do
			if values[bind] ~= "" and values[bind] == input.KeyCode.Name then
				ZenithHub.setters[feature](not values[feature])
			end
		end
	end)
	local reportedErrors = {}
	local function safeStep(name, callback, ...)
		local ok, err = pcall(callback, ...)
		if not ok and not reportedErrors[name] then
			reportedErrors[name] = true
			warn("Zenith " .. name .. ": " .. tostring(err))
		end
	end
	connectInput(runService.Heartbeat, function(dt)
		if not state.alive then return end
		safeStep("movement", updateMovement)
		state.scanElapsed += dt
		state.visualElapsed += dt
		state.blindElapsed += dt
		if state.scanElapsed >= 0.1 then
			state.scanElapsed = 0
			safeStep("parry", updateParry)
			safeStep("actions", updateActions, rootOf(localPlayer.Character))
		end
		if state.visualElapsed >= 0.75 then
			state.visualElapsed = 0
			safeStep("visuals", updateVisuals, rootOf(localPlayer.Character))
		end
		if state.blindElapsed >= 0.75 then state.blindElapsed = 0 safeStep("anti-blind", updateBlind) end
	end)
	connectInput(runService.RenderStepped, function()
		if not state.alive then return end
		local ok, err = pcall(updateAim)
		if not ok and not state.aimReported then state.aimReported = true warn("Zenith aim: " .. tostring(err)) end
	end)
	-- Use the interaction-point tags exposed by the inspected game scripts.
	function state.teleportTo(tag)
		if state.teleporting then return end
		local character = localPlayer.Character
		local root = rootOf(character)
		local map = inRound() and workspace:FindFirstChild("Map")
		if not root or not map then return ZenithHub.notify("No active round", false) end
		local nearest, nearestDistance
		for _, point in ipairs(collectionService:GetTagged(tag)) do
			if point:IsA("BasePart") and point:IsDescendantOf(map) then
				local distance = (point.Position - root.Position).Magnitude
				if not nearestDistance or distance < nearestDistance then nearest, nearestDistance = point, distance end
			end
		end
		if not nearest then return ZenithHub.notify("No " .. tag .. " found", false) end
		state.teleporting = true
		task.spawn(function()
			local ok, err = pcall(function()
				local target = nearest.Position + Vector3.new(0, 3, 0)
				local deadline = os.clock() + 10
				while state.alive and root.Parent and nearest.Parent and localPlayer.Character == character and inRound() and os.clock() < deadline do
					local delta = target - root.Position
					if delta.Magnitude <= 24 then root.CFrame = root.CFrame + delta break end
					root.CFrame = root.CFrame + delta.Unit * 24
					task.wait(0.1)
				end
			end)
			state.teleporting = false
			if not ok then warn("Zenith teleport: " .. tostring(err)) end
		end)
	end
	local function stop()
		if not state.alive then return end
		state.alive = false
		disconnectParry()
		restoreMovement()
		restoreEffects()
		removeMarkers({})
		visualFolder:Destroy()
	end
	screenGui.Destroying:Connect(stop)
	return state
end

	local function trackTransparency(instance)
		if instance.Name == "Ripple" or transparencyDefaults[instance] then return end
		local properties = {}
		if instance:IsA("GuiObject") then
			table.insert(properties, "BackgroundTransparency")
		end
		if instance:IsA("TextLabel") or instance:IsA("TextButton") or instance:IsA("TextBox") then
			table.insert(properties, "TextTransparency")
			table.insert(properties, "TextStrokeTransparency")
		end
		if instance:IsA("ImageLabel") or instance:IsA("ImageButton") then
			table.insert(properties, "ImageTransparency")
		end
		if instance:IsA("ScrollingFrame") then
			table.insert(properties, "ScrollBarImageTransparency")
		end
		if instance:IsA("UIStroke") then
			table.insert(properties, "Transparency")
		end
		if #properties == 0 then return end
		local defaults = {}
		transparencyDefaults[instance] = defaults
		for _, property in properties do
			defaults[property] = instance[property]
			connectInput(instance:GetPropertyChangedSignal(property), function()
				if applyingTransparency then return end
				local original = instance[property]
				defaults[property] = original
				if transparency > 0 then
					applyingTransparency = true
					instance[property] = original + (1 - original) * transparency
					applyingTransparency = false
				end
			end)
			if transparency > 0 then
				applyingTransparency = true
				instance[property] = defaults[property] + (1 - defaults[property]) * transparency
				applyingTransparency = false
			end
		end
	end
	local function setWindowTransparency(value)
		transparency = math.clamp(value, 0, 0.9)
		applyingTransparency = true
		for instance, defaults in transparencyDefaults do
			if instance.Parent then
				for property, original in defaults do
					instance[property] = original + (1 - original) * transparency
				end
			end
		end
		applyingTransparency = false
	end

	local function validName(name)
		return typeof(name) == "string" and #name > 0 and #name <= 40 and name:match("^[%w_%-]+$") ~= nil
	end
	local function configPath(name)
		if not validName(name) then return nil end
		return CONFIG_FOLDER .. "/" .. name .. ".json"
	end
	local function ensureFolder()
		if not (isfolder and makefolder) then error("Executor folder functions unavailable") end
		if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end
	end
	local selectorUpdate
	local autoloadPath = CONFIG_FOLDER .. "/autoload.txt"
	local autoloadName = "None"
	local autoloadLabel
	local function showAutoload(name)
		autoloadName = name or "None"
		if autoloadLabel then autoloadLabel.Text = "Autoload config: " .. autoloadName end
	end
	local function refreshConfigs(preferred)
		local options = { "Select Config..." }
		if listfiles and isfolder and isfolder(CONFIG_FOLDER) then
			local ok, files = pcall(listfiles, CONFIG_FOLDER)
			if ok and typeof(files) == "table" then
				local found = {}
				for _, file in files do
					local name = file:match("([^/\\]+)%.json$")
					if name and validName(name) and not found[name] then
						found[name] = true
						table.insert(options, name)
					end
				end
			end
		end
		table.sort(options, function(a, b)
			if a == b then return false end
			if a == "Select Config..." then return true end
			if b == "Select Config..." then return false end
			return string.lower(a) < string.lower(b)
		end)
		selectorUpdate(options)
		if preferred then ZenithHub.setters["Select Config"](preferred) end
	end
	local function saveConfig(name, allowOverwrite)
		local path = configPath(name)
		if not path then return setStatus("Use letters, numbers, _ or - for the config name.") end
		if not (writefile and isfile) then return setStatus("Saving is not supported here.") end
		local ok, result = pcall(function()
			ensureFolder()
			if isfile(path) and not allowOverwrite then error("Config already exists") end
			if not isfile(path) and allowOverwrite then error("Select an existing config") end
			local data = {}
			for key, value in ZenithHub.values do
				if key ~= "Config Name" and key ~= "Select Config" then data[key] = value end
			end
			writefile(path, httpService:JSONEncode(data))
		end)
		if not ok then return setStatus("Save failed: " .. tostring(result)) end
		refreshConfigs(name)
		setStatus("Config saved: " .. name)
	end
	local function loadConfig(name)
		local path = configPath(name)
		if not path or not (readfile and isfile and isfile(path)) then
			return setStatus("Select an existing config.")
		end
		local ok, data = pcall(function() return httpService:JSONDecode(readfile(path)) end)
		if not ok or typeof(data) ~= "table" then return setStatus("Config file is invalid.") end
		for key, value in data do
			local setter = ZenithHub.setters[key]
			if setter and key ~= "Config Name" and key ~= "Select Config" then pcall(setter, value) end
		end
		setStatus("Config loaded: " .. name)
	end

		local query = string.lower(searchBox.Text)
		clearSearch.Visible = query ~= ""
		tween(searchStroke, FAST_TWEEN, {
			Transparency = searchFocused and 0 or (query == "" and 1 or 0.25),
		})

		for _, element in ZenithHub.searchables do
			element.Visible = (query == "" or string.find(string.lower(element.Name), query, 1, true) ~= nil) and not element:GetAttribute("GroupHidden")
		end

		for _, box in ZenithHub.boxes do
			local anyVisible = query == ""

			for _, child in box:GetChildren() do
				if child:IsA("GuiObject") and child.Visible and not table.find(ZenithHub.sections, child) then
					anyVisible = true
					break
				end
			end

			-- Keep matching section headings visible so filtered results have context.
			for _, section in ZenithHub.sections do
				if section.Parent == box then section.Visible = anyVisible end
			end
			box.Visible = anyVisible
		end
	end)

	-- The menu keybind hides and shows the menu.
	connectInput(userInputService.InputBegan, function(input, gameProcessed)
		if input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end

		-- A keybind button is waiting for the next key.
		if ZenithHub.listening then
			return ZenithHub.listening(input.KeyCode)
		end

		if gameProcessed then
			return
		end

		if input.KeyCode == ZenithHub.menuKey then
			screenGui.Enabled = not screenGui.Enabled
			return
		end

		-- Toggle any feature bound to this key.
		for featureName, keyCode in ZenithHub.keybinds do
			if keyCode == input.KeyCode then
				ZenithHub.setters[featureName](not ZenithHub.values[featureName])
				ZenithHub.notify(featureName, ZenithHub.values[featureName])
			end
		end
	end)

		while screenGui.Parent do
			local visible = ZenithHub.values["Keybind Hints"] ~= false

			for _, keybindButton in ZenithHub.keyLabels do
				keybindButton.Visible = visible
			end

			task.wait(0.5)
		end
	end)

-- ZENITH compatibility dispatcher.
-- Chess and Vocabulary are independent products; this file only selects one.

if not game:IsLoaded() then
	game.Loaded:Wait()
end

local CHESS_PLACES = {
	[130878180532794] = true, -- RoChess
	[89083628291452] = true, -- Chess Duels
	[93365009198479] = true, -- Play Chess
}

local productFile = CHESS_PLACES[game.PlaceId] and "ZenithChess.txt" or "ZenithVocabulary.txt"
local productUrl = "https://raw.githubusercontent.com/XzAngel19/Models/arena/cbc36148-models/" .. productFile
local source = game:HttpGet(productUrl, true)
local chunk, compileError = loadstring(source, "Zenith/" .. productFile)
if not chunk then
	error("[Zenith] No se pudo compilar " .. productFile .. ": " .. tostring(compileError), 0)
end

return chunk()

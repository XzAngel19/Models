-- Flow: if Zenith/KeySystem.log holds a key that has not expired yet, the Word Helper
-- opens directly. Otherwise the Key System opens; after "Success!" it closes itself,
-- writes the log (entered + expiry time) and opens the Word Helper.

-- Key log. Stored in the executor workspace as Zenith/KeySystem.log (append-only,
-- one line per accepted key). The newest "expires=" value is the active one.
local ZenithKeyLog = {
	Folder = "Zenith",
	File = "Zenith/KeySystem.log",
	DurationSeconds = 12 * 60 * 60,
	SessionAttribute = "ZenithStandardExpiresAt",
}

local function zenithFileFunction(name)
	local environment = getfenv and getfenv() or _G
	local value = environment[name]
	if type(value) ~= "function" and type(getgenv) == "function" then
		local ok, genv = pcall(getgenv)
		if ok and type(genv) == "table" then
			value = genv[name]
		end
	end
	return type(value) == "function" and value or nil
end

function ZenithKeyLog.FormatDate(timestamp)
	local ok, text = pcall(os.date, "%Y-%m-%d %H:%M:%S", timestamp)
	if ok and type(text) == "string" then
		return text
	end
	return tostring(timestamp)
end

function ZenithKeyLog.ReadExpiry()
	local readFile = zenithFileFunction("readfile")
	local isFile = zenithFileFunction("isfile")
	local latest = nil

	if readFile then
		local ok, content = pcall(function()
			if isFile and not isFile(ZenithKeyLog.File) then
				return nil
			end
			return readFile(ZenithKeyLog.File)
		end)
		if ok and type(content) == "string" then
			for value in string.gmatch(content, "expires=(%d+)") do
				latest = tonumber(value)
			end
		end
	end

	-- Studio / executors without file access: remember the key for this session only.
	if not latest then
		local player = game:GetService("Players").LocalPlayer
		local attribute = player and player:GetAttribute(ZenithKeyLog.SessionAttribute)
		if type(attribute) == "number" then
			latest = attribute
		end
	end
	return latest
end

function ZenithKeyLog.Record()
	local now = os.time()
	local expiresAt = now + ZenithKeyLog.DurationSeconds
	local line = string.format(
		"[%s] KEY ACCEPTED | entered=%d (%s) | expires=%d (%s)\n",
		ZenithKeyLog.FormatDate(now),
		now,
		ZenithKeyLog.FormatDate(now),
		expiresAt,
		ZenithKeyLog.FormatDate(expiresAt)
	)

	local writeFile = zenithFileFunction("writefile")
	local appendFile = zenithFileFunction("appendfile")
	local readFile = zenithFileFunction("readfile")
	local isFile = zenithFileFunction("isfile")
	local makeFolder = zenithFileFunction("makefolder")

	if writeFile then
		pcall(function()
			if makeFolder then
				pcall(makeFolder, ZenithKeyLog.Folder)
			end
			local exists = isFile and isFile(ZenithKeyLog.File)
			if exists and appendFile then
				appendFile(ZenithKeyLog.File, line)
			elseif exists and readFile then
				writeFile(ZenithKeyLog.File, readFile(ZenithKeyLog.File) .. line)
			else
				writeFile(ZenithKeyLog.File, "-- Zenith Key System log. Each line: when the key was entered and when a new key is required.\n" .. line)
			end
		end)
	end

	pcall(function()
		game:GetService("Players").LocalPlayer:SetAttribute(ZenithKeyLog.SessionAttribute, expiresAt)
	end)
	return expiresAt
end

function ZenithKeyLog.FormatRemaining(seconds)
	seconds = math.max(0, math.floor(seconds))
	if seconds <= 0 then
		return "Expired"
	end
	local hours = math.floor(seconds / 3600)
	local minutes = math.floor((seconds % 3600) / 60)
	return string.format("%02d:%02d:%02d - Remaining", hours, minutes, seconds % 60)
end

	-- Pure Luau base64 decoder, used for the bundled transparent PNGs.
	local BASE64_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
	local BASE64_LOOKUP = {}
	for index = 1, #BASE64_ALPHABET do
		BASE64_LOOKUP[string.sub(BASE64_ALPHABET, index, index)] = index - 1
	end

	local function decodeBase64(data)
		local output = table.create(math.floor(#data * 0.75))
		local outputLength = 0
		local accumulator = 0
		local groupLength = 0

		local function appendByte(value)
			outputLength = outputLength + 1
			output[outputLength] = string.char(value)
		end

		-- Decode one four-character group at a time. Keeping the accumulator small
		-- avoids floating-point overflow on the embedded 15 KB division logo.
		for index = 1, #data do
			local character = string.sub(data, index, index)
			local value = BASE64_LOOKUP[character]
			if value then
				accumulator = accumulator * 64 + value
				groupLength = groupLength + 1
				if groupLength == 4 then
					appendByte(math.floor(accumulator / 65536) % 256)
					appendByte(math.floor(accumulator / 256) % 256)
					appendByte(accumulator % 256)
					accumulator = 0
					groupLength = 0
				end
			end
		end

		-- Padding characters are skipped above, leaving two or three useful values.
		if groupLength == 2 then
			appendByte(math.floor(accumulator / 16) % 256)
		elseif groupLength == 3 then
			appendByte(math.floor(accumulator / 1024) % 256)
			appendByte(math.floor(accumulator / 4) % 256)
		end

		return table.concat(output, "", 1, outputLength)
	end

	local function normalizeAsset(value)
		if value == nil then
			return ""
		end

		local text = tostring(value)
		if text == "" or text == "0" then
			return ""
		end
		if string.find(text, "://", 1, true) then
			return text
		end
		if string.match(text, "^%d+$") then
			return "rbxassetid://" .. text
		end
		return text
	end

	local function getExecutorFunction(name)
		local value = sharedEnvironment[name]
		if type(value) == "function" then
			return value
		end
		return nil
	end

	local writeFile = getExecutorFunction("writefile")

	local function resolveEmbeddedAsset(key, fileName)
		local configured = normalizeAsset(Config.Assets[key])
		if configured ~= "" then
			return configured
		end

		if not writeFile or not getCustomAsset then
			return ""
		end

		local folder = "ZNTHMainMenuAssets"
		local path = folder .. "/" .. fileName
		local ok = pcall(function()
			-- Some executors omit isfolder and throw when makefolder sees an existing
			-- directory, so folder creation is intentionally its own ignored pcall.
			if makeFolder then
				pcall(makeFolder, folder)
			end
			-- Rewriting guarantees a stale file from an older script can never win.
			writeFile(path, decodeBase64(EMBEDDED_ASSETS[key]))
		end)

		if not ok then
			return ""
		end

		local assetOk, asset = pcall(getCustomAsset, path)
		if assetOk and type(asset) == "string" then
			return asset
		end
		return ""
	end

	-- UI utility functions.
	local function create(className, properties, parent)
		local object = Instance.new(className)
		for property, value in pairs(properties or {}) do
			object[property] = value
		end
		if parent then
			object.Parent = parent
		end
		return object
	end

	local function connect(signal, callback)
		local connection = signal:Connect(callback)
		table.insert(Menu.connections, connection)
		return connection
	end

	local function tween(object, info, goals)
		if Menu.destroyed or not object or not object.Parent then
			return nil
		end
		local animation = tweenService:Create(object, info, goals)
		animation:Play()
		return animation
	end

	local function bindPressScale(button, target)
		local scaleObject = create("UIScale", { Scale = 1 }, target or button)
		connect(button.MouseButton1Down, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 0.975 })
		end)
		connect(button.MouseButton1Up, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 1 })
		end)
		connect(button.MouseLeave, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 1 })
		end)
		return scaleObject
	end

		local function restoreKeyPrompt()
			if Menu.destroyed or Menu.keyAccepted then
				return
			end
			keyInput.Visible = false
			keyInput.Text = ""
			keyLabel.Text = "Key System"
			keyLabel.TextColor3 = Config.White
			keyLabel.Visible = true
			keyButton.Visible = true
		end

		local function submitKey()
			if Menu.destroyed or Menu.keyAccepted then
				return
			end

			if not string.find(keyInput.Text, "%S") then
				keyInput.Text = ""
				keyInput.PlaceholderText = "ENTER ANY KEY FIRST"
				tween(keyInput, FAST_TWEEN, { TextColor3 = Config.Accent })
				task.delay(0.2, function()
					if not Menu.destroyed then
						tween(keyInput, FAST_TWEEN, { TextColor3 = Config.White })
					end
				end)
				return
			end

			local acceptedKey = keyInput.Text
			Menu.keyAccepted = true
			-- The Standard countdown starts now, at the moment a key is accepted.
			-- It is written to Zenith/KeySystem.log so the Word Helper remembers it for 12 hours.
			Menu.standardExpiresAt = ZenithKeyLog.Record()
			if Menu.references.CountdownValue then
				Menu.references.CountdownValue.Text = "12 Hours - Remaining"
			end
			keyInput.Visible = false
			keyLabel.Visible = true
			keyLabel.Text = "Success!"
			keyLabel.TextColor3 = Color3.fromRGB(128, 255, 164)
			keyButton.Visible = true
			tween(keyBar, FAST_TWEEN, {
				BackgroundColor3 = Color3.fromRGB(12, 36, 23),
				BackgroundTransparency = 0.02,
			})
			if keyStroke then
				tween(keyStroke, FAST_TWEEN, { Color = Color3.fromRGB(128, 255, 164), Transparency = 0.1 })
			end

			local callback = Menu.callbacks.KeyAccepted
			if type(callback) == "function" then
				task.spawn(callback, Menu, acceptedKey)
			end

			-- Slide the Key System out, remove it completely, then open the Word Helper.
			task.delay(0.7, function()
				if Menu.destroyed then
					return
				end
				Menu:SetOpen(false)
				task.wait(OPEN_TWEEN.Time + 0.1)
				Menu:Destroy()
				if type(onKeyAccepted) == "function" then
					onKeyAccepted(Menu.standardExpiresAt)
				end
			end)
		end

		connect(keyButton.Activated, function()
			if Menu.keyAccepted then
				return
			end
			keyLabel.Visible = false
			keyButton.Visible = false
			keyInput.Visible = true
			keyInput.PlaceholderText = "TYPE ANY KEY  •  PRESS ENTER"
			task.defer(function()
				if not Menu.destroyed and keyInput.Visible then
					keyInput:CaptureFocus()
				end
			end)
		end)
		connect(keyInput.FocusLost, function(enterPressed)
			if enterPressed then
				submitKey()
			else
				restoreKeyPrompt()
			end
		end)
		connect(updatesButton.Activated, function()
			Menu:OpenUpdates()
		end)

		return root
	end

	local function captureCoreGuiState()
		if Menu.coreGuiCaptured then
			return
		end
		Menu.coreGuiCaptured = true

		local ok, coreTypes = pcall(function()
			return Enum.CoreGuiType:GetEnumItems()
		end)
		if ok then
			for _, coreType in ipairs(coreTypes) do
				if coreType ~= Enum.CoreGuiType.All then
					local stateOk, enabled = pcall(function()
						return starterGui:GetCoreGuiEnabled(coreType)
					end)
					if stateOk then
						Menu.coreGuiStates[coreType] = enabled
					end
				end
			end
		end

		local topbarOk, topbarEnabled = pcall(function()
			return starterGui:GetCore("TopbarEnabled")
		end)
		if topbarOk and type(topbarEnabled) == "boolean" then
			Menu.topbarWasEnabled = topbarEnabled
		end

		-- Executors can usually reach the modern RobloxGui container even when the
		-- legacy TopbarEnabled bridge is unavailable. Normal Studio clients simply
		-- fail this protected lookup and continue with the supported StarterGui APIs.
		pcall(function()
			local robloxGui = coreGuiService:FindFirstChild("RobloxGui")
			if robloxGui and robloxGui:IsA("ScreenGui") then
				Menu.robloxGui = robloxGui
				Menu.robloxGuiWasEnabled = robloxGui.Enabled
			end
		end)
	end

	local function setRobloxChromeHidden(hidden)
		captureCoreGuiState()
		if hidden then
			-- The modern mobile Chrome bar responds more reliably when All is disabled
			-- before individual CoreGui types and TopbarEnabled are applied.
			pcall(function()
				starterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, false)
			end)
		end
		for coreType, wasEnabled in pairs(Menu.coreGuiStates) do
			pcall(function()
				starterGui:SetCoreGuiEnabled(coreType, hidden and false or wasEnabled)
			end)
		end
		pcall(function()
			starterGui:SetCore("TopbarEnabled", hidden and false or Menu.topbarWasEnabled)
		end)
		if Menu.robloxGui then
			pcall(function()
				Menu.robloxGui.Enabled = hidden and false or Menu.robloxGuiWasEnabled
			end)
		end
	end

	local function setMenuEnvironment(active, instant)
		setRobloxChromeHidden(active)
		blurEffect.Enabled = true
		if instant then
			blurEffect.Size = active and Config.BlurSize or 0
		else
			tween(blurEffect, OPEN_TWEEN, { Size = active and Config.BlurSize or 0 })
		end

		if not active then
			task.delay(OPEN_TWEEN.Time + 0.03, function()
				if not Menu.destroyed and not Menu.open and blurEffect.Size <= 0.1 then
					blurEffect.Enabled = false
				end
			end)
		else
			-- Mobile Chrome can register after LocalScripts. Reapply during the short
			-- opening window, but only while this menu is still active.
			for _, retryDelay in ipairs({ 0.1, 0.35, 0.8, 1.4 }) do
				task.delay(retryDelay, function()
					if not Menu.destroyed and Menu.open then
						setRobloxChromeHidden(true)
					end
				end)
			end
		end
	end

	local function expApproach(current, target, speed, deltaTime)
		return current + (target - current) * (1 - math.exp(-speed * math.min(deltaTime, 0.1)))
	end

	-- Responsive sizing now controls world-space plane dimensions rather than flat GUI scale.
	function Menu:UpdateScale()

		end
		local camera = workspaceService.CurrentCamera
		local viewport = camera and camera.ViewportSize or Vector2.new(Config.ReferenceWidth, Config.ReferenceHeight)

	end

	local function placeSurface(part, canvasSize, centerX, centerY, yaw, pitch, camera, viewport)
		local depth = Config.SurfaceDepth
		local verticalWorldSize = 2 * depth * math.tan(math.rad(camera.FieldOfView * 0.5))
		local worldPerPixel = verticalWorldSize / math.max(viewport.Y, 1)
		-- Perspective expands the near edge; compressing physical width preserves
		-- the reference's readable on-screen footprint while retaining a true trapezoid.
		local width = canvasSize.X * Menu.scale * worldPerPixel * Config.SurfaceWidthCompression
		local height = canvasSize.Y * Menu.scale * worldPerPixel
		local localX = (centerX - viewport.X * 0.5) * worldPerPixel
		local localY = -(centerY - viewport.Y * 0.5) * worldPerPixel

		part.Size = Vector3.new(width, height, 0.04)
		part.CFrame = camera.CFrame
			* CFrame.new(localX, localY, -depth)
			* CFrame.Angles(math.rad(pitch), math.rad(yaw), 0)
	end

	local KEY_INPUT_CANVAS_BOUNDS = {
		left = 50,
		top = 60,
		width = 335,
		height = 54,
	}

	local function surfaceCanvasPointToViewport(part, canvasSize, x, y, camera)
		local localPoint = Vector3.new(
			(x / canvasSize.X - 0.5) * part.Size.X,
			(0.5 - y / canvasSize.Y) * part.Size.Y,
			part.Size.Z * 0.5
		)
		local worldPoint = part.CFrame:PointToWorldSpace(localPoint)
		local viewportPoint, onScreen = camera:WorldToViewportPoint(worldPoint)
		return Vector2.new(viewportPoint.X, viewportPoint.Y), onScreen and viewportPoint.Z > 0
	end

	local function triangleSign(point, first, second)
		return (point.X - second.X) * (first.Y - second.Y) - (first.X - second.X) * (point.Y - second.Y)
	end

	local function pointInTriangle(point, first, second, third)
		local firstSign = triangleSign(point, first, second)
		local secondSign = triangleSign(point, second, third)
		local thirdSign = triangleSign(point, third, first)
		local hasNegative = firstSign < 0 or secondSign < 0 or thirdSign < 0
		local hasPositive = firstSign > 0 or secondSign > 0 or thirdSign > 0
		return not (hasNegative and hasPositive)
	end

	local function pointInsideSurfaceRoot(point, part, canvasSize, left, top, width, height, camera)
		local topLeft, topLeftVisible = surfaceCanvasPointToViewport(part, canvasSize, left, top, camera)
		local topRight, topRightVisible = surfaceCanvasPointToViewport(part, canvasSize, left + width, top, camera)
		local bottomLeft, bottomLeftVisible = surfaceCanvasPointToViewport(part, canvasSize, left, top + height, camera)
		local bottomRight, bottomRightVisible = surfaceCanvasPointToViewport(part, canvasSize, left + width, top + height, camera)
		if not (topLeftVisible or topRightVisible or bottomLeftVisible or bottomRightVisible) then
			return false
		end
		return pointInTriangle(point, topLeft, topRight, bottomRight)
			or pointInTriangle(point, topLeft, bottomRight, bottomLeft)
	end

	local function pointInsideMenu(point)
		if not Menu.open then
			return false
		end
		local camera = workspaceService.CurrentCamera
		if not camera then
			return false
		end
		return pointInsideSurfaceRoot(point, leftPart, LEFT_CANVAS_SIZE, 50, 12, 335, 452, camera)
			or pointInsideSurfaceRoot(point, rightPart, RIGHT_CANVAS_SIZE, 14, 8, 323, 313, camera)
	end

	-- No separate orientation square is needed: a drag that begins directly on
	-- either menu plane rotates both planes. A small movement threshold preserves
	-- normal taps on Home, Announcements, Updates, Key System, and the footer.
	connect(userInputService.InputBegan, function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		local keyInput = Menu.references.KeySystemInput
		if keyInput and keyInput.Visible then
			return
		end
		local position = Vector2.new(input.Position.X, input.Position.Y)
		if not pointInsideMenu(position) then
			return
		end
		Menu.orientationDragging = true
		Menu.orientationDragInput = input
		Menu.orientationDragStart = position
		Menu.orientationDragOriginYaw = Menu.targetOrientationYaw
		Menu.orientationDragOriginPitch = Menu.targetOrientationPitch
		Menu.orientationDragDistance = 0
	end)

	connect(userInputService.InputChanged, function(input)
		if not Menu.orientationDragging then
			return
		end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if Menu.orientationDragInput.UserInputType == Enum.UserInputType.Touch and input ~= Menu.orientationDragInput then
			return
		end
		local position = Vector2.new(input.Position.X, input.Position.Y)
		local totalDelta = position - Menu.orientationDragStart
		Menu.orientationDragDistance = totalDelta.Magnitude
		if Menu.orientationDragDistance >= 4 then
			Menu:SetOrientation(
				Menu.orientationDragOriginYaw + totalDelta.X * 0.22,
				Menu.orientationDragOriginPitch - totalDelta.Y * 0.18
			)
		end
	end)

	connect(userInputService.InputEnded, function(input)
		if not Menu.orientationDragging then
			return
		end
		local activeInput = Menu.orientationDragInput
		local matchingMouse = activeInput.UserInputType == Enum.UserInputType.MouseButton1
			and input.UserInputType == Enum.UserInputType.MouseButton1
		local matchingTouch = activeInput.UserInputType == Enum.UserInputType.Touch and input == activeInput
		if not matchingMouse and not matchingTouch then
			return
		end
		Menu.orientationDragging = false
		Menu.orientationDragInput = nil
	end)

	local function updateKeyInputOverlay(camera)
		local keyInput = Menu.references.KeySystemInput
		if not keyInput or not keyInput.Visible then
			return
		end

		local bounds = KEY_INPUT_CANVAS_BOUNDS
		local topLeft, topLeftVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left, bounds.top, camera)
		local topRight, topRightVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left + bounds.width, bounds.top, camera)
		local bottomLeft, bottomLeftVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left, bounds.top + bounds.height, camera)
		local bottomRight, bottomRightVisible = surfaceCanvasPointToViewport(
			leftPart,
			LEFT_CANVAS_SIZE,
			bounds.left + bounds.width,
			bounds.top + bounds.height,
			camera
		)
		if not (topLeftVisible or topRightVisible or bottomLeftVisible or bottomRightVisible) then
			return
		end

		local centerX = (topLeft.X + topRight.X + bottomLeft.X + bottomRight.X) * 0.25
		local centerY = (topLeft.Y + topRight.Y + bottomLeft.Y + bottomRight.Y) * 0.25
		local projectedWidth = ((topRight - topLeft).Magnitude + (bottomRight - bottomLeft).Magnitude) * 0.5
		local projectedHeight = ((bottomLeft - topLeft).Magnitude + (bottomRight - topRight).Magnitude) * 0.5
		local edgeCenterLeft = (topLeft + bottomLeft) * 0.5
		local edgeCenterRight = (topRight + bottomRight) * 0.5
		local edge = edgeCenterRight - edgeCenterLeft
		local rotation = math.deg(math.atan2(edge.Y, edge.X))
		-- Back-face canvas X can project in reverse; normalize so mobile text never
		-- flips upside down while still following the bar's visible slope.
		if rotation > 90 then
			rotation = rotation - 180
		elseif rotation < -90 then
			rotation = rotation + 180
		end

		keyInput.Position = UDim2.fromOffset(centerX, centerY)
		keyInput.Size = UDim2.fromOffset(math.max(120, projectedWidth - 26), math.max(34, projectedHeight - 4))
		keyInput.Rotation = rotation
	end

	local function updateWorldPresentation(deltaTime)
		if Menu.destroyed then
			return
		end
		local camera = workspaceService.CurrentCamera
		if not camera then
			return
		end
		local viewport = camera.ViewportSize
		if viewport.X <= 0 or viewport.Y <= 0 then
			return
		end
		Menu.scale = math.clamp(math.min(viewport.X / Config.ReferenceWidth, viewport.Y / Config.ReferenceHeight), 0.62, 1.35)

		Menu.openAmount = expApproach(Menu.openAmount, Menu.targetOpenAmount, 10.5, deltaTime)
		Menu.orientationYaw = expApproach(Menu.orientationYaw, Menu.targetOrientationYaw, 9, deltaTime)
		Menu.orientationPitch = expApproach(Menu.orientationPitch, Menu.targetOrientationPitch, 9, deltaTime)

		if math.abs(Menu.openAmount - Menu.targetOpenAmount) < 0.001 then
			Menu.openAmount = Menu.targetOpenAmount
		end
		local reveal = 1 - ((1 - Menu.openAmount) ^ 3)
		local leftClosedX = -LEFT_CANVAS_SIZE.X * Menu.scale * 0.58
		local leftOpenX = 210 * Menu.scale
		local rightClosedX = viewport.X + RIGHT_CANVAS_SIZE.X * Menu.scale * 0.58
		local rightOpenX = viewport.X - 193 * Menu.scale
		local leftCenterX = leftClosedX + (leftOpenX - leftClosedX) * reveal
		local rightCenterX = rightClosedX + (rightOpenX - rightClosedX) * reveal
		-- Align the visible root tops even though each SurfaceGui has a different
		-- canvas height and internal top padding.
		local sharedRootTop = viewport.Y * 0.245
		local leftCenterY = sharedRootTop + (LEFT_CANVAS_SIZE.Y * 0.5 - 12) * Menu.scale
		local rightCenterY = sharedRootTop + (RIGHT_CANVAS_SIZE.Y * 0.5 - 8) * Menu.scale

		placeSurface(
			leftPart,
			LEFT_CANVAS_SIZE,
			leftCenterX,
			leftCenterY,
			Config.SurfaceInwardYaw + Menu.orientationYaw,
			Menu.orientationPitch,
			camera,
			viewport
		)
		placeSurface(
			rightPart,
			RIGHT_CANVAS_SIZE,
			rightCenterX,
			rightCenterY,
			-Config.SurfaceInwardYaw + Menu.orientationYaw,
			Menu.orientationPitch,
			camera,
			viewport
		)
		updateKeyInputOverlay(camera)

		local shouldRender = Menu.openAmount > 0.005 or Menu.targetOpenAmount > 0
		leftSurface.Enabled = shouldRender
		rightSurface.Enabled = shouldRender
	end

	connect(runService.RenderStepped, updateWorldPresentation)

		if self.destroyed then
			return
		end

		shouldOpen = shouldOpen == true
		self.open = shouldOpen
		self.targetOpenAmount = shouldOpen and 1 or 0

		if shouldOpen then
			leftSurface.Enabled = true
			rightSurface.Enabled = true
		else
			self.orientationDragging = false
			self.orientationDragInput = nil
		end
		if not shouldOpen and self.references.KeySystemInput and self.references.KeySystemInput.Visible then

			self.references.KeySystemInput.Visible = false
			self.references.KeySystemButton.Visible = true
			self.references.KeySystemLabel.Visible = true
		end

		if instant then
			self.openAmount = self.targetOpenAmount
			updateWorldPresentation(0)
		end
		setMenuEnvironment(shouldOpen, instant == true)
	end

	local function sanitizeDescription(value)
		if type(value) ~= "string" or value == "" then
			return Config.DefaultDescription
		end
		local sanitized = string.gsub(value, "[%c]+", " ")
		sanitized = string.gsub(sanitized, "%s+", " ")

		-- Truncate by UTF-8 character boundary so an emoji or accented title can
		-- never leave invalid text in the Roblox label.
		local characterCount = utf8.len(sanitized)
		if characterCount and characterCount > 175 then
			local byteAfterLimit = utf8.offset(sanitized, 173)
			if byteAfterLimit then
				sanitized = string.sub(sanitized, 1, byteAfterLimit - 1) .. "..."
			end
		end
		return sanitized
	end

	local function formatRemaining(seconds)
		seconds = math.max(0, math.floor(seconds + 0.5))
		if seconds <= 0 then
			return "Expired"
		end
		if seconds >= Config.StandardDurationSeconds then
			return "12 Hours - Remaining"
		end
		local hours = math.floor(seconds / 3600)
		local minutes = math.floor((seconds % 3600) / 60)
		local remainingSeconds = seconds % 60
		return string.format("%02d:%02d:%02d - Remaining", hours, minutes, remainingSeconds)
	end

	local function formatPlayed(seconds)
		seconds = math.max(0, math.floor(seconds))
		local days = math.floor(seconds / 86400)
		local hours = math.floor((seconds % 86400) / 3600)
		local minutes = math.floor((seconds % 3600) / 60)
		return string.format("%dd %dh %dm", days, hours, minutes)
	end

	local function updateTimers()
		local now = getServerTime()
		Menu.references.CountdownValue.Text = (Menu.keyAccepted and Menu.standardExpiresAt) and formatRemaining(Menu.standardExpiresAt - os.time()) or "Pending..."
		Menu.references.TimePlayedValue.Text = formatPlayed(now - startedAt)
	end

	connect(runService.Heartbeat, function(deltaTime)
		timerAccumulator = timerAccumulator + deltaTime
		if timerAccumulator >= 1 then
			timerAccumulator = timerAccumulator % 1
			updateTimers()
		end
	end)
	updateTimers()

	-- RightShift always toggles the complete presentation, even while the key box is focused.
	connect(userInputService.InputBegan, function(input, gameProcessed)
		if input.KeyCode == Enum.KeyCode.RightShift then
			Menu:Toggle()
			return
		end
		if gameProcessed or userInputService:GetFocusedTextBox() then
			return
		end

		if input.KeyCode == Enum.KeyCode.Z then
			if not Menu.open then
				Menu:SetOpen(true)
			end
			Menu:OpenHome()
		elseif input.KeyCode == Enum.KeyCode.F then
			Menu:OpenAnnouncements()
		elseif input.KeyCode == Enum.KeyCode.X then
			Menu:PulseHotkey("X")
			Menu:CopyKeyLink()
			local callback = Menu.callbacks.KeyLink or Menu.callbacks.X
			if type(callback) == "function" then
				task.spawn(callback, Menu)
			end
		elseif input.KeyCode == Enum.KeyCode.C then
			Menu:OpenDiscord()
		end
	end)

	local function bindViewport()
		local camera = workspaceService.CurrentCamera
		if camera then
			connect(camera:GetPropertyChangedSignal("ViewportSize"), function()
				Menu:UpdateScale()
			end)
		end
	end

	connect(workspaceService:GetPropertyChangedSignal("CurrentCamera"), function()
		Menu:UpdateScale()
	end)

	function Menu:Destroy()
		if self.destroyed then
			return
		end

		if self.references.KeySystemInput and self.references.KeySystemInput:IsFocused() then
			self.references.KeySystemInput:ReleaseFocus()
		end
		if self.coreGuiCaptured then
			setRobloxChromeHidden(false)
		end
		self.destroyed = true

		for _, connection in ipairs(self.connections) do
			if connection and connection.Connected then
				connection:Disconnect()
			end
		end
		table.clear(self.connections)

		for _, referenceName in ipairs({ "ScreenGui", "LeftSurface", "RightSurface", "SurfaceFolder", "BlurEffect" }) do
			local object = self.references[referenceName]
			if object and object.Parent then
				object:Destroy()
			end
		end

		if sharedEnvironment.ZNTHMainMenu == self then
			sharedEnvironment.ZNTHMainMenu = nil
		end
	end

	bindViewport()
	Menu:UpdateScale()
	bindCurrentGame()
	bindLocalPlayer()
	sharedEnvironment.ZNTHMainMenu = Menu

	-- Start just off-screen, then reproduce the two-sided entrance from the reference.
	Menu:SetOpen(false, true)
	task.defer(function()
		if not Menu.destroyed then
			Menu:SetOpen(true)
		end
	end)

	return Menu
end

	local function getSharedEnvironment()
		local environment = _G
		local ok, executorEnvironment = pcall(function()
			if type(getgenv) == "function" then
				return getgenv()
			end
			return nil
		end)
		if ok and type(executorEnvironment) == "table" then
			environment = executorEnvironment
		end
		return environment
	end

	local function create(className, properties, parent)
		local object = Instance.new(className)
		for property, value in pairs(properties or {}) do
			object[property] = value
		end
		if parent then
			object.Parent = parent
		end
		return object
	end

	local function connect(signal, callback)
		local connection = signal:Connect(callback)
		table.insert(WordHelper.connections, connection)
		return connection
	end

		end

	local function normalize(text)
		text = string.lower(tostring(text or ""))
		text = string.gsub(text, "^%s+", "")
		text = string.gsub(text, "%s+$", "")
		return text
	end

	local function normalizeLetters(text)
		return string.gsub(normalize(text), "[^a-z]", "")
	end

	local function escapeRichText(text)
		text = string.gsub(text, "&", "&amp;")
		text = string.gsub(text, "<", "&lt;")
		text = string.gsub(text, ">", "&gt;")
		text = string.gsub(text, '"', "&quot;")
		return text
	end

	local function extractCurrentFragment(text, cursorPosition)
		text = string.lower(tostring(text or ""))
		cursorPosition = tonumber(cursorPosition)
		if cursorPosition and cursorPosition > 0 then
			text = string.sub(text, 1, math.min(#text, cursorPosition - 1))
		end
		return string.match(text, "([a-z]+)$") or ""
	end

	local WORD_BASE64_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
	local WORD_BASE64_LOOKUP = {}
	for index = 1, #WORD_BASE64_ALPHABET do
		WORD_BASE64_LOOKUP[string.sub(WORD_BASE64_ALPHABET, index, index)] = index - 1
	end

	local function decodeWordAsset(data)
		local output = table.create(math.floor(#data * 0.75))
		local outputLength, accumulator, groupLength = 0, 0, 0
		local function appendByte(value)
			outputLength = outputLength + 1
			output[outputLength] = string.char(value)
		end
		for index = 1, #data do
			local value = WORD_BASE64_LOOKUP[string.sub(data, index, index)]
			if value then
				accumulator = accumulator * 64 + value
				groupLength = groupLength + 1
				if groupLength == 4 then
					appendByte(math.floor(accumulator / 65536) % 256)
					appendByte(math.floor(accumulator / 256) % 256)
					appendByte(accumulator % 256)
					accumulator, groupLength = 0, 0
				end
			end
		end
		if groupLength == 2 then
			appendByte(math.floor(accumulator / 16) % 256)
		elseif groupLength == 3 then
			appendByte(math.floor(accumulator / 1024) % 256)
			appendByte(math.floor(accumulator / 4) % 256)
		end
		return table.concat(output, "", 1, outputLength)
	end

	local function normalizeWordAsset(value)
		local text = tostring(value or "")
		if text == "" or text == "0" then return "" end
		if string.find(text, "://", 1, true) then return text end
		if string.match(text, "^%d+$") then return "rbxassetid://" .. text end
		return text
	end

	local function resolveWordAsset(key, fileName)
		local configured = normalizeWordAsset(Config.Assets[key])
		if configured ~= "" then return configured end
		if not writeFile or not getCustomAsset then return "" end
		local path = "Zenith/" .. fileName
		local wrote = pcall(function()
			if makeFolder then pcall(makeFolder, "Zenith") end
			writeFile(path, decodeWordAsset(EMBEDDED_WORD_ASSETS[key]))
		end)
		if not wrote then return "" end
		local ok, asset = pcall(getCustomAsset, path)
		return ok and type(asset) == "string" and asset or ""
	end

	local function parseDictionary(content)
		local output = {}
		local seen = {}
		local sorted = true
		local previousWord = nil
		local processed = 0
		for rawWord in string.gmatch(tostring(content or ""), "[^\r\n]+") do
			processed = processed + 1
			if processed % 5000 == 0 then
				if WordHelper.destroyed then return {} end
				task.wait()
			end
			local word = normalize(rawWord)
			if #word >= 2 and #word <= 32 and string.match(word, "^[a-z]+$") and not seen[word] then
				seen[word] = true
				if previousWord and word < previousWord then
					sorted = false
				end
				previousWord = word
				table.insert(output, word)
			end
		end
		if not sorted then
			table.sort(output)
		end
		return output
	end

	local function buildBuckets(words)
		local firstLetterBuckets = {}
		local prefixBuckets = {}
		for index, word in ipairs(words) do
			if index % 5000 == 0 then
				if WordHelper.destroyed then return {}, {} end
				task.wait()
			end
			local first = string.sub(word, 1, 1)
			local firstBucket = firstLetterBuckets[first]
			if not firstBucket then
				firstBucket = {}
				firstLetterBuckets[first] = firstBucket
			end
			table.insert(firstBucket, word)

			if #word >= 2 then
				local prefix = string.sub(word, 1, 2)
				local prefixBucket = prefixBuckets[prefix]
				if not prefixBucket then
					prefixBucket = {}
					prefixBuckets[prefix] = prefixBucket
				end
				table.insert(prefixBucket, word)
			end
		end
		return firstLetterBuckets, prefixBuckets
	end

	local function installCleanWords(words)
		local merged, seen = {}, {}
		for _, word in ipairs(words) do
			if not seen[word] then
				seen[word] = true
				table.insert(merged, word)
			end
		end
		for _, word in ipairs(WordHelper.derivedVocabulary) do
			if not seen[word] then
				seen[word] = true
				table.insert(merged, word)
			end
		end
		table.sort(merged)
		WordHelper.baseWords = words
		WordHelper.words = merged
		WordHelper.firstLetterBuckets, WordHelper.prefixBuckets = buildBuckets(merged)
		local wordSet = {}
		for index, word in ipairs(merged) do
			if index % 5000 == 0 then
				if WordHelper.destroyed then return end
				task.wait()
			end
			wordSet[word] = true
		end
		WordHelper.wordSet = wordSet
		for word in pairs(WordHelper.pendingUsedWords) do
			if wordSet[word] then WordHelper.pendingUsedWords[word] = nil end
		end
	end

	local function formatWord(word, query)
		local startIndex = (query ~= "" and string.find(word, query, 1, true)) or 1
		local before = escapeRichText(string.sub(word, 1, startIndex - 1))
		local match = escapeRichText(string.sub(word, startIndex, startIndex + #query - 1))
		local after = escapeRichText(string.sub(word, startIndex + #query))
		return '<font color="#B5B9BF">' .. before .. '</font><font color="#F2F3F5"><b>' .. match
			.. '</b></font><font color="#B5B9BF">' .. after .. "</font>"
	end

	local function decodeHtml(text)
		text = string.gsub(text, "<.->", "")
		text = string.gsub(text, "&nbsp;", " ")
		text = string.gsub(text, "&quot;", '"')
		text = string.gsub(text, "&#39;", "'")
		text = string.gsub(text, "&lt;", "<")
		text = string.gsub(text, "&gt;", ">")
		text = string.gsub(text, "&amp;", "&")
		text = string.gsub(text, "%s+", " ")
		text = string.gsub(text, "^%s+", "")
		text = string.gsub(text, "[%s:]+$", "")
		return text
	end

	local function fetchDefinition(word)
		local page = httpGet(Config.CambridgeUrl .. word)
		if page then
			local raw = string.match(page, 'class="def ddef_d db">(.-)</div>')
			if raw then
				local definition = decodeHtml(raw)
				if definition ~= "" then
					return definition, "Cambridge"
				end
			end
		end
		local json = httpGet(Config.BackupDefinitionUrl .. word)
		if json then
			local ok, data = pcall(function()
				return game:GetService("HttpService"):JSONDecode(json)
			end)
			local entry = ok and type(data) == "table" and data[1]
			local meaning = entry and entry.meanings and entry.meanings[1]
			local first = meaning and meaning.definitions and meaning.definitions[1]
			if first and type(first.definition) == "string" then
				return first.definition, "Free Dictionary"
			end
		end
		return nil, nil
	end

	local function isGameTextBox(textBox)
		if not textBox or textBox == searchBox then return false end
		local valid = false
		pcall(function() valid = textBox:IsA("TextBox") end)
		if not valid then return false end
		local isCoreChat = false
		if coreGui then pcall(function() isCoreChat = textBox:IsDescendantOf(coreGui) end) end
		if isCoreChat then return false end
		local current = textBox
		while current do
			local name = string.lower(tostring(current.Name or ""))
			if name == "chat" or name == "experiencechat" then return false end
			if current == playerGui then break end
			current = current.Parent
		end
		return true
	end

	-- A focused game TextBox is the safest universal target. Build a paced edit
	-- plan instead of replacing the whole value instantly so the WPM setting also
	-- applies to native game inputs.
	local function prepareExternalTextBox(word)
		if isRobloxMenuOpen() then return nil end
		local textBox = nil
		pcall(function() textBox = userInputService:GetFocusedTextBox() end)
		if textBox == searchBox then textBox = nil end
		if not textBox then textBox = WordHelper.activeExternalTextBox end
		if not textBox and WordHelper.detectionSource == "external" then
			textBox = WordHelper.lastExternalTextBox
		end
		if not textBox or not textBox.Parent or not isGameTextBox(textBox) then return nil end

		local ok, text = pcall(function() return tostring(textBox.Text or "") end)
		if not ok then return nil end
		local fragment = string.lower(string.match(text, "([%a]+)$") or "")
		local baseText = string.sub(text, 1, #text - #fragment)
		local common = 0
		while common < #fragment and common < #word
			and string.sub(fragment, common + 1, common + 1) == string.sub(word, common + 1, common + 1) do
			common = common + 1
		end
		local actions = {}
		for _ = #fragment, common + 1, -1 do table.insert(actions, "back") end
		for index = common + 1, #word do table.insert(actions, string.sub(word, index, index)) end
		local buffer, expectedText = fragment, text
		local function sender(action)
			local currentOk, currentText = pcall(function() return tostring(textBox.Text or "") end)
			if not currentOk or currentText ~= expectedText then return false end
			if action == "back" then
				buffer = string.sub(buffer, 1, math.max(0, #buffer - 1))
			else
				buffer = string.sub(buffer .. action, -32)
			end
			local completed = baseText .. buffer
			local wrote = pcall(function()
				textBox.Text = completed
				textBox.CursorPosition = #completed + 1
			end)
			if wrote then expectedText = completed end
			return wrote
		end
		local function validator()
			if not textBox.Parent or not isGameTextBox(textBox) then return false end
			local focused = userInputService:GetFocusedTextBox()
			return focused == nil or focused == textBox
		end
		return actions, sender, validator, textBox
	end

	local typeWordIntoGame

	-- Single activation types/completes. Clipboard access is intentionally kept
	-- out of this method; a second activation handles copying below.
	function WordHelper:SelectWord(word)
		if self.destroyed or isRobloxMenuOpen() then return false end
		word = normalizeLetters(word)
		if word == "" then return false end
		if searchBox:IsFocused() then searchBox:ReleaseFocus() end
		self.selectedWord = word
		styleRows()
		showDefinition(word)

		local started, method = false, nil
		if Settings.autoType and type(typeWordIntoGame) == "function" then
			started, method = typeWordIntoGame(word)
		end
		local completedImmediately = method == "complete"
		if not Settings.autoType then
			setStatus("Seleccionado: " .. word, Config.Secondary)
		else
			setStatus(started and ((completedImmediately and "Escrito: " or "Escribiendo: ") .. word)
				or ("No pude escribir automáticamente: " .. word), started and Config.Secondary or Config.Muted)
		end

		local callback = self.callbacks.WordSelected
		-- Keep the historical third callback argument as "copied" (always false on
		-- a first activation); append the automatic-typing result for integrations.
		if type(callback) == "function" then task.spawn(callback, self, word, false, started, method) end
		return started
	end

	function WordHelper:ActivateWord(word)
		if self.destroyed or isRobloxMenuOpen() then return false end
		word = normalizeLetters(word)
		if word == "" then return false end
		local now = os.clock()
		if self.lastWordActivation == word and now - self.lastWordActivationAt <= Config.DoubleActivationSeconds then
			self.lastWordActivation = nil
			self.lastWordActivationAt = 0
			local copied = clipboardWriter and pcall(clipboardWriter, word) or false
			setStatus(copied and ("Copiado: " .. word) or ("No se pudo copiar: " .. word),
				copied and Config.Secondary or Config.Muted)
			local callback = self.callbacks.WordCopied
			if type(callback) == "function" then task.spawn(callback, self, word, copied) end
			return copied
		end
		self.lastWordActivation = word
		self.lastWordActivationAt = now
		return self:SelectWord(word)
	end

	local function refreshWords(expectedVersion)
		if WordHelper.destroyed or (expectedVersion and expectedVersion ~= WordHelper.refreshVersion) then return end
		clearRows()

		local query = WordHelper.currentQuery
		if query == "" then
			emptyLabel.Visible = true
			emptyLabel.Text = "Escribe una letra en el chat o juego"
			wordList.CanvasPosition = Vector2.new(0, 0)
			setStatus("Esperando texto...", Config.Muted)
			WordHelper.bestWord = nil
			updatePreview()
			return
		end

		local containsMode = WordHelper.searchMode == "contains"
		local used = WordHelper.usedWords
		local matches = {}
		if containsMode then
			-- "Contiene": word-battle style games give a syllable that can be anywhere.
			for _, word in ipairs(WordHelper.words) do
				if (Settings.derivations or not WordHelper.derivedWordSet[word])
					and (not Settings.hideUsed or not used[word]) and string.find(word, query, 1, true) then
					table.insert(matches, word)
				end
			end
		else
			local source
			if #query == 1 then
				source = WordHelper.firstLetterBuckets[query] or {}
			else
				source = WordHelper.prefixBuckets[string.sub(query, 1, 2)] or {}
			end
			for _, word in ipairs(source) do
				if (Settings.derivations or not WordHelper.derivedWordSet[word])
					and (not Settings.hideUsed or not used[word]) and string.sub(word, 1, #query) == query then
					table.insert(matches, word)
				end
			end
		end
		local totalMatches = #matches

		-- Sort: Comunes = most frequent first; Largas = longest first; Cortas = shortest first.
		local rank = WordHelper.frequencyRank
		local sortMode = WordHelper.sortMode
		table.sort(matches, function(a, b)
			if sortMode == "long" and #a ~= #b then
				return #a > #b
			elseif sortMode == "short" and #a ~= #b then
				return #a < #b
			end
			local rankA, rankB = rank[a] or math.huge, rank[b] or math.huge
			if rankA ~= rankB then
				return rankA < rankB
			end
			if #a ~= #b then
				return #a < #b
			end
			return a < b
		end)

		-- Preview word: the shortest common word that is clearly longer than the
		-- prefix ("s" -> short word, "sim" -> "simple"), best frequency on ties.
		local best, bestLength, bestRank = nil, math.huge, math.huge
		for _, word in ipairs(matches) do
			if #word >= #query + 2 then
				local wordRank = rank[word] or math.huge
				local common = wordRank ~= math.huge
				local bestCommon = bestRank ~= math.huge
				if (common and not bestCommon)
					or (common == bestCommon and (#word < bestLength or (#word == bestLength and wordRank < bestRank))) then
					best, bestLength, bestRank = word, #word, wordRank
				end
			end
		end
		WordHelper.bestWord = best or matches[1]
		updatePreview()

		local cappedLimit = isTouchDevice and Config.MobileMaximumResults or Config.MaximumResults
		local resultLimit = Settings.allResults and totalMatches or cappedLimit
		WordHelper.resultLimit = resultLimit
		local shown = math.min(totalMatches, resultLimit)
		local batchSize = isTouchDevice and 20 or 40
		for index = 1, shown do
			createWordRow(matches[index], query, index)
			if Settings.allResults and index % batchSize == 0 then
				task.wait()
				if WordHelper.destroyed or (expectedVersion and expectedVersion ~= WordHelper.refreshVersion) then return end
			end
		end

		emptyLabel.Visible = shown == 0
		emptyLabel.Text = "No se encontraron palabras"
		wordList.CanvasPosition = Vector2.new(0, 0)
		if shown == 0 then
			setStatus('Sin resultados para "' .. query .. '"', Config.Muted)
		elseif totalMatches > shown then
			setStatus(string.format("%d de %d resultados", shown, totalMatches), Config.Muted)
		else
			setStatus(string.format("%d resultado%s", shown, shown == 1 and "" or "s"), Config.Muted)
		end
	end

	local function scheduleRefresh()
		WordHelper.refreshVersion = WordHelper.refreshVersion + 1
		local version = WordHelper.refreshVersion
		task.delay(isTouchDevice and 0.075 or 0.045, function()
			if not WordHelper.destroyed and version == WordHelper.refreshVersion then
				refreshWords(version)
			end
		end)
	end

	local synchronizingSearch = false
	local function applyDetectedQuery(query, source, reflectInSearch)
		if WordHelper.destroyed then return end
		if not Settings.autoDetect and source ~= "manual" then return end
		query = normalizeLetters(query)
		local queryChanged = query ~= WordHelper.currentQuery
		WordHelper.currentQuery = query
		WordHelper.detectionSource = source or "automatic"
		local searchFocused = false
		pcall(function() searchFocused = searchBox:IsFocused() end)
		if reflectInSearch and not searchFocused and searchBox.Text ~= query then
			synchronizingSearch = true
			searchBox.Text = query
			synchronizingSearch = false
		end
		if queryChanged and WordHelper.selectedWord then
			WordHelper.definitionVersion = WordHelper.definitionVersion + 1
			WordHelper.selectedWord = nil
			definitionWord.Text = "Definición"
			definitionText.Text = "Toca una palabra para ver su significado."
			definitionText.TextColor3 = Config.Muted
		end
		if query == "" then
			detectionLabel.Text = "Detección automática activa"
		else
			local SOURCE_LABELS = {
				manual = "manual",
				external = "chat / caja de texto",
				passive = "teclado",
				remote = "teclado del juego",
				visual = "interfaz del juego",
				turn = "tu turno",
			}
			local sourceLabel = SOURCE_LABELS[source] or "detectado automáticamente"
			detectionLabel.Text = "Texto: " .. query .. "  |  " .. sourceLabel
		end
		if queryChanged then scheduleRefresh() end
	end

	local function applySettingChange(key)
		updateSettingsControls()
		if key == "autoType" and not Settings.autoType then
			cancelAutomaticTyping(true)
			setStatus("Auto Type desactivado", Config.Muted)
		elseif key == "definitions" then
			updateDefinitionLayout()
		elseif key == "allResults" or key == "hideUsed" or key == "derivations" then
			scheduleRefresh()
		elseif key == "autoDetect" then
			cancelAutomaticTyping(true)
			if Settings.autoDetect then
				applyDetectedQuery(WordHelper.currentQuery, WordHelper.detectionSource, false)
			else
				WordHelper.passiveBuffer = ""
				WordHelper.lastVisualWord = ""
				WordHelper.lastLetterState = ""
				WordHelper.lastLetterVisualCandidate = ""
				WordHelper.lastLetterVisualCandidateCount = 0
				WordHelper.lastLetterPacketAt = 0
				detectionLabel.Text = "Automatic detection disabled"
			end
		end
	end

	function WordHelper:SetSetting(key, value)
		if self.destroyed or DEFAULT_SETTINGS[key] == nil then return false end
		if key == "wpm" then return self:SetWPM(value) end
		if key == "legitMinWPM" or key == "legitMaxWPM" then return self:SetLegitWPM(key, value) end
		if type(DEFAULT_SETTINGS[key]) ~= "boolean" then return false end
		Settings[key] = value == true
		applySettingChange(key)
		return true
	end

	function WordHelper:SetWPM(value)
		if self.destroyed then return false end
		value = tonumber(value)
		if not value then return false end
		Settings.wpm = math.clamp(
			math.floor(value / Config.WPMIncrement + 0.5) * Config.WPMIncrement,
			Config.MinimumWPM,
			Config.MaximumWPM
		)
		updateSettingsControls()
		return true
	end

	function WordHelper:SetLegitWPM(key, value)
		if self.destroyed or (key ~= "legitMinWPM" and key ~= "legitMaxWPM") then return false end
		value = tonumber(value)
		if not value then return false end
		value = math.clamp(
			math.floor(value / Config.LegitWPMIncrement + 0.5) * Config.LegitWPMIncrement,
			Config.MinimumWPM,
			Config.MaximumWPM
		)
		Settings[key] = value
		if key == "legitMinWPM" and Settings.legitMaxWPM < value then
			Settings.legitMaxWPM = value
		elseif key == "legitMaxWPM" and Settings.legitMinWPM > value then
			Settings.legitMinWPM = value
		end
		updateSettingsControls()
		return true
	end

	local function queryForTypedBuffer(typed)
		typed = normalizeLetters(typed)
		local prompt = WordHelper.turnPrompt
		if WordHelper.localTurn == true and prompt ~= "" then
			-- The confirmed game pre-fills the required letters and transmits only
			-- what the local player adds. Build "s" + "o" as "so" exactly once.
			return string.sub(prompt .. typed, 1, 32)
		end
		return typed
	end

	function WordHelper:SetWords(words)
		if self.destroyed or type(words) ~= "table" then
			return
		end
		local clean = {}
		local seen = {}
		for _, value in ipairs(words) do
			local word = normalize(value)
			if #word >= 2 and #word <= 32 and string.match(word, "^[a-z]+$") and not seen[word] then
				seen[word] = true
				table.insert(clean, word)
			end
		end
		table.sort(clean)
		installCleanWords(clean)
		scheduleRefresh()
	end

	local function disconnectFocusedTextBox()
		for _, connection in ipairs(WordHelper.focusConnections) do
			if connection and connection.Connected then
				connection:Disconnect()
			end
		end
		table.clear(WordHelper.focusConnections)

	local function syncFromFocusedTextBox(textBox)
		if WordHelper.destroyed or textBox ~= WordHelper.activeExternalTextBox
			or (WordHelper.isAutoTyping and textBox == WordHelper.autoTypeTextBox) then
			return
		end
		local ok, text, cursorPosition = pcall(function()
			return textBox.Text, textBox.CursorPosition
		end)
		if ok then
			WordHelper:DetectText(text, cursorPosition)
		end
	end

	local function bindFocusedTextBox(textBox)
		if not Settings.autoDetect or isRobloxMenuOpen() or WordHelper.isAutoTyping then return end
		-- Last Letter is driven by Overbar.Frame.Keyboard and CurrentWord, never by
		-- Roblox chat or an arbitrary focused menu TextBox.
		if currentPlaceId == LAST_LETTER_PLACE_ID and textBox ~= searchBox then
			disconnectFocusedTextBox()
			return
		end
		if textBox == searchBox then
			disconnectFocusedTextBox()
			return
		end
		if textBox == WordHelper.activeExternalTextBox then
			return
		end
		disconnectFocusedTextBox()
		if not textBox then
			return
		end
		if not isGameTextBox(textBox) then return end

		WordHelper.activeExternalTextBox = textBox
		WordHelper.lastExternalTextBox = textBox
		local ok, textConnection = pcall(function()
			return textBox:GetPropertyChangedSignal("Text"):Connect(function()
				syncFromFocusedTextBox(textBox)
			end)
		end)
		if ok and textConnection then
			table.insert(WordHelper.focusConnections, textConnection)
		end
		local cursorOk, cursorConnection = pcall(function()
			return textBox:GetPropertyChangedSignal("CursorPosition"):Connect(function()
				syncFromFocusedTextBox(textBox)
			end)
		end)
		if cursorOk and cursorConnection then
			table.insert(WordHelper.focusConnections, cursorConnection)
		end
		local focusOk, focusConnection = pcall(function()
			return textBox.FocusLost:Connect(function(enterPressed)
				if enterPressed and markUsed then
					local okText, text = pcall(function() return textBox.Text end)
					if okText then markUsed(extractCurrentFragment(text, #tostring(text) + 1)) end
				end
			end)
		end)
		if focusOk and focusConnection then
			table.insert(WordHelper.focusConnections, focusConnection)
		end
		syncFromFocusedTextBox(textBox)
	end

	function WordHelper:BindTextBox(textBox)
		bindFocusedTextBox(textBox)
	end

	function WordHelper:Destroy()
		if self.destroyed then
			return
		end
		if searchBox:IsFocused() then
			searchBox:ReleaseFocus()
		end
		self.destroyed = true
		self.autoTypeGeneration = self.autoTypeGeneration + 1
		self.isAutoTyping = false
		if self.remoteHook and self.remoteHook.handler == self.ProcessRemoteCall then
			self.remoteHook.handler = nil
		end
		disconnectFocusedTextBox()
		disconnectRows()
		for _, connection in ipairs(self.connections) do
			if connection and connection.Connected then
				connection:Disconnect()
			end
		end
		table.clear(self.connections)
		if screenGui and screenGui.Parent then
			screenGui:Destroy()
		end
		local replacement = sharedEnvironment.ZenithChess
		if type(replacement) ~= "table" or replacement.destroyed then replacement = nil end
		if sharedEnvironment.ZenithVocabulary == self then sharedEnvironment.ZenithVocabulary = nil end
		if sharedEnvironment.ZenithAssistant == self then sharedEnvironment.ZenithAssistant = replacement end
		if sharedEnvironment.ZenithWordHelper == self then sharedEnvironment.ZenithWordHelper = replacement end
		if type(shared) == "table" then
			local sharedReplacement = shared.ZenithChess
			if type(sharedReplacement) ~= "table" or sharedReplacement.destroyed then sharedReplacement = replacement end
			if shared.ZenithVocabulary == self then shared.ZenithVocabulary = nil end
			if shared.ZenithAssistant == self then shared.ZenithAssistant = sharedReplacement end
			if shared.ZenithWordHelper == self then shared.ZenithWordHelper = sharedReplacement end
		end
	end

	local function targetScale()
		local viewport = currentViewport()
		local preferredScale = isTouchDevice and Config.MobileUIScale or Config.UIScale
		return math.clamp(math.min(
			(viewport.X - 12) / Config.WindowWidth,
			(viewport.Y - 12) / Config.WindowHeight,
			preferredScale
		), Config.MinimumUIScale, preferredScale)
	end

	local function clampWindowPosition(x, y)
		local viewport = currentViewport()
		local scale = windowScale.Scale
		local width = Config.WindowWidth * scale
		local headerHeight = Config.HeaderHeight * scale
		-- Deliberately allow most of the menu outside any edge. Keep only a small
		-- draggable part of the header reachable so it can always be recovered.
		local visibleX = math.min(width, math.max(48, 56 * scale))
		local visibleHeaderY = math.min(headerHeight, math.max(22, 24 * scale))
		local minimumX = -width + visibleX
		local maximumX = viewport.X - visibleX
		local minimumY = -headerHeight + visibleHeaderY
		local maximumY = viewport.Y - visibleHeaderY
		return math.clamp(x, minimumX, maximumX), math.clamp(y, minimumY, maximumY)
	end

	updateResponsivePlacement = function()
		if WordHelper.destroyed then
			return
		end
		local viewport = currentViewport()
		local scale = targetScale()
		windowScale.Scale = scale
		if not WordHelper.hasUserPosition then
			local x = viewport.X - Config.WindowWidth * scale - 16
			local y = (viewport.Y - Config.WindowHeight * scale) * 0.5
			x, y = clampWindowPosition(x, y)
			window.Position = UDim2.fromOffset(x, y)
		else
			local x, y = clampWindowPosition(window.Position.X.Offset, window.Position.Y.Offset)
			window.Position = UDim2.fromOffset(x, y)
		end
	end

	-- Dragging stays entirely in ScreenGui offset space. Mixing AbsolutePosition
	-- with InputObject.Position caused the old menu to jump upward on first grab,
	-- especially with Roblox's top-bar inset and UIScale enabled.
	local function pointInside(guiObject, point)
		local topLeft, size = guiObject.AbsolutePosition, guiObject.AbsoluteSize
		return point.X >= topLeft.X and point.X <= topLeft.X + size.X
			and point.Y >= topLeft.Y and point.Y <= topLeft.Y + size.Y
	end

	connect(header.InputBegan, function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		local pointer = Vector2.new(input.Position.X, input.Position.Y)
		-- Header controls must click without briefly starting a window drag.
		if pointInside(settingsButton, pointer) or pointInside(minimizeButton, pointer) or pointInside(closeButton, pointer)
			or (previewLabel.Visible and pointInside(previewButton, pointer)) then
			return
		end
		WordHelper.dragging = true
		WordHelper.dragInput = input
		WordHelper.dragStart = pointer
		WordHelper.dragWindowStart = Vector2.new(window.Position.X.Offset, window.Position.Y.Offset)
	end)

	connect(userInputService.InputChanged, function(input)
		if not WordHelper.dragging then
			return
		end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if WordHelper.dragInput.UserInputType == Enum.UserInputType.Touch and input ~= WordHelper.dragInput then
			return
		end
		local position = Vector2.new(input.Position.X, input.Position.Y)
		local delta = position - WordHelper.dragStart
		local x, y = clampWindowPosition(
			WordHelper.dragWindowStart.X + delta.X,
			WordHelper.dragWindowStart.Y + delta.Y
		)
		WordHelper.hasUserPosition = true
		window.Position = UDim2.fromOffset(x, y)
	end)

	connect(userInputService.InputEnded, function(input)
		if not WordHelper.dragging then
			return
		end
		local activeInput = WordHelper.dragInput
		local matchingMouse = activeInput.UserInputType == Enum.UserInputType.MouseButton1
			and input.UserInputType == Enum.UserInputType.MouseButton1
		local matchingTouch = activeInput.UserInputType == Enum.UserInputType.Touch and input == activeInput
		if matchingMouse or matchingTouch then
			WordHelper.dragging = false
			WordHelper.dragInput = nil
		end
	end)

	-- Shared buffer for input that never reaches a real TextBox.
	-- action: a single letter, "back" (delete one) or "end" (space/enter = new word).
	-- Every submitted match word is remembered so it is hidden if suggested later
	-- (most word games forbid repeating a word).
	markUsed = function(word)
		word = normalizeLetters(word)
		if #word < 2 or WordHelper.usedWords[word] then return end
		-- Match history is authoritative even when the downloaded dictionary does
		-- not contain a valid game word. Remember it now; if a later dictionary
		-- load adds it, Hide Used already excludes it without double-counting.
		WordHelper.usedWords[word] = true
		if not (WordHelper.wordSet and WordHelper.wordSet[word]) then
			WordHelper.pendingUsedWords[word] = true
		else
			WordHelper.pendingUsedWords[word] = nil
		end
		WordHelper.usedCount = WordHelper.usedCount + 1
		updateUsedChip()
	end

	local function matchParticipantKey(player)
		if player == localPlayer then return "local" end
		local userId, name = nil, nil
		pcall(function()
			userId = player.UserId
			name = player.Name
		end)
		if type(userId) == "number" and userId > 0 then return "user:" .. tostring(userId) end
		if type(name) == "string" and name ~= "" then return "player:" .. name end
		if type(player) == "table" then
			userId = player.UserId
			name = player.PlayerName or player.Name
			if type(userId) == "number" and userId > 0 then return "user:" .. tostring(userId) end
			if type(name) == "string" and name ~= "" then return "player:" .. name end
		end
		return "unknown"
	end

	local function setMatchCurrentWord(player, word)
		local key = matchParticipantKey(player)
		word = normalizeLetters(word)
		WordHelper.matchCurrentWords[key] = word
		return word
	end

	local function rememberMatchWord(player, word)
		word = normalizeLetters(word)
		if #word < 2 then return false end
		local key = matchParticipantKey(player)
		local history = WordHelper.matchWordHistory[key]
		if not history then
			history = {}
			WordHelper.matchWordHistory[key] = history
		end
		history[word] = true
		WordHelper.matchCurrentWords[key] = ""
		markUsed(word)
		return true
	end

	WordHelper.RememberLastLetterWord = function(word)
		word = normalizeLetters(word)
		if currentPlaceId ~= LAST_LETTER_PLACE_ID or #word < 2 then return false end
		markUsed(word)
		local history = WordHelper.matchWordHistory["last-letter"]
		if not history then
			history = {}
			WordHelper.matchWordHistory["last-letter"] = history
		end
		history[word] = true
		return true
	end

	local function passiveType(action, resetSeconds)
		if not Settings.autoDetect or WordHelper.isAutoTyping or isRobloxMenuOpen() then return end
		-- In the confirmed two-player game, consumed physical keys and world-space
		-- keyboard buttons exist for both players. Never treat either as input unless
		-- MatchStatus or a local outgoing typing call has proved that this is our turn.
		if currentPlaceId == WORD_BATTLE_PLACE_ID and WordHelper.localTurn ~= true then return end
		local now = os.clock()
		if now - WordHelper.lastPassiveInput > resetSeconds then
			WordHelper.passiveBuffer = ""
		end
		WordHelper.lastPassiveInput = now
		if action == "back" then
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer, 1, math.max(0, #WordHelper.passiveBuffer - 1))
		elseif action == "end" then
			if currentPlaceId == WORD_BATTLE_PLACE_ID then
				-- The server's confirm event is authoritative. Preserve the complete
				-- attempt so confirm/wrongAnswer can attribute it before clearing input.
				WordHelper.awaitingSubmission = true
				setMatchCurrentWord(localPlayer, WordHelper.turnPrompt .. WordHelper.passiveBuffer)
				WordHelper.passiveBuffer = ""
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			else
				if currentPlaceId == LAST_LETTER_PLACE_ID then
					WordHelper.RememberLastLetterWord(WordHelper.passiveBuffer)
				else
					markUsed(WordHelper.passiveBuffer)
				end
				WordHelper.passiveBuffer = ""
				applyDetectedQuery("", "passive", true)
			end
			return
		else
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer .. action, -32)
		end
		if currentPlaceId == WORD_BATTLE_PLACE_ID then
			setMatchCurrentWord(localPlayer, WordHelper.turnPrompt .. WordHelper.passiveBuffer)
		end
		applyDetectedQuery(queryForTypedBuffer(WordHelper.passiveBuffer), "passive", true)
	end

	-- The supplied live trace emits the exact MatchStatus protocol below:
	--   updateId(sessionUUID, { players... })
	--   type(prompt, typingRemote, startedAt, Player, turnConfig)
	--   confirm(Player, word, ...) / wrongAnswer(Player, reason, ...)
	-- Use the protocol's fixed owner positions first. This matters when another
	-- argument happens to contain the local player: the active Player, not merely
	-- the first familiar value, decides whether Zenith may type.
	local function protocolPlayer(value)
		if value == localPlayer then return localPlayer end
		local player = nil
		pcall(function()
			if typeof(value) == "Instance" and value:IsA("Player") then player = value end
		end)
		if player then return player end
		if type(value) == "table" then
			local ownerName = value.Player or value.PlayerName or value.Name
			if (type(ownerName) == "string" and (ownerName == localPlayer.Name or ownerName == localPlayer.DisplayName))
				or (value.UserId ~= nil and localPlayer.UserId ~= nil and value.UserId == localPlayer.UserId) then
				return localPlayer
			end
			if type(ownerName) == "string" or value.UserId ~= nil then return value end
		end
		return nil
	end

	local function processMatchStatus(...)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return end
		local arguments = table.pack(...)
		local eventType = type(arguments[1]) == "string" and string.lower(arguments[1]) or ""
		if eventType == "updateid" then
			local sessionId, participants = arguments[2], arguments[3]
			-- A new match invalidates the prior match's dynamic remote even if its
			-- Instance still exists. Relearn it only from this match's type event.
			WordHelper.wordBattleRemote = nil
			WordHelper.wordBattleSessionId = nil
			WordHelper.knownWordRemote = nil
			WordHelper.knownSessionId = nil
			if type(sessionId) == "string" and string.match(sessionId, "^[%x%-]+$") then
				WordHelper.wordBattleSessionId = sessionId
				WordHelper.knownSessionId = sessionId
			end
			table.clear(WordHelper.knownMatchPlayers)
			if type(participants) == "table" then
				for _, participant in pairs(participants) do
					local player = protocolPlayer(participant)
					if player then table.insert(WordHelper.knownMatchPlayers, player) end
				end
			end
			return
		end

		local exactOwnerIndex = eventType == "type" and 5 or 2
		local owner = protocolPlayer(arguments[exactOwnerIndex])
		if not owner then
			-- Compatibility fallback for a server revision that appends fields while
			-- retaining one unambiguous Player argument.
			for index = 2, arguments.n do
				owner = protocolPlayer(arguments[index])
				if owner then break end
			end
		end

		if eventType == "type" then
			local candidate = arguments[3]
			pcall(function()
				if candidate and candidate:IsA("RemoteEvent")
					and string.find(candidate:GetFullName(), "MatchmakingFolders", 1, true) ~= nil then
					WordHelper.wordBattleRemote = candidate
					WordHelper.knownWordRemote = candidate
					if bindWordBattleRemote then bindWordBattleRemote(candidate) end
				end
			end)
		end

		local function waitForLocalTurn()
			cancelAutomaticTyping(true)
			WordHelper.lastWordActivation = nil
			WordHelper.lastWordActivationAt = 0
			WordHelper.localTurn = false
			WordHelper.turnPrompt = ""
			WordHelper.passiveBuffer = ""
			WordHelper.awaitingSubmission = false
			applyDetectedQuery("", "turn", true)
			detectionLabel.Text = "Esperando tu turno"
		end

		if eventType == "type" then
			WordHelper.letterSelectionActive = false
			WordHelper.letterSelectionOwner = nil
			table.clear(WordHelper.letterSelectionLetters)
			cancelAutomaticTyping(true)
			WordHelper.lastWordActivation = nil
			WordHelper.lastWordActivationAt = 0
			WordHelper.lastTurnEventAt = os.clock()
			local prompt = normalizeLetters(arguments[2])
			local ownTurn = owner == localPlayer
			WordHelper.turnOwner = owner
			WordHelper.turnSource = "match"
			WordHelper.awaitingSubmission = false
			WordHelper.searchMode = "prefix"
			modeChip.Text = "Empieza con"
			setMatchCurrentWord(owner, prompt)
			if ownTurn then
				WordHelper.localTurn = true
				local recentVisual = os.clock() - WordHelper.visiblePromptAt <= 0.75 and WordHelper.visiblePrompt or ""
				WordHelper.turnPrompt = prompt ~= "" and prompt or recentVisual
				WordHelper.passiveBuffer = ""
				WordHelper.lastPassiveInput = os.clock()
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			else
				waitForLocalTurn()
			end
		elseif eventType == "confirm" then
			WordHelper.lastTurnEventAt = os.clock()
			local confirmed = type(arguments[3]) == "string" and arguments[3]
				or WordHelper.matchCurrentWords[matchParticipantKey(owner)]
			rememberMatchWord(owner, confirmed)
			if owner == localPlayer then
				WordHelper.turnSource = "match"
				waitForLocalTurn()
			end
		elseif eventType == "wronganswer" then
			WordHelper.lastTurnEventAt = os.clock()
			-- Remember complete attempts for either participant as well. Even a word
			-- absent from the local dictionary stays in history, so a later dictionary
			-- refresh cannot offer that same server-seen attempt again.
			rememberMatchWord(owner, WordHelper.matchCurrentWords[matchParticipantKey(owner)])
			if owner ~= localPlayer then return end
			cancelAutomaticTyping(true)
			WordHelper.lastWordActivation = nil
			WordHelper.lastWordActivationAt = 0
			WordHelper.awaitingSubmission = false
			local reason = type(arguments[3]) == "string" and string.lower(arguments[3]) or ""
			if reason == "healthsubtracted" then
				WordHelper.turnSource = "match"
				waitForLocalTurn()
			else
				-- WordDoesNotExist (and equivalent retryable errors) does not pass
				-- the turn. Preserve the required letters and offer another word.
				WordHelper.localTurn = true
				WordHelper.turnOwner = localPlayer
				WordHelper.turnSource = "match"
				WordHelper.passiveBuffer = ""
				setMatchCurrentWord(localPlayer, WordHelper.turnPrompt)
				if WordHelper.turnPrompt == "" and os.clock() - WordHelper.visiblePromptAt <= 0.75 then
					WordHelper.turnPrompt = WordHelper.visiblePrompt
				end
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			end
		end
	end

	local function processLetterEvent(...)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return end
		local arguments = table.pack(...)
		local eventType = type(arguments[1]) == "string" and string.lower(arguments[1]) or ""
		if eventType == "enable" then
			WordHelper.letterSelectionActive = true
			WordHelper.letterSelectionOwner = protocolPlayer(arguments[5])
			table.clear(WordHelper.letterSelectionLetters)
			if type(arguments[2]) == "table" then
				for _, value in pairs(arguments[2]) do
					value = normalizeLetters(value)
					if #value == 1 then WordHelper.letterSelectionLetters[value] = true end
				end
			end
			-- LetterEvent is a separate choose-a-letter phase, not word entry. Its
			-- visible alphabet must never be mistaken for authority to type a word.
			cancelAutomaticTyping(true)
			WordHelper.localTurn = false
			WordHelper.turnOwner = WordHelper.letterSelectionOwner
			WordHelper.turnSource = "letter-selection"
			WordHelper.turnPrompt = ""
			WordHelper.passiveBuffer = ""
			applyDetectedQuery("", "turn", true)
		elseif eventType == "update" and type(arguments[2]) == "table" then
			table.clear(WordHelper.letterSelectionLetters)
			for _, value in pairs(arguments[2]) do
				value = normalizeLetters(value)
				if #value == 1 then WordHelper.letterSelectionLetters[value] = true end
			end
		elseif eventType == "disable" or eventType == "clear" then
			WordHelper.letterSelectionActive = false
			WordHelper.letterSelectionOwner = nil
			table.clear(WordHelper.letterSelectionLetters)
		end
	end

	WordHelper.ProcessMatchStatus = processMatchStatus
	WordHelper.ProcessLetterEvent = processLetterEvent
	local boundMatchStatus = setmetatable({}, { __mode = "k" })
	local function bindMatchStatus(remote)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID or not remote or boundMatchStatus[remote] then return end
		local valid = false
		pcall(function() valid = remote:IsA("RemoteEvent") and remote.Name == "MatchStatus" end)
		if not valid then return end
		boundMatchStatus[remote] = true
		pcall(function() connect(remote.OnClientEvent, processMatchStatus) end)
	end
	local boundLetterEvents = setmetatable({}, { __mode = "k" })
	local function bindLetterEvent(remote)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID or not remote or boundLetterEvents[remote] then return end
		local valid = false
		pcall(function()
			valid = remote:IsA("RemoteEvent") and remote.Name == "LetterEvent"
				and string.find(remote:GetFullName(), "MatchmakingFolders", 1, true) ~= nil
		end)
		if not valid then return end
		boundLetterEvents[remote] = true
		pcall(function() connect(remote.OnClientEvent, processLetterEvent) end)
	end
	if currentPlaceId == WORD_BATTLE_PLACE_ID then
		-- A game-side remote rename must disable only this adapter, never abort the
		-- entire Vocabulary UI during startup.
		pcall(function()
			bindMatchStatus(replicatedStorage:FindFirstChild("MatchStatus", true))
			bindLetterEvent(workspaceService:FindFirstChild("LetterEvent", true))
			connect(replicatedStorage.DescendantAdded, bindMatchStatus)
			connect(workspaceService.DescendantAdded, bindLetterEvent)
		end)
	end

	-- TextBoxFocused reports both chat and game interfaces; isGameTextBox filters
	-- Roblox chat before binding. The passive gameProcessed fallback also catches
	-- custom interfaces that render text in a label instead of focusing a TextBox.
	connect(userInputService.TextBoxFocused, function(textBox)
		local typingGuiEvent = false
		pcall(function()
			typingGuiEvent = WordHelper.wordBattleTypingGui ~= nil
				and textBox:IsDescendantOf(WordHelper.wordBattleTypingGui)
		end)
		local verifiedGameKeyboard = currentPlaceId == WORD_BATTLE_PLACE_ID
			and WordHelper.localTurn == true and WordHelper.autoTypeMethod == "game-keyboard"
			and typingGuiEvent
		if WordHelper.isAutoTyping and not verifiedGameKeyboard then
			cancelAutomaticTyping(true)
		elseif not WordHelper.isAutoTyping then
			WordHelper.suppressGeneratedUntil = 0
		end
		if not isRobloxMenuOpen() then bindFocusedTextBox(textBox) end
	end)
	connect(userInputService.TextBoxFocusReleased, function(textBox)
		local typingGuiEvent = false
		pcall(function()
			typingGuiEvent = WordHelper.wordBattleTypingGui ~= nil
				and textBox:IsDescendantOf(WordHelper.wordBattleTypingGui)
		end)
		local verifiedGameKeyboard = currentPlaceId == WORD_BATTLE_PLACE_ID
			and WordHelper.localTurn == true and WordHelper.autoTypeMethod == "game-keyboard"
			and typingGuiEvent
		if WordHelper.isAutoTyping and not verifiedGameKeyboard then cancelAutomaticTyping(true) end
		if textBox == WordHelper.activeExternalTextBox then
			disconnectFocusedTextBox()
		end
	end)

	local function bridgeBlockedSearchInput(input)
		local letter = LETTER_BY_KEYCODE[input.KeyCode]
		local isBackspace = input.KeyCode == Enum.KeyCode.Backspace
		if not letter and not isBackspace then return false end
		local inputAt = os.clock()
		local inputEpoch = WordHelper.inputEpoch
		WordHelper.manualCaptureUntil = inputAt + 2.5
		local before = searchBox.Text
		local cursor = tonumber(searchBox.CursorPosition) or (#before + 1)
		if cursor <= 0 then cursor = #before + 1 end
		local nativeAlreadyHandled = WordHelper.lastNativeSearchEditAt > 0
			and inputAt - WordHelper.lastNativeSearchEditAt <= 0.04
		-- Give Roblox's normal TextBox editor a frame first. Only synthesize when no
		-- native edit occurred either immediately before or after InputBegan.
		task.delay(0.035, function()
			if WordHelper.destroyed or WordHelper.inputEpoch ~= inputEpoch or isRobloxMenuOpen()
				or nativeAlreadyHandled or searchBox.Text ~= before
				or (not searchBox:IsFocused() and os.clock() > WordHelper.manualCaptureUntil) then
				return
			end
			local updated, nextCursor = before, cursor
			if letter then
				updated = string.sub(before, 1, cursor - 1) .. letter .. string.sub(before, cursor)
				nextCursor = cursor + 1
			elseif isBackspace and cursor > 1 then
				updated = string.sub(before, 1, cursor - 2) .. string.sub(before, cursor)
				nextCursor = cursor - 1
			end
			updated = string.sub(normalizeLetters(updated), 1, 32)
			synchronizingSearch = true
			searchBox.Text = updated
			synchronizingSearch = false
			WordHelper.passiveBuffer = updated
			applyDetectedQuery(updated, "manual", false)
			pcall(function() searchBox.CursorPosition = math.min(#updated + 1, nextCursor) end)
		end)
		return true
	end

	connect(userInputService.TextBoxFocusReleased, function(textBox)
		if textBox ~= searchBox then return end
		if WordHelper.destroyed or WordHelper.minimized or not WordHelper.visible then
			WordHelper.manualCaptureUntil = 0
			return
		end
		-- Some game controllers release the box immediately before forwarding the
		-- same consumed hardware key. Keep only a tiny keyboard-only bridge window;
		-- ordinary taps/clicks must never recapture Zenith's search box.
		WordHelper.manualCaptureUntil = os.clock() + 0.18
	end)

	local function exactRemoteHandledNear(action, inputAt)
		return currentPlaceId == WORD_BATTLE_PLACE_ID
			and WordHelper.lastRemoteAction == action
			and math.abs(WordHelper.lastRemoteActionAt - inputAt) <= 0.55
	end

	connect(userInputService.InputBegan, function(input, gameProcessed)
		if WordHelper.isAutoTyping or WordHelper.settingsOpen or isRobloxMenuOpen() then return end
		WordHelper.suppressGeneratedUntil = 0
		if input.KeyCode == Enum.KeyCode.RightShift then
			WordHelper:Toggle()
			return
		end

		local focusedTextBox = userInputService:GetFocusedTextBox()
		if focusedTextBox then
			if focusedTextBox == searchBox then
				bridgeBlockedSearchInput(input)
			else
				bindFocusedTextBox(focusedTextBox)
			end
			return
		end
		-- Some custom-keyboard controllers forcibly release the TextBox before this
		-- observer runs. Keep a short, explicit manual-search lease after the user
		-- focused Zenith so that the same physical key still edits Zenith, not the
		-- game-word detector.
		if WordHelper.manualCaptureUntil > 0 and os.clock() <= WordHelper.manualCaptureUntil then
			local canBridgeReleasedSearch = gameProcessed == true
				and input.UserInputType == Enum.UserInputType.Keyboard
				and (LETTER_BY_KEYCODE[input.KeyCode] ~= nil or input.KeyCode == Enum.KeyCode.Backspace)
			if canBridgeReleasedSearch then
				pcall(function() searchBox:CaptureFocus() end)
				bridgeBlockedSearchInput(input)
				return
			end
			-- A pointer/touch outside the box explicitly ends the short lease.
			WordHelper.manualCaptureUntil = 0
		end

		local letter = LETTER_BY_KEYCODE[input.KeyCode]
		local action = letter
			or (input.KeyCode == Enum.KeyCode.Backspace and "back")
			or ((input.KeyCode == Enum.KeyCode.Space
				or input.KeyCode == Enum.KeyCode.Return
				or input.KeyCode == Enum.KeyCode.KeypadEnter) and "end")
			or nil
		if not action then
			return
		end
		if currentPlaceId == WORD_BATTLE_PLACE_ID and WordHelper.localTurn ~= true then
			return
		end

		-- Record activity even when the game's ContextAction does not mark the key
		-- processed. This lets the read-only remote adapter validate the exact key,
		-- but raw movement/chat keys are never added unless the game consumed them.
		local pressedAt = os.clock()
		local inputEpoch = WordHelper.inputEpoch
		WordHelper.lastTypingActivity = pressedAt
		WordHelper.lastTypedAction = action
		local knownOwnTurn = WordHelper.localTurn == true
			and (currentPlaceId == WORD_BATTLE_PLACE_ID or currentPlaceId == LAST_LETTER_PLACE_ID)
		if not gameProcessed and not knownOwnTurn then return end
		-- With the confirmed hook active, the outgoing call is the sole source of
		-- truth. A second physical fallback would append the same letter twice.
		if WordHelper.exactRemoteHookActive then return end

		-- Wait briefly: an exact game remote or visible-word update wins, preventing
		-- one physical/on-screen key from being counted twice. The exact remote may
		-- occur just before InputBegan on touch controllers, so compare both ways.
		local fallbackDelay = currentPlaceId == WORD_BATTLE_PLACE_ID and 0.5 or 0.25
		task.delay(fallbackDelay, function()
			local remoteWon = exactRemoteHandledNear(action, pressedAt)
				or (currentPlaceId ~= WORD_BATTLE_PLACE_ID and WordHelper.lastRemoteTyping >= pressedAt)
			if not WordHelper.destroyed and WordHelper.inputEpoch == inputEpoch
				and not isRobloxMenuOpen() and not remoteWon
				and WordHelper.lastVisualUpdate < pressedAt then
				passiveType(action, Config.PassiveResetSeconds)
			end
		end)
	end)

	-- Custom on-screen keyboards: games that draw their own A-Z buttons instead of
	-- using a TextBox. Each key press is read from the button label when clicked.
	local BACK_WORDS = { "back", "del", "borrar", "⌫", "←", "<-" }
	local ENTER_WORDS = { "enter", "done", "submit", "send", "enviar", "ok", "go", "✓", "✔", "space", "espacio" }

	local function buttonKeyLabel(button)
		local text = ""
		pcall(function()
			if button:IsA("TextButton") then text = button.Text end
		end)
		if not string.find(text, "%S") then
			pcall(function()
				local label = button:FindFirstChildWhichIsA("TextLabel", true)
				text = label and label.Text or ""
			end)
		end
		if not string.find(text, "%S") then
			pcall(function()
				for _, attributeName in ipairs({ "Key", "Letter", "Value" }) do
					local value = button:GetAttribute(attributeName)
					if type(value) == "string" and value ~= "" then
						text = value
						break
					end
				end
			end)
		end
		-- Several mobile keyboards use ImageButtons named Q, Delete, or Done.
		if not string.find(text, "%S") then
			pcall(function() text = button.Name end)
		end
		text = string.gsub(text, "<.->", "")
		text = string.lower((string.gsub(text, "^%s*(.-)%s*$", "%1")))
		if string.match(text, "^[a-z]$") then
			return text
		end
		for _, word in ipairs(BACK_WORDS) do
			if text == word or (#word > 2 and string.find(text, word, 1, true)) then
				return "back"
			end
		end
		for _, word in ipairs(ENTER_WORDS) do
			if text == word then
				return "end"
			end
		end
		return nil
	end

	local function bindKeyboardButton(object)
		if WordHelper.keyboardButtons[object] or object:IsDescendantOf(screenGui) then
			return
		end
		if currentPlaceId == LAST_LETTER_PLACE_ID then
			local exactKeyboard = WordHelper.lastLetterKeyboardRoot
			local insideExactKeyboard = false
			pcall(function()
				insideExactKeyboard = exactKeyboard ~= nil and object:IsDescendantOf(exactKeyboard)
			end)
			if not insideExactKeyboard then return end
		elseif currentPlaceId == WORD_BATTLE_PLACE_ID and WordHelper.wordBattleTypingGui then
			local insideTypingGui = false
			pcall(function() insideTypingGui = object:IsDescendantOf(WordHelper.wordBattleTypingGui) end)
			if not insideTypingGui then return end
		end
		local isButton = false
		pcall(function()
			isButton = object:IsA("GuiButton")
		end)
		if not isButton then return end
		WordHelper.keyboardButtons[object] = true
		local localGuiButton = false
		pcall(function() localGuiButton = object:IsDescendantOf(playerGui) end)
		local initialAction = buttonKeyLabel(object)
		if initialAction and localGuiButton then WordHelper.keyboardByAction[initialAction] = object end
		connect(object.Activated, function()
			local action = buttonKeyLabel(object)
			if not action then return end
			if localGuiButton then WordHelper.keyboardByAction[action] = object end
			if WordHelper.isAutoTyping or isRobloxMenuOpen() then return end
			WordHelper.suppressGeneratedUntil = 0
			if currentPlaceId == WORD_BATTLE_PLACE_ID and WordHelper.localTurn ~= true then return end
			local pressedAt = os.clock()
			local inputEpoch = WordHelper.inputEpoch
			WordHelper.lastTypingActivity = pressedAt
			WordHelper.lastTypedAction = action
			-- Exact remote/visual text wins, preventing duplicate letters even when
			-- Activated is delivered just after the game's outgoing remote.
			local fallbackDelay = currentPlaceId == WORD_BATTLE_PLACE_ID and 0.5 or 0.25
			task.delay(fallbackDelay, function()
				local remoteWon = exactRemoteHandledNear(action, pressedAt)
					or (currentPlaceId ~= WORD_BATTLE_PLACE_ID and WordHelper.lastRemoteTyping >= pressedAt)
				if not WordHelper.destroyed and WordHelper.inputEpoch == inputEpoch
					and not isRobloxMenuOpen() and not remoteWon
					and WordHelper.lastVisualUpdate < pressedAt then
					passiveType(action, 6)
				end
			end)
		end)
	end

	-- Scan screen GUIs and in-world keyboards (SurfaceGuis on tables, etc.).
	local function scanForKeyboards(root, immediate)
		local function scan()
			local ok, descendants = pcall(function()
				return root:GetDescendants()
			end)
			if not ok then return end
			for index, object in ipairs(descendants) do
				if WordHelper.destroyed then return end
				bindKeyboardButton(object)
				if not immediate and index % 400 == 0 then task.wait() end
			end
		end
		if immediate then scan() else task.spawn(scan) end
		pcall(function()
			connect(root.DescendantAdded, bindKeyboardButton)
		end)
	end

	-- Place 129866685202296 renders the live entry under InGame.Frame.CurrentWord.
	-- Reading those visible labels is more reliable than guessing the contents of
	-- its shared Packet buffer and remains completely read-only.
	local function guiObjectVisible(object, boundary)
		local current = object
		while current and current ~= boundary do
			local hidden = false
			pcall(function()
				if current:IsA("GuiObject") and not current.Visible then hidden = true end
				if current:IsA("LayerCollector") and not current.Enabled then hidden = true end
			end)
			if hidden then return false end
			current = current.Parent
		end
		return true
	end

	local function visibleText(object)
		local text = ""
		local ok = pcall(function()
			if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
				text = object.Text
			end
		end)
		if not ok then return "" end
		text = string.gsub(tostring(text), "<.->", "")
		text = string.lower((string.gsub(text, "^%s*(.-)%s*$", "%1")))
		return string.match(text, "^[a-z]+$") and text or ""
	end

	local function readLargePromptLetters()
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return nil end
		local camera = workspaceService.CurrentCamera
		local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
		local records = {}
		local ok, descendants = pcall(function() return playerGui:GetDescendants() end)
		if not ok then return nil end
		for _, object in ipairs(descendants) do
			local isOwnGui = false
			pcall(function() isOwnGui = object:IsDescendantOf(screenGui) end)
			if not isOwnGui and guiObjectVisible(object, playerGui) then
				local text = visibleText(object)
				if #text == 1 then
					local position, size, textSize = Vector2.new(0, 0), Vector2.new(0, 0), 0
					pcall(function()
						position, size, textSize = object.AbsolutePosition, object.AbsoluteSize, object.TextSize
					end)
					local centerX, centerY = position.X + size.X * 0.5, position.Y + size.Y * 0.5
					if centerX >= viewport.X * 0.14 and centerX <= viewport.X * 0.86
						and centerY >= viewport.Y * 0.05 and centerY <= viewport.Y * 0.65
						and (size.Y >= 28 or textSize >= 20) then
						table.insert(records, {
							text = text, x = centerX, y = centerY,
							height = math.max(size.Y, textSize), area = math.max(1, size.X * size.Y),
						})
					end
				end
			end
		end
		local best, bestScore = nil, 0
		for _, seed in ipairs(records) do
			local row, score = {}, 0
			for _, record in ipairs(records) do
				if math.abs(record.y - seed.y) <= math.max(22, seed.height * 0.45) then
					table.insert(row, record)
					score = score + record.area
				end
			end
			if #row >= 1 and #row <= 4 and score > bestScore then best, bestScore = row, score end
		end
		if not best then return nil end
		table.sort(best, function(a, b) return a.x < b.x end)
		local letters = table.create(#best)
		for index, record in ipairs(best) do letters[index] = record.text end
		return table.concat(letters)
	end

	local function pollWordBattlePrompt()
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return end
		local prompt = readLargePromptLetters()
		if not prompt or prompt == "" then return end
		WordHelper.visiblePrompt = prompt
		WordHelper.visiblePromptAt = os.clock()
		-- A visible prompt alone is not proof of ownership: every client can see the
		-- opponent's letters. It may only supplement an already verified local turn.
		if WordHelper.localTurn == true and WordHelper.turnPrompt == "" then
			WordHelper.turnPrompt = prompt
			WordHelper.searchMode = "prefix"
			modeChip.Text = "Empieza con"
			applyDetectedQuery(prompt, "turn", true)
		end
	end

	WordHelper.FindLastLetterGuiRoots = function()
		if currentPlaceId ~= LAST_LETTER_PLACE_ID then return nil, nil end
		-- These are the two hierarchies required by the game's own modules. Do not
		-- fall back to similarly named descendants from menus, results, or chat.
		local overbar = playerGui:FindFirstChild("Overbar")
		local overbarFrame = overbar and overbar:FindFirstChild("Frame")
		local keyboard = overbarFrame and overbarFrame:FindFirstChild("Keyboard")
		local inGame = playerGui:FindFirstChild("InGame")
		local inGameFrame = inGame and inGame:FindFirstChild("Frame")
		local currentWord = inGameFrame and inGameFrame:FindFirstChild("CurrentWord")
		WordHelper.lastLetterKeyboardRoot = keyboard
		WordHelper.lastLetterCurrentWordRoot = currentWord
		return keyboard, currentWord
	end

	local function readLastLetterVisibleWord()
		if currentPlaceId ~= LAST_LETTER_PLACE_ID then return nil end
		local _, root = WordHelper.FindLastLetterGuiRoots()
		if not root or not guiObjectVisible(root, playerGui) then return nil end
		local rootText = visibleText(root)
		if #rootText >= 1 and #rootText <= 32 then return rootText end

		-- InGame.AddLetter clones one direct Case per logical character. Preserve
		-- child/insertion order; sorting LayoutOrder or animated X positions caused
		-- transient reversed words while RotateLetters was moving cases.
		local letters, wholeWords = {}, {}
		local children = {}
		pcall(function() children = root:GetChildren() end)
		for childIndex, child in ipairs(children) do
			if guiObjectVisible(child, root) then
				local text = visibleText(child)
				if text == "" then
					local descendants = {}
					pcall(function() descendants = child:GetDescendants() end)
					for _, object in ipairs(descendants) do
						local candidate = guiObjectVisible(object, child) and visibleText(object) or ""
						if #candidate == 1 then
							text = candidate
							break
						elseif #candidate >= 2 and #candidate <= 32 then
							table.insert(wholeWords, candidate)
						end
					end
				end
				if #text == 1 then
					table.insert(letters, { text = text, index = childIndex })
				elseif #text >= 2 and #text <= 32 then
					table.insert(wholeWords, text)
				end
			end
		end
		if #letters > 0 then
			table.sort(letters, function(a, b) return a.index < b.index end)
			local output = table.create(#letters)
			for index, record in ipairs(letters) do output[index] = record.text end
			return table.concat(output)
		end
		if #wholeWords > 0 then
			table.sort(wholeWords, function(a, b) return #a > #b end)
			return wholeWords[1]
		end
		return ""
	end

	WordHelper.ApplyLastLetterState = function(candidate, source, operation)
		if currentPlaceId ~= LAST_LETTER_PLACE_ID then return false end
		candidate = normalizeLetters(candidate)
		local now = os.clock()
		local current = normalizeLetters(WordHelper.lastLetterState)
		local fromPacket = source == "remote"

		if fromPacket then
			WordHelper.lastLetterPacketAt = now
			WordHelper.lastRemoteTyping = now
		else
			-- The namecall hook queues before the game's callback can repaint. Let the
			-- ordered packet drain first so one locally rendered key is never appended
			-- a second time when its deferred observer runs.
			if WordHelper.remoteHook and (WordHelper.remoteHook.pendingCount or 0) > 0 then return false end
			if candidate == WordHelper.lastLetterVisualCandidate then
				WordHelper.lastLetterVisualCandidateCount = WordHelper.lastLetterVisualCandidateCount + 1
			else
				WordHelper.lastLetterVisualCandidate = candidate
				WordHelper.lastLetterVisualCandidateCount = 1
			end
			if candidate == current then
				WordHelper.lastVisualWord = candidate
				WordHelper.lastVisualUpdate = now
				return true
			end

			local extension = current == "" or string.sub(candidate, 1, #current) == current
			local contraction = candidate ~= "" and string.sub(current, 1, #candidate) == candidate
			local recentBackspace = WordHelper.lastTypedAction == "back"
				and now - WordHelper.lastTypingActivity <= 1
			-- An ordered packet state is newer than tweening UI. A visual frame may
			-- confirm it, but cannot overwrite it with a stale prefix/reversal.
			if now - WordHelper.lastLetterPacketAt <= 0.75 and candidate ~= current then
				return false
			end
			-- Prefix growth is monotonic and can stay fluid. Shrinks need a proven
			-- backspace; resets/reorders must survive two consecutive polls.
			if not extension and not (contraction and recentBackspace)
				and WordHelper.lastLetterVisualCandidateCount < 2 then
				return false
			end
		end

		if candidate ~= current then
			local extension = current == "" or string.sub(candidate, 1, #current) == current
			local contraction = candidate ~= "" and string.sub(current, 1, #candidate) == candidate
			local recentBackspace = operation == "back" or (WordHelper.lastTypedAction == "back"
				and now - WordHelper.lastTypingActivity <= 1)
			-- A completed word disappears when the next first letter/empty round is
			-- rendered. This records words from either player, while ordinary edits do
			-- not inflate Used.
			if #current >= 2 and not extension and not (contraction and recentBackspace) then
				WordHelper.RememberLastLetterWord(current)
			end
			WordHelper.lastLetterState = candidate
			WordHelper.lastLetterStateAt = now
			WordHelper.passiveBuffer = candidate
			WordHelper.lastPassiveInput = now
			if not WordHelper.isAutoTyping then
				applyDetectedQuery(candidate, fromPacket and "remote" or "visual", true)
			end
		end
		if not fromPacket then
			WordHelper.lastVisualWord = candidate
			WordHelper.lastVisualUpdate = now
		end
		return true
	end

	local function pollVisibleGameWord()
		local word = readLastLetterVisibleWord()
		if word == nil then
			if currentPlaceId == LAST_LETTER_PLACE_ID and WordHelper.lastLetterState ~= ""
				and os.clock() - WordHelper.lastVisualUpdate > 0.5 then
				WordHelper.ApplyLastLetterState("", "visual", "reset")
			end
			return
		end
		WordHelper.ApplyLastLetterState(word, "visual", "display")
	end

	-- ===== Isolated game-remote observation (executor only) =====
	-- Word Battle accepts exactly (session UUID, boolean, character) on its dynamic
	-- match remote. Last Letter observes only the exact Packet.RemoteEvent path.
	-- Decode only Last Letter's four observed, declared packet shapes. This is an
	-- observer, not a serializer: typing still goes through InputService/buttons.
	-- A buffer can batch several frames (for example 0x0A followed by 0x09), so
	-- walk it in order instead of treating the first byte as the whole call.
	WordHelper.DecodeLastLetterPacket = function(value)
		local records = {}
		if typeof(value) ~= "buffer" then return records end
		local ok, raw = pcall(buffer.tostring, value)
		if not ok or type(raw) ~= "string" then return records end
		local index, size = 1, #raw
		while index <= size do
			local packetId = string.byte(raw, index)
			if packetId == 0x09 and index + 5 <= size
				and string.byte(raw, index + 1) == 0x01
				and string.byte(raw, index + 2) == 0x33
				and (string.byte(raw, index + 3) == 0x00 or string.byte(raw, index + 3) == 0x01)
				and string.byte(raw, index + 4) == 0x0B then
				local length = string.byte(raw, index + 5)
				local last = index + 5 + length
				local text = last <= size and string.sub(raw, index + 6, last) or ""
				if length >= 1 and length <= 32 and string.match(text, "^[A-Za-z]+$") then
					table.insert(records, {
						kind = "update",
						adding = string.byte(raw, index + 3) == 0x01,
						text = string.lower(text),
					})
					index = last + 1
				else
					index = index + 1
				end
			elseif packetId == 0x0E and index + 3 <= size and string.byte(raw, index + 1) >= 0x80 then
				local length = string.byte(raw, index + 2) + string.byte(raw, index + 3) * 256
				local last = index + 3 + length
				local text = last <= size and string.sub(raw, index + 4, last) or ""
				if length >= 2 and length <= 32 and string.match(text, "^[A-Za-z]+$") then
					table.insert(records, { kind = "submit", text = string.lower(text) })
					index = last + 1
				else
					index = index + 1
				end
			elseif packetId == 0x0F and index + 2 <= size and string.byte(raw, index + 1) >= 0x80 then
				local length = string.byte(raw, index + 2)
				local last = index + 2 + length
				local text = last <= size and string.sub(raw, index + 3, last) or ""
				if length == 1 and string.match(text, "^[A-Za-z]$") then
					table.insert(records, { kind = "first", text = string.lower(text) })
					index = last + 1
				else
					index = index + 1
				end
			else
				index = index + 1
			end
		end
		return records
	end

	WordHelper.ProcessLastLetterPacket = function(arguments, callSequence)
		if currentPlaceId ~= LAST_LETTER_PLACE_ID or type(arguments) ~= "table" then return false end
		if callSequence then
			if callSequence <= WordHelper.lastLetterPacketSequence then return false end
			WordHelper.lastLetterPacketSequence = callSequence
		end
		local handled = false
		for argumentIndex = 1, arguments.n or #arguments do
			for _, record in ipairs(WordHelper.DecodeLastLetterPacket(arguments[argumentIndex])) do
				handled = true
				if record.kind == "update" then
					if record.adding then
						local current = normalizeLetters(WordHelper.lastLetterState)
						local expected
						if current == "" then
							local visible = normalizeLetters(readLastLetterVisibleWord() or "")
							expected = #visible >= 2 and string.sub(visible, -1) == record.text
								and visible or (visible .. record.text)
						else
							expected = current .. record.text
						end
						expected = string.sub(expected, -32)
						WordHelper.lastTypedAction = record.text
						WordHelper.lastTypingActivity = os.clock()
						WordHelper.ApplyLastLetterState(expected, "remote", "add")
					else
						WordHelper.lastTypedAction = "back"
						WordHelper.lastTypingActivity = os.clock()
						WordHelper.ApplyLastLetterState(record.text, "remote", "back")
					end
				elseif record.kind == "first" then
					WordHelper.ApplyLastLetterState(record.text, "remote", "first")
				elseif record.kind == "submit" then
					WordHelper.RememberLastLetterWord(record.text)
					WordHelper.lastTypedAction = "end"
					WordHelper.lastTypingActivity = os.clock()
					WordHelper.lastSubmitAt = WordHelper.lastTypingActivity
				end
			end
		end
		return handled
	end

	local function instancePath(instance)
		local ok, path = pcall(function() return instance:GetFullName() end)
		return ok and type(path) == "string" and path or ""
	end

	local function processWordBattleIncoming(remote, ...)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID or remote ~= WordHelper.wordBattleRemote then return end
		local arguments = table.pack(...)
		local eventType = type(arguments[1]) == "string" and string.lower(arguments[1]) or ""
		if eventType ~= "syncfilteredword" or type(arguments[2]) ~= "string" then return end
		local owner = WordHelper.turnOwner
		local fullWord = normalizeLetters(arguments[2])
		local prompt = normalizeLetters(WordHelper.turnPrompt)
		-- A delayed opponent broadcast can arrive just after ownership changes. A
		-- local word is valid only when it retains this turn's required prefix.
		if owner == localPlayer and prompt ~= "" and string.sub(fullWord, 1, #prompt) ~= prompt then return end
		setMatchCurrentWord(owner, fullWord)
		-- The same channel broadcasts the opponent's live text. Keep it in the
		-- participant history, but only the verified local owner may drive Zenith's
		-- query or suffix buffer.
		if owner ~= localPlayer or WordHelper.localTurn ~= true then return end
		local suffix = fullWord
		if prompt ~= "" and string.sub(fullWord, 1, #prompt) == prompt then
			suffix = string.sub(fullWord, #prompt + 1)
		end
		WordHelper.passiveBuffer = suffix
		WordHelper.lastPassiveInput = os.clock()
		WordHelper.lastRemoteTyping = os.clock()
		if not WordHelper.isAutoTyping then
			applyDetectedQuery(fullWord, "remote", true)
		end
	end

	bindWordBattleRemote = function(remote)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID or not remote or WordHelper.wordBattleRemoteListeners[remote] then return end
		local valid = false
		pcall(function()
			valid = remote:IsA("RemoteEvent")
				and string.find(instancePath(remote), "MatchmakingFolders", 1, true) ~= nil
		end)
		if not valid then return end
		local connected = false
		pcall(function()
			connect(remote.OnClientEvent, function(...)
				processWordBattleIncoming(remote, ...)
			end)
			connected = true
		end)
		if connected then WordHelper.wordBattleRemoteListeners[remote] = true end
	end
	-- A turn can arrive during script startup after MatchStatus is connected but
	-- before this callback is defined; finish that binding deterministically.
	if WordHelper.wordBattleRemote then bindWordBattleRemote(WordHelper.wordBattleRemote) end
	WordHelper.ProcessWordBattleIncoming = processWordBattleIncoming
	WordHelper.ProcessWordBombIncoming = processWordBattleIncoming

	local function handleWordBattleRemoteCall(remote, method, arguments)
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID or method ~= "FireServer" or arguments.n < 3 then
			return false
		end
		local sessionId, adding, value = arguments[1], arguments[2], arguments[3]
		local path = instancePath(remote)
		if type(sessionId) ~= "string" or not string.match(sessionId, "^[%x%-]+$")
			or type(adding) ~= "boolean" or type(value) ~= "string"
			or not string.find(path, "MatchmakingFolders", 1, true) then
			return false
		end
		local text = string.lower(value)
		if not (string.match(text, "^[a-z]$") or text == " ") then return false end
		-- Once MatchStatus identifies this match, a matching-looking call from any
		-- other UUID or RemoteEvent is not authoritative and cannot replace it.
		if WordHelper.wordBattleSessionId and WordHelper.wordBattleSessionId ~= sessionId then return false end
		if WordHelper.wordBattleRemote and WordHelper.wordBattleRemote ~= remote then return false end

		local now = os.clock()
		local signature = sessionId .. "|" .. tostring(adding) .. "|" .. text
		if WordHelper.lastRemoteSignature == signature and now - WordHelper.lastRemoteSignatureAt <= 0.01 then
			return true
		end
		WordHelper.lastRemoteSignature = signature
		WordHelper.lastRemoteSignatureAt = now
		WordHelper.wordBattleRemote = remote
		WordHelper.wordBattleSessionId = sessionId
		WordHelper.knownWordRemote = remote
		WordHelper.knownSessionId = sessionId
		if bindWordBattleRemote then bindWordBattleRemote(remote) end
		local remoteAction = text == " " and "end" or (adding and text or "back")
		WordHelper.lastRemoteTyping = now
		WordHelper.lastRemoteAction = remoteAction
		WordHelper.lastRemoteActionAt = now
		WordHelper.localTurn = true -- a local outgoing typing call is authoritative
		WordHelper.turnOwner = localPlayer
		WordHelper.turnSource = "remote"
		WordHelper.lastLocalRemoteCall = now
		WordHelper.lastRemoteTyping = now
		if WordHelper.turnPrompt == "" then
			local visiblePrompt = readLargePromptLetters()
			if visiblePrompt and visiblePrompt ~= "" then
				WordHelper.visiblePrompt = visiblePrompt
				WordHelper.visiblePromptAt = now
			elseif now - WordHelper.visiblePromptAt <= 0.75 then
				visiblePrompt = WordHelper.visiblePrompt
			end
			if visiblePrompt and visiblePrompt ~= "" then WordHelper.turnPrompt = visiblePrompt end
		end
		WordHelper.searchMode = "prefix"
		modeChip.Text = "Empieza con"
		if text == " " then
			WordHelper.lastSubmitAt = now
			WordHelper.awaitingSubmission = true
			-- Preserve the complete attempt until confirm/wrongAnswer attributes it
			-- to this player; both outcomes arrive after the local suffix is cleared.
			setMatchCurrentWord(localPlayer, WordHelper.turnPrompt .. WordHelper.passiveBuffer)
			WordHelper.passiveBuffer = ""
			applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			return true
		end
		WordHelper.awaitingSubmission = false
		if now - WordHelper.lastPassiveInput > 8 then WordHelper.passiveBuffer = "" end
		if adding then
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer .. text, -32)
		else
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer, 1, math.max(0, #WordHelper.passiveBuffer - 1))
		end
		WordHelper.lastPassiveInput = now
		setMatchCurrentWord(localPlayer, WordHelper.turnPrompt .. WordHelper.passiveBuffer)
		-- Generated letters update the authoritative suffix but keep the current
		-- suggestion row stable long enough for a second tap to copy it.
		if not WordHelper.isAutoTyping then
			applyDetectedQuery(queryForTypedBuffer(WordHelper.passiveBuffer), "remote", true)
		end
		return true
	end

	local function onRemoteCall(remote, method, arguments, callSequence)
		if WordHelper.destroyed or not Settings.autoDetect or WordHelper.robloxMenuOpen then return end

		if activeGameAdapter == "word-battle" then
			-- Word Battle owns only the UUID + boolean + character protocol exposed
			-- by its MatchStatus event. No Packet or generic typing remote is accepted.
			handleWordBattleRemoteCall(remote, method, arguments)
			return
		end
		if activeGameAdapter ~= "last-letter" then return end

		-- Last Letter owns only ReplicatedStorage.Modules.Packet.RemoteEvent. Decode
		-- its observed ordered frames for state, but never fabricate a packet.
		if method ~= "FireServer"
			or instancePath(remote) ~= "ReplicatedStorage.Modules.Packet.RemoteEvent" then return end
		WordHelper.lastLetterPacketRemote = remote
		WordHelper.ProcessLastLetterPacket(arguments, callSequence)
	end

	-- Exposed for deterministic adapters/tests; it only observes arguments and never
	-- invokes a remote or changes game controls.
	WordHelper.ProcessRemoteCall = onRemoteCall

	do
		local remoteHookKey = activeGameAdapter == "word-battle" and "ZenithWordBattleRemoteHookV1"
			or (activeGameAdapter == "last-letter" and "ZenithLastLetterRemoteHookV2" or nil)
		local remoteHook = remoteHookKey and sharedEnvironment[remoteHookKey] or nil
		if remoteHookKey and type(remoteHook) ~= "table" then
			-- Keep installation in a child function. Delta compiles executor chunks
			-- without register optimization, so these temporary locals must not consume
			-- registers in launchWordHelper's already-large lexical scope.
			remoteHook = (function()
				local hookMetamethod = zenithFileFunction("hookmetamethod")
				local getNamecallMethod = zenithFileFunction("getnamecallmethod")
				local newCClosure = zenithFileFunction("newcclosure") or function(callback)
					return callback
				end
				if not hookMetamethod or not getNamecallMethod then return nil end
				-- Installed once per game session; re-executions just swap the handler.
				local state = {}
				local installed = pcall(function()
					local original
					original = hookMetamethod(game, "__namecall", newCClosure(function(self, ...)
						if state.handler then
							local method = getNamecallMethod()
							if method == "FireServer" or method == "InvokeServer" or method == "Fire" then
								state.sequence = (state.sequence or 0) + 1
								state.pendingCount = (state.pendingCount or 0) + 1
								state.queue = state.queue or {}
								table.insert(state.queue, {
									remote = self,
									method = method,
									arguments = table.pack(...),
									sequence = state.sequence,
								})
								if not state.draining then
									state.draining = true
									task.defer(function()
										while #state.queue > 0 do
											local call = table.remove(state.queue, 1)
											local handler = state.handler
											if handler then
												pcall(handler, call.remote, call.method, call.arguments, call.sequence)
											end
											state.pendingCount = math.max(0, (state.pendingCount or 1) - 1)
										end
										state.draining = false
									end)
								end
							end
						end
						return original(self, ...)
					end))
				end)
				if not installed then return nil end
				sharedEnvironment[remoteHookKey] = state
				return state
			end)()
		end
		if type(remoteHook) == "table" then
			remoteHook.adapter = activeGameAdapter
			if activeGameAdapter == "last-letter" then
				remoteHook.queue = {}
				remoteHook.pendingCount = 0
				remoteHook.draining = false
			end
			remoteHook.handler = onRemoteCall
			WordHelper.remoteHook = remoteHook
		end
		WordHelper.exactRemoteHookActive = (currentPlaceId == WORD_BATTLE_PLACE_ID
			or currentPlaceId == LAST_LETTER_PLACE_ID) and type(remoteHook) == "table"
	end

	-- Bind only the authoritative keyboard hierarchy for each supported game.
	-- This prevents chat, shop buttons, and stale result screens from becoming
	-- typing sources merely because they contain a one-letter GuiButton.
	if currentPlaceId == WORD_BATTLE_PLACE_ID then
		pcall(function()
			local function bindWordBattleTypingGui(object)
				if not object or object.Name ~= "TypingSystemGui" then return end
				WordHelper.wordBattleTypingGui = object
				scanForKeyboards(object, true)
			end
			bindWordBattleTypingGui(playerGui:FindFirstChild("TypingSystemGui", true))
			connect(playerGui.DescendantAdded, bindWordBattleTypingGui)
		end)
	elseif currentPlaceId == LAST_LETTER_PLACE_ID then
		pcall(function()
			local function bindLastLetterKeyboard()
				local keyboard = WordHelper.FindLastLetterGuiRoots()
				if keyboard then scanForKeyboards(keyboard, true) end
			end
			bindLastLetterKeyboard()
			connect(playerGui.DescendantAdded, function(object)
				if object.Name == "Keyboard" or object.Name == "CurrentWord"
					or object.Name == "Overbar" or object.Name == "InGame" then
					task.defer(bindLastLetterKeyboard)
				end
			end)
		end)
	else
		-- Unsupported places retain the non-invasive universal TextBox/keyboard
		-- fallback, but never gain either supported game's remote adapter.
		scanForKeyboards(playerGui)
		scanForKeyboards(workspaceService)
	end

	-- Keep late-stage helpers on the runtime table instead of adding locals to the
	-- parent function; Delta's loadstring compiler enforces a 200-register limit.
	WordHelper.AutoTypeStillValid = function(generation, inputEpoch, validator)
		if WordHelper.destroyed or isRobloxMenuOpen()
			or WordHelper.autoTypeGeneration ~= generation
			or WordHelper.inputEpoch ~= inputEpoch then
			return false
		end
		if validator then
			local ok, valid = pcall(validator)
			return ok and valid == true
		end
		return true
	end

	WordHelper.MappedButtonUsable = function(button)
		if not button or not button.Parent then return false end
		local usable = false
		pcall(function()
			usable = button:IsA("GuiButton") and button.Active ~= false
				and guiObjectVisible(button, playerGui)
		end)
		return usable
	end

	WordHelper.LastLetterKeyboardActive = function()
		if currentPlaceId ~= LAST_LETTER_PLACE_ID then return false end
		local previousKeyboard = WordHelper.lastLetterKeyboardRoot
		local keyboard = WordHelper.FindLastLetterGuiRoots()
		if not keyboard or not guiObjectVisible(keyboard, playerGui) then return false end
		if keyboard ~= previousKeyboard or not WordHelper.keyboardByAction.a
			or not WordHelper.keyboardByAction.a.Parent then
			scanForKeyboards(keyboard, true)
		end
		local visibleLetters = 0
		for letter in string.gmatch("abcdefghijklmnopqrstuvwxyz", ".") do
			local button = WordHelper.keyboardByAction[letter]
			local inside = false
			pcall(function() inside = button and button:IsDescendantOf(keyboard) end)
			if inside and WordHelper.MappedButtonUsable(button) then visibleLetters = visibleLetters + 1 end
		end
		return visibleLetters >= 8
	end

	WordHelper.SendLastLetterAction = function(action)
		if not WordHelper.LastLetterKeyboardActive() then return false end
		local button = WordHelper.keyboardByAction[action]
		local inside = false
		pcall(function()
			inside = button and button:IsDescendantOf(WordHelper.lastLetterKeyboardRoot)
		end)
		if not inside or not WordHelper.MappedButtonUsable(button) then return false end
		WordHelper.lastTypingActivity = os.clock()
		WordHelper.suppressGeneratedUntil = os.clock() + 0.12
		WordHelper.lastTypedAction = action
		if fireSignal then
			if action == "back" or action == "end" then
				if pcall(fireSignal, button.MouseButton1Click) then return true end
			else
				local pressed = pcall(fireSignal, button.MouseButton1Down)
				pcall(fireSignal, button.MouseButton1Up)
				if pressed then return true end
			end
		end
		if virtualInputManager then
			local ok = pcall(function()
				local position, size = button.AbsolutePosition, button.AbsoluteSize
				local x, y = position.X + size.X * 0.5, position.Y + size.Y * 0.5
				virtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
				virtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
			end)
			if ok then return true end
		end
		return false
	end

	WordHelper.WordBattleTypingUiActive = function()
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return false end
		local root = WordHelper.wordBattleTypingGui
		if not root or not root.Parent then
			root = playerGui:FindFirstChild("TypingSystemGui", true)
			if root then
				WordHelper.wordBattleTypingGui = root
				scanForKeyboards(root, true)
			end
		end
		if not root or not guiObjectVisible(root, playerGui) then return false end

		local focused = userInputService:GetFocusedTextBox()
		local focusInside = false
		pcall(function()
			focusInside = focused ~= nil and focused ~= searchBox and focused:IsDescendantOf(root)
		end)
		if focusInside then return true end

		-- Mobile can move focus to Zenith after the turn already opened. The game's
		-- short, editable current-word box remains visible only while this client's
		-- TypingSystem is active, so it is also safe launch-mid-turn evidence.
		local descendants = {}
		pcall(function() descendants = root:GetDescendants() end)
		for _, object in ipairs(descendants) do
			local activeInput = false
			pcall(function()
				local text, size = normalizeLetters(object.Text), object.AbsoluteSize
				activeInput = object:IsA("TextBox") and object ~= searchBox
					and object.TextEditable ~= false and guiObjectVisible(object, root)
					and size.X >= 20 and size.Y >= 16 and #text >= 1 and #text <= 4
			end)
			if activeInput then return true end
		end

		-- A visible alphabet inside the exact TypingSystemGui is the game's local
		-- input surface. Require several keys so an unrelated one-letter control can
		-- never turn an opponent's visible prompt into typing authority.
		local visibleLetters = 0
		for letter in string.gmatch("abcdefghijklmnopqrstuvwxyz", ".") do
			local button = WordHelper.keyboardByAction[letter]
			local inside = false
			pcall(function() inside = button and button:IsDescendantOf(root) end)
			if inside and WordHelper.MappedButtonUsable(button) then visibleLetters = visibleLetters + 1 end
		end
		return visibleLetters >= 8
	end

	WordHelper.RecoverWordBattleTurnFromUi = function()
		if currentPlaceId ~= WORD_BATTLE_PLACE_ID then return false end
		-- Never overrule an explicit opponent/post-confirm state or LetterEvent's
		-- separate choose-a-letter phase. UI recovery is only for an unknown launch.
		if WordHelper.localTurn == false and WordHelper.turnSource ~= nil then return false end
		if WordHelper.letterSelectionActive or not WordHelper.WordBattleTypingUiActive() then return false end
		local prompt = normalizeLetters(readLargePromptLetters() or "")
		if prompt == "" then
			local query = normalizeLetters(WordHelper.currentQuery)
			if #query >= 1 and #query <= 4 then prompt = query end
		end
		if prompt == "" then return false end
		WordHelper.localTurn = true
		WordHelper.turnOwner = localPlayer
		WordHelper.turnSource = "typing-ui"
		WordHelper.turnPrompt = prompt
		WordHelper.passiveBuffer = ""
		WordHelper.searchMode = "prefix"
		WordHelper.detectionSource = "turn"
		WordHelper.lastTurnEventAt = os.clock()
		setMatchCurrentWord(localPlayer, prompt)
		modeChip.Text = "Empieza con"
		detectionLabel.Text = "Texto: " .. prompt .. "  |  tu turno"
		return true
	end
	WordHelper.WordBombTypingUiActive = WordHelper.WordBattleTypingUiActive
	WordHelper.RecoverWordBombTurnFromUi = WordHelper.RecoverWordBattleTurnFromUi

	WordHelper.SendUniversalAction = function(action)
		WordHelper.lastTypingActivity = os.clock()
		WordHelper.suppressGeneratedUntil = os.clock() + 0.12
		WordHelper.lastTypedAction = action
		local button = WordHelper.keyboardByAction[action]
		if WordHelper.MappedButtonUsable(button) then
			-- Invoke the exact game key first when the executor supports signals. A
			-- coordinate click can otherwise land on Zenith when its window overlaps
			-- the mobile keyboard even though the mapped game button is underneath.
			if fireSignal and pcall(fireSignal, button.Activated) then return true end
			if virtualInputManager then
				local ok = pcall(function()
					local position, size = button.AbsolutePosition, button.AbsoluteSize
					local x, y = position.X + size.X * 0.5, position.Y + size.Y * 0.5
					virtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
					virtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
				end)
				if ok then return true end
			end
		end
		local keyCode = action == "back" and Enum.KeyCode.Backspace or KEYCODE_BY_LETTER[action]
		if not keyCode then return false end
		if virtualInputManager then
			local ok = pcall(function()
				virtualInputManager:SendKeyEvent(true, keyCode, false, game)
				virtualInputManager:SendKeyEvent(false, keyCode, false, game)
			end)
			if ok then return true end
		end
		if executorKeyPress and executorKeyRelease then
			local virtualKey = action == "back" and 0x08 or string.byte(string.upper(action))
			local ok = pcall(function()
				executorKeyPress(virtualKey)
				executorKeyRelease(virtualKey)
			end)
			if ok then return true end
		end
		if virtualUser then
			local character = action == "back" and "\b" or action
			local ok = pcall(function()
				virtualUser:CaptureController()
				virtualUser:SetKeyDown(character)
				virtualUser:SetKeyUp(character)
			end)
			if ok then return true end
		end
		return false
	end

	WordHelper.BeginAutoTyping = function(actions, sender, word, method, validator)
		WordHelper.autoTypeGeneration = WordHelper.autoTypeGeneration + 1
		local generation = WordHelper.autoTypeGeneration
		local inputEpoch = WordHelper.inputEpoch
		local typingWPM = Settings.wpm
		if Settings.legitAutoType then
			typingWPM = math.random(Settings.legitMinWPM, Settings.legitMaxWPM)
		end
		WordHelper.activeTypingWPM = typingWPM
		WordHelper.isAutoTyping = true
		WordHelper.autoTypeMethod = method
		task.spawn(function()
			local completed = true
			for index, action in ipairs(actions) do
				if not WordHelper.AutoTypeStillValid(generation, inputEpoch, validator) or not sender(action) then
					completed = false
					break
				end
				if index < #actions then
					local secondsPerCharacter = math.clamp(60 / (typingWPM * 5), 0.02, 0.4)
					task.wait(secondsPerCharacter)
				end
			end
			if WordHelper.autoTypeGeneration == generation then
				WordHelper.isAutoTyping = false
				WordHelper.autoTypeTextBox = nil
				WordHelper.autoTypeMethod = nil
				if not WordHelper.destroyed and not isRobloxMenuOpen() then
					setStatus(completed and ("Escrito: " .. word)
						or ("No se pudo completar: " .. word), completed and Config.Secondary or Config.Muted)
				end
			end
		end)
		return true, method
	end

	typeWordIntoGame = function(word)
		if WordHelper.destroyed or isRobloxMenuOpen() then return false end
		-- Every new activation invalidates delayed physical/custom-key callbacks and
		-- any older generated word before choosing the safest current target.
		cancelAutomaticTyping(true)

		if currentPlaceId == WORD_BATTLE_PLACE_ID then
			if WordHelper.localTurn ~= true and not WordHelper.RecoverWordBattleTurnFromUi() then return false end
			local prompt = normalizeLetters(WordHelper.turnPrompt)
			if prompt == "" then
				prompt = normalizeLetters(readLargePromptLetters() or "")
				if prompt == "" then
					local query = normalizeLetters(WordHelper.currentQuery)
					if #query >= 1 and #query <= 4 then prompt = query end
				end
				WordHelper.turnPrompt = prompt
			end
			if prompt == "" or string.sub(word, 1, #prompt) ~= prompt then return false end

			local currentSuffix = normalizeLetters(WordHelper.passiveBuffer)
			local desiredSuffix = string.sub(word, #prompt + 1)
			local common = 0
			while common < #currentSuffix and common < #desiredSuffix
				and string.sub(currentSuffix, common + 1, common + 1) == string.sub(desiredSuffix, common + 1, common + 1) do
				common = common + 1
			end

			local remote, sessionId = WordHelper.wordBattleRemote, WordHelper.wordBattleSessionId
			local remoteValid = false
			pcall(function()
				remoteValid = remote and remote:IsA("RemoteEvent")
					and string.find(instancePath(remote), "MatchmakingFolders", 1, true) ~= nil
			end)
			if remoteValid and type(sessionId) == "string" and string.match(sessionId, "^[%x%-]+$") then
				local actions = {}
				for index = #currentSuffix, common + 1, -1 do
					table.insert(actions, { kind = "back", value = string.sub(currentSuffix, index, index) })
				end
				for index = common + 1, #desiredSuffix do
					table.insert(actions, { kind = "letter", value = string.sub(desiredSuffix, index, index) })
				end
				if #actions == 0 then return true, "complete" end
				return WordHelper.BeginAutoTyping(actions, function(action)
					local adding = action.kind == "letter"
					local arguments = table.pack(sessionId, adding, action.value)
					WordHelper.suppressGeneratedUntil = os.clock() + 0.12
					local ok = pcall(function() remote:FireServer(sessionId, adding, action.value) end)
					if ok then handleWordBattleRemoteCall(remote, "FireServer", arguments) end
					return ok
				end, word, "remote", function()
					return WordHelper.localTurn == true
						and WordHelper.wordBattleRemote == remote
						and WordHelper.wordBattleSessionId == sessionId
				end)
			end

			-- If Zenith began during an already-running turn, updateId/type may have
			-- happened before it connected. Drive TypingSystemGui/ContextAction keys so
			-- the game's own controller emits the exact remote; never assign its
			-- TextBox.Text, which the live trace proves does not send input.
			local actions = {}
			for _ = #currentSuffix, common + 1, -1 do table.insert(actions, "back") end
			for index = common + 1, #desiredSuffix do
				table.insert(actions, string.sub(desiredSuffix, index, index))
			end
			if #actions == 0 then return true, "complete" end
			local canSend = virtualInputManager ~= nil or virtualUser ~= nil
				or (executorKeyPress ~= nil and executorKeyRelease ~= nil)
			if not canSend and fireSignal then
				canSend = true
				for _, action in ipairs(actions) do
					if not WordHelper.MappedButtonUsable(WordHelper.keyboardByAction[action]) then
						canSend = false
						break
					end
				end
			end
			if not canSend then return false end
			local shadowSuffix = currentSuffix
			return WordHelper.BeginAutoTyping(actions, function(action)
				local sent = WordHelper.SendUniversalAction(action)
				if sent then
					if action == "back" then
						shadowSuffix = string.sub(shadowSuffix, 1, math.max(0, #shadowSuffix - 1))
					else
						shadowSuffix = string.sub(shadowSuffix .. action, -32)
					end
					-- Assignment (not append) is safe whether the outgoing hook ran
					-- synchronously or this controller reports the call a frame later.
					WordHelper.passiveBuffer = shadowSuffix
					setMatchCurrentWord(localPlayer, prompt .. shadowSuffix)
				end
				return sent
			end, word, "game-keyboard", function()
				if WordHelper.localTurn ~= true then return false end
				return WordHelper.turnSource ~= "typing-ui" or WordHelper.WordBattleTypingUiActive()
			end)
		end

		if currentPlaceId == LAST_LETTER_PLACE_ID then
			-- Packet.RemoteEvent remains observation-only. Every generated character
			-- presses Overbar.Frame.Keyboard so the game's loaded InputService and
			-- Packet serializer retain ownership of validation, framing, and batching.
			if not WordHelper.LastLetterKeyboardActive() then return false end
			local visible = normalizeLetters(readLastLetterVisibleWord() or "")
			local current = Settings.autoDetect and normalizeLetters(WordHelper.lastLetterState) or visible
			if current == "" then current = visible end
			if current == "" then current = normalizeLetters(WordHelper.currentQuery) end
			if current == "" or string.sub(word, 1, #current) ~= current then return false end
			local actions = {}
			for index = #current + 1, #word do
				table.insert(actions, string.sub(word, index, index))
			end
			if #actions == 0 then return true, "complete" end
			if not fireSignal and not virtualInputManager then return false end
			for _, action in ipairs(actions) do
				local button = WordHelper.keyboardByAction[action]
				local inside = false
				pcall(function()
					inside = button and button:IsDescendantOf(WordHelper.lastLetterKeyboardRoot)
				end)
				if not inside or not WordHelper.MappedButtonUsable(button) then return false end
			end
			local shadowWord = current
			return WordHelper.BeginAutoTyping(actions, function(action)
				local sent = WordHelper.SendLastLetterAction(action)
				if sent then
					shadowWord = string.sub(shadowWord .. action, -32)
					WordHelper.passiveBuffer = shadowWord
				end
				return sent
			end, word, "last-letter-keyboard", function()
				return WordHelper.LastLetterKeyboardActive()
			end)
		end

		local textActions, textSender, textValidator, textBox = prepareExternalTextBox(word)
		if textActions then
			if #textActions == 0 then return true, "complete" end
			WordHelper.autoTypeTextBox = textBox
			return WordHelper.BeginAutoTyping(textActions, textSender, word, "TextBox", textValidator)
		end

		local initialFocus = userInputService:GetFocusedTextBox()
		local query = normalizeLetters(WordHelper.currentQuery)
		local existing = normalizeLetters(WordHelper.passiveBuffer)
		local actions = {}
		local remaining = word
		if WordHelper.detectionSource ~= "manual" and query ~= ""
			and string.sub(word, 1, #query) == query then
			remaining = string.sub(word, #query + 1)
		elseif WordHelper.detectionSource ~= "manual" then
			for _ = 1, #existing do table.insert(actions, "back") end
		end
		-- Fill the target only. Zenith never synthesizes Enter/Space or submits a word.
		for index = 1, #remaining do table.insert(actions, string.sub(remaining, index, index)) end
		if #actions == 0 then return true, "complete" end
		local canSend = virtualInputManager ~= nil or virtualUser ~= nil
			or (executorKeyPress ~= nil and executorKeyRelease ~= nil)
		if not canSend and fireSignal then
			canSend = true
			for _, action in ipairs(actions) do
				if not WordHelper.MappedButtonUsable(WordHelper.keyboardByAction[action]) then
					canSend = false
					break
				end
			end
		end
		if not canSend then return false end
		return WordHelper.BeginAutoTyping(actions, WordHelper.SendUniversalAction, word, "universal", function()
			return userInputService:GetFocusedTextBox() == initialFocus
		end)
	end

	WordHelper.TypeWordIntoGame = typeWordIntoGame
	WordHelper.IsRobloxMenuOpen = isRobloxMenuOpen

	WordHelper.focusPollAccumulator = 0
	WordHelper.visualPollAccumulator = 0
	WordHelper.menuPollAccumulator = 0
	connect(runService.Heartbeat, function(deltaTime)
		WordHelper.menuPollAccumulator = WordHelper.menuPollAccumulator + deltaTime
		if WordHelper.menuPollAccumulator >= Config.FocusPollSeconds then
			WordHelper.menuPollAccumulator = 0
			isRobloxMenuOpen()
		end
		if WordHelper.robloxMenuOpen or WordHelper.isAutoTyping then return end
		if not Settings.autoDetect then
			if WordHelper.activeExternalTextBox then disconnectFocusedTextBox() end
			return
		end
		if currentPlaceId == LAST_LETTER_PLACE_ID then
			WordHelper.visualPollAccumulator = WordHelper.visualPollAccumulator + deltaTime
			if WordHelper.visualPollAccumulator >= Config.VisualPollSeconds then
				WordHelper.visualPollAccumulator = 0
				pollVisibleGameWord()
			end
		end
		WordHelper.focusPollAccumulator = WordHelper.focusPollAccumulator + deltaTime
		if WordHelper.focusPollAccumulator < Config.FocusPollSeconds then return end
		WordHelper.focusPollAccumulator = 0
		if WordHelper.robloxMenuOpen or WordHelper.isAutoTyping then return end
		local focusedTextBox = userInputService:GetFocusedTextBox()
		if focusedTextBox and focusedTextBox ~= searchBox then
			bindFocusedTextBox(focusedTextBox)
			-- Polling as well as listening makes detection survive protected/custom
			-- TextBoxes whose property-changed signal cannot be connected to.
			syncFromFocusedTextBox(focusedTextBox)
		elseif not focusedTextBox and WordHelper.activeExternalTextBox then
			disconnectFocusedTextBox()
		end
	end)

	WordHelper.BindViewport = function()
		local camera = workspaceService.CurrentCamera
		if camera then
			connect(camera:GetPropertyChangedSignal("ViewportSize"), updateResponsivePlacement)
		end
		updateResponsivePlacement()
	end
	connect(workspaceService:GetPropertyChangedSignal("CurrentCamera"), WordHelper.BindViewport)
	WordHelper.BindViewport()

	sharedEnvironment.ZenithVocabulary = WordHelper
	sharedEnvironment.ZenithAssistant = WordHelper
	sharedEnvironment.ZenithWordHelper = WordHelper -- compatibility alias
	if type(shared) == "table" then
		shared.ZenithVocabulary = WordHelper
		shared.ZenithAssistant = WordHelper
		shared.ZenithWordHelper = WordHelper
	end

	WordHelper:SetWords(parseDictionary(FALLBACK_WORDS))

	-- Upgrade to the full cached/downloaded English dictionary in the background.
	-- Read a cached file, otherwise download it from the first working URL and cache it.
	WordHelper.LoadCachedOrDownload = function(fileName, urls)
		if readFile then
			local ok, value = pcall(function()
				if isFile and not isFile(fileName) then
					return nil
				end
				return readFile(fileName)
			end)
			if ok and type(value) == "string" and #value > 1000 then
				return value
			end
		end
		for _, url in ipairs(urls) do
			local value = httpGet(url)
			if value and #value > 1000 then
				if writeFile then
					pcall(function()
						if makeFolder then
							pcall(makeFolder, "Zenith")
						end
						writeFile(fileName, value)
					end)
				end
				return value
			end
		end
		return nil
	end

	task.spawn(function()
		setStatus("Cargando diccionario...", Config.Muted)

		local frequency = WordHelper.LoadCachedOrDownload(Config.FrequencyFile, { Config.FrequencyUrl })
		if frequency and not WordHelper.destroyed then
			local rank = {}
			local position, scanned = 0, 0
			for word in string.gmatch(frequency, "[^\r\n]+") do
				scanned = scanned + 1
				if scanned % 5000 == 0 then
					if WordHelper.destroyed then return end
					task.wait()
				end
				word = normalize(word)
				if string.match(word, "^[a-z]+$") and not rank[word] then
					position = position + 1
					rank[word] = position
				end
			end
			WordHelper.frequencyRank = rank
		end

		local content = WordHelper.LoadCachedOrDownload(Config.DictionaryFile, Config.DictionaryUrls)
		if WordHelper.destroyed then
			return
		end
		if content then
			local words = parseDictionary(content)
			if WordHelper.destroyed then return end
			if #words > 0 then
				installCleanWords(words)
				if WordHelper.destroyed then return end
				WordHelper.dictionaryLabel = "ENABLE dictionary"
			end
		end
		scheduleRefresh()
		task.wait(0.1)
		if not WordHelper.destroyed and WordHelper.currentQuery == "" then
			setStatus(content and string.format("Diccionario listo: %d palabras", #WordHelper.words)
				or "Diccionario básico (sin conexión)", Config.Muted)
		end
	end)

	-- Keep the existing twelve-hour entitlement without adding a timer/stat panel.
	standardExpiresAt = tonumber(standardExpiresAt) or (os.time() + ZenithKeyLog.DurationSeconds)
	task.spawn(function()
		while not WordHelper.destroyed do
			if standardExpiresAt <= os.time() then
				WordHelper:Destroy()
				launchKeySystem(function(newExpiresAt)
					launchWordHelper(newExpiresAt)
				end)
				return
			end
			task.wait(1)
		end
	end)

	-- Small, non-blocking entrance; there is deliberately no BlurEffect.
	WordHelper.finalScale = targetScale()
	windowScale.Scale = WordHelper.finalScale * 0.9
	updateResponsivePlacement()
	windowScale.Scale = WordHelper.finalScale * 0.9
	tween(windowScale, OPEN_TWEEN, { Scale = WordHelper.finalScale })

	return WordHelper
end

-- Bootstrap: skip the Key System while the logged key is still valid.
local function startZenith()
	local expiresAt = ZenithKeyLog.ReadExpiry()
	if expiresAt and expiresAt > os.time() then
		launchWordHelper(expiresAt)
	else
		launchKeySystem(function(newExpiresAt)
			launchWordHelper(newExpiresAt)
		end)
	end
end

startZenith()

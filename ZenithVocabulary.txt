-- Flow: if Zenith/KeySystem.log holds a key that has not expired yet, the Word Helper
-- opens directly. Otherwise the Key System opens; after "Success!" it closes itself,
-- writes the log (entered + expiry time) and opens the Word Helper.

-- Key log. Stored in the executor workspace as Zenith/KeySystem.log (append-only,
-- one line per accepted key). The newest "expires=" value is the active one.
local ZenithKeyLog = {
	Folder = "Zenith",
	File = "Zenith/KeySystem.log",
	DurationSeconds = 12 * 60 * 60,
	SessionAttribute = "ZenithStandardExpiresAt",
}

local function zenithFileFunction(name)
	local environment = getfenv and getfenv() or _G
	local value = environment[name]
	if type(value) ~= "function" and type(getgenv) == "function" then
		local ok, genv = pcall(getgenv)
		if ok and type(genv) == "table" then
			value = genv[name]
		end
	end
	return type(value) == "function" and value or nil
end

function ZenithKeyLog.FormatDate(timestamp)
	local ok, text = pcall(os.date, "%Y-%m-%d %H:%M:%S", timestamp)
	if ok and type(text) == "string" then
		return text
	end
	return tostring(timestamp)
end

function ZenithKeyLog.ReadExpiry()
	local readFile = zenithFileFunction("readfile")
	local isFile = zenithFileFunction("isfile")
	local latest = nil

	if readFile then
		local ok, content = pcall(function()
			if isFile and not isFile(ZenithKeyLog.File) then
				return nil
			end
			return readFile(ZenithKeyLog.File)
		end)
		if ok and type(content) == "string" then
			for value in string.gmatch(content, "expires=(%d+)") do
				latest = tonumber(value)
			end
		end
	end

	-- Studio / executors without file access: remember the key for this session only.
	if not latest then
		local player = game:GetService("Players").LocalPlayer
		local attribute = player and player:GetAttribute(ZenithKeyLog.SessionAttribute)
		if type(attribute) == "number" then
			latest = attribute
		end
	end
	return latest
end

function ZenithKeyLog.Record()
	local now = os.time()
	local expiresAt = now + ZenithKeyLog.DurationSeconds
	local line = string.format(
		"[%s] KEY ACCEPTED | entered=%d (%s) | expires=%d (%s)\n",
		ZenithKeyLog.FormatDate(now),
		now,
		ZenithKeyLog.FormatDate(now),
		expiresAt,
		ZenithKeyLog.FormatDate(expiresAt)
	)

	local writeFile = zenithFileFunction("writefile")
	local appendFile = zenithFileFunction("appendfile")
	local readFile = zenithFileFunction("readfile")
	local isFile = zenithFileFunction("isfile")
	local makeFolder = zenithFileFunction("makefolder")

	if writeFile then
		pcall(function()
			if makeFolder then
				pcall(makeFolder, ZenithKeyLog.Folder)
			end
			local exists = isFile and isFile(ZenithKeyLog.File)
			if exists and appendFile then
				appendFile(ZenithKeyLog.File, line)
			elseif exists and readFile then
				writeFile(ZenithKeyLog.File, readFile(ZenithKeyLog.File) .. line)
			else
				writeFile(ZenithKeyLog.File, "-- Zenith Key System log. Each line: when the key was entered and when a new key is required.\n" .. line)
			end
		end)
	end

	pcall(function()
		game:GetService("Players").LocalPlayer:SetAttribute(ZenithKeyLog.SessionAttribute, expiresAt)
	end)
	return expiresAt
end

function ZenithKeyLog.FormatRemaining(seconds)
	seconds = math.max(0, math.floor(seconds))
	if seconds <= 0 then
		return "Expired"
	end
	local hours = math.floor(seconds / 3600)
	local minutes = math.floor((seconds % 3600) / 60)
	return string.format("%02d:%02d:%02d - Remaining", hours, minutes, seconds % 60)
end

	-- Pure Luau base64 decoder, used for the bundled transparent PNGs.
	local BASE64_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
	local BASE64_LOOKUP = {}
	for index = 1, #BASE64_ALPHABET do
		BASE64_LOOKUP[string.sub(BASE64_ALPHABET, index, index)] = index - 1
	end

	local function decodeBase64(data)
		local output = table.create(math.floor(#data * 0.75))
		local outputLength = 0
		local accumulator = 0
		local groupLength = 0

		local function appendByte(value)
			outputLength = outputLength + 1
			output[outputLength] = string.char(value)
		end

		-- Decode one four-character group at a time. Keeping the accumulator small
		-- avoids floating-point overflow on the embedded 15 KB division logo.
		for index = 1, #data do
			local character = string.sub(data, index, index)
			local value = BASE64_LOOKUP[character]
			if value then
				accumulator = accumulator * 64 + value
				groupLength = groupLength + 1
				if groupLength == 4 then
					appendByte(math.floor(accumulator / 65536) % 256)
					appendByte(math.floor(accumulator / 256) % 256)
					appendByte(accumulator % 256)
					accumulator = 0
					groupLength = 0
				end
			end
		end

		-- Padding characters are skipped above, leaving two or three useful values.
		if groupLength == 2 then
			appendByte(math.floor(accumulator / 16) % 256)
		elseif groupLength == 3 then
			appendByte(math.floor(accumulator / 1024) % 256)
			appendByte(math.floor(accumulator / 4) % 256)
		end

		return table.concat(output, "", 1, outputLength)
	end

	local function normalizeAsset(value)
		if value == nil then
			return ""
		end

		local text = tostring(value)
		if text == "" or text == "0" then
			return ""
		end
		if string.find(text, "://", 1, true) then
			return text
		end
		if string.match(text, "^%d+$") then
			return "rbxassetid://" .. text
		end
		return text
	end

	local function getExecutorFunction(name)
		local value = sharedEnvironment[name]
		if type(value) == "function" then
			return value
		end
		return nil
	end

	local writeFile = getExecutorFunction("writefile")

	local function resolveEmbeddedAsset(key, fileName)
		local configured = normalizeAsset(Config.Assets[key])
		if configured ~= "" then
			return configured
		end

		if not writeFile or not getCustomAsset then
			return ""
		end

		local folder = "ZNTHMainMenuAssets"
		local path = folder .. "/" .. fileName
		local ok = pcall(function()
			-- Some executors omit isfolder and throw when makefolder sees an existing
			-- directory, so folder creation is intentionally its own ignored pcall.
			if makeFolder then
				pcall(makeFolder, folder)
			end
			-- Rewriting guarantees a stale file from an older script can never win.
			writeFile(path, decodeBase64(EMBEDDED_ASSETS[key]))
		end)

		if not ok then
			return ""
		end

		local assetOk, asset = pcall(getCustomAsset, path)
		if assetOk and type(asset) == "string" then
			return asset
		end
		return ""
	end

	-- UI utility functions.
	local function create(className, properties, parent)
		local object = Instance.new(className)
		for property, value in pairs(properties or {}) do
			object[property] = value
		end
		if parent then
			object.Parent = parent
		end
		return object
	end

	local function connect(signal, callback)
		local connection = signal:Connect(callback)
		table.insert(Menu.connections, connection)
		return connection
	end

	local function tween(object, info, goals)
		if Menu.destroyed or not object or not object.Parent then
			return nil
		end
		local animation = tweenService:Create(object, info, goals)
		animation:Play()
		return animation
	end

	local function bindPressScale(button, target)
		local scaleObject = create("UIScale", { Scale = 1 }, target or button)
		connect(button.MouseButton1Down, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 0.975 })
		end)
		connect(button.MouseButton1Up, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 1 })
		end)
		connect(button.MouseLeave, function()
			tween(scaleObject, PRESS_TWEEN, { Scale = 1 })
		end)
		return scaleObject
	end

		local function restoreKeyPrompt()
			if Menu.destroyed or Menu.keyAccepted then
				return
			end
			keyInput.Visible = false
			keyInput.Text = ""
			keyLabel.Text = "Key System"
			keyLabel.TextColor3 = Config.White
			keyLabel.Visible = true
			keyButton.Visible = true
		end

		local function submitKey()
			if Menu.destroyed or Menu.keyAccepted then
				return
			end

			if not string.find(keyInput.Text, "%S") then
				keyInput.Text = ""
				keyInput.PlaceholderText = "ENTER ANY KEY FIRST"
				tween(keyInput, FAST_TWEEN, { TextColor3 = Config.Accent })
				task.delay(0.2, function()
					if not Menu.destroyed then
						tween(keyInput, FAST_TWEEN, { TextColor3 = Config.White })
					end
				end)
				return
			end

			local acceptedKey = keyInput.Text
			Menu.keyAccepted = true
			-- The Standard countdown starts now, at the moment a key is accepted.
			-- It is written to Zenith/KeySystem.log so the Word Helper remembers it for 12 hours.
			Menu.standardExpiresAt = ZenithKeyLog.Record()
			if Menu.references.CountdownValue then
				Menu.references.CountdownValue.Text = "12 Hours - Remaining"
			end
			keyInput.Visible = false
			keyLabel.Visible = true
			keyLabel.Text = "Success!"
			keyLabel.TextColor3 = Color3.fromRGB(128, 255, 164)
			keyButton.Visible = true
			tween(keyBar, FAST_TWEEN, {
				BackgroundColor3 = Color3.fromRGB(12, 36, 23),
				BackgroundTransparency = 0.02,
			})
			if keyStroke then
				tween(keyStroke, FAST_TWEEN, { Color = Color3.fromRGB(128, 255, 164), Transparency = 0.1 })
			end

			local callback = Menu.callbacks.KeyAccepted
			if type(callback) == "function" then
				task.spawn(callback, Menu, acceptedKey)
			end

			-- Slide the Key System out, remove it completely, then open the Word Helper.
			task.delay(0.7, function()
				if Menu.destroyed then
					return
				end
				Menu:SetOpen(false)
				task.wait(OPEN_TWEEN.Time + 0.1)
				Menu:Destroy()
				if type(onKeyAccepted) == "function" then
					onKeyAccepted(Menu.standardExpiresAt)
				end
			end)
		end

		connect(keyButton.Activated, function()
			if Menu.keyAccepted then
				return
			end
			keyLabel.Visible = false
			keyButton.Visible = false
			keyInput.Visible = true
			keyInput.PlaceholderText = "TYPE ANY KEY  •  PRESS ENTER"
			task.defer(function()
				if not Menu.destroyed and keyInput.Visible then
					keyInput:CaptureFocus()
				end
			end)
		end)
		connect(keyInput.FocusLost, function(enterPressed)
			if enterPressed then
				submitKey()
			else
				restoreKeyPrompt()
			end
		end)
		connect(updatesButton.Activated, function()
			Menu:OpenUpdates()
		end)

		return root
	end

	local function captureCoreGuiState()
		if Menu.coreGuiCaptured then
			return
		end
		Menu.coreGuiCaptured = true

		local ok, coreTypes = pcall(function()
			return Enum.CoreGuiType:GetEnumItems()
		end)
		if ok then
			for _, coreType in ipairs(coreTypes) do
				if coreType ~= Enum.CoreGuiType.All then
					local stateOk, enabled = pcall(function()
						return starterGui:GetCoreGuiEnabled(coreType)
					end)
					if stateOk then
						Menu.coreGuiStates[coreType] = enabled
					end
				end
			end
		end

		local topbarOk, topbarEnabled = pcall(function()
			return starterGui:GetCore("TopbarEnabled")
		end)
		if topbarOk and type(topbarEnabled) == "boolean" then
			Menu.topbarWasEnabled = topbarEnabled
		end

		-- Executors can usually reach the modern RobloxGui container even when the
		-- legacy TopbarEnabled bridge is unavailable. Normal Studio clients simply
		-- fail this protected lookup and continue with the supported StarterGui APIs.
		pcall(function()
			local robloxGui = coreGuiService:FindFirstChild("RobloxGui")
			if robloxGui and robloxGui:IsA("ScreenGui") then
				Menu.robloxGui = robloxGui
				Menu.robloxGuiWasEnabled = robloxGui.Enabled
			end
		end)
	end

	local function setRobloxChromeHidden(hidden)
		captureCoreGuiState()
		if hidden then
			-- The modern mobile Chrome bar responds more reliably when All is disabled
			-- before individual CoreGui types and TopbarEnabled are applied.
			pcall(function()
				starterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, false)
			end)
		end
		for coreType, wasEnabled in pairs(Menu.coreGuiStates) do
			pcall(function()
				starterGui:SetCoreGuiEnabled(coreType, hidden and false or wasEnabled)
			end)
		end
		pcall(function()
			starterGui:SetCore("TopbarEnabled", hidden and false or Menu.topbarWasEnabled)
		end)
		if Menu.robloxGui then
			pcall(function()
				Menu.robloxGui.Enabled = hidden and false or Menu.robloxGuiWasEnabled
			end)
		end
	end

	local function setMenuEnvironment(active, instant)
		setRobloxChromeHidden(active)
		blurEffect.Enabled = true
		if instant then
			blurEffect.Size = active and Config.BlurSize or 0
		else
			tween(blurEffect, OPEN_TWEEN, { Size = active and Config.BlurSize or 0 })
		end

		if not active then
			task.delay(OPEN_TWEEN.Time + 0.03, function()
				if not Menu.destroyed and not Menu.open and blurEffect.Size <= 0.1 then
					blurEffect.Enabled = false
				end
			end)
		else
			-- Mobile Chrome can register after LocalScripts. Reapply during the short
			-- opening window, but only while this menu is still active.
			for _, retryDelay in ipairs({ 0.1, 0.35, 0.8, 1.4 }) do
				task.delay(retryDelay, function()
					if not Menu.destroyed and Menu.open then
						setRobloxChromeHidden(true)
					end
				end)
			end
		end
	end

	local function expApproach(current, target, speed, deltaTime)
		return current + (target - current) * (1 - math.exp(-speed * math.min(deltaTime, 0.1)))
	end

	-- Responsive sizing now controls world-space plane dimensions rather than flat GUI scale.
	function Menu:UpdateScale()

		end
		local camera = workspaceService.CurrentCamera
		local viewport = camera and camera.ViewportSize or Vector2.new(Config.ReferenceWidth, Config.ReferenceHeight)

	end

	local function placeSurface(part, canvasSize, centerX, centerY, yaw, pitch, camera, viewport)
		local depth = Config.SurfaceDepth
		local verticalWorldSize = 2 * depth * math.tan(math.rad(camera.FieldOfView * 0.5))
		local worldPerPixel = verticalWorldSize / math.max(viewport.Y, 1)
		-- Perspective expands the near edge; compressing physical width preserves
		-- the reference's readable on-screen footprint while retaining a true trapezoid.
		local width = canvasSize.X * Menu.scale * worldPerPixel * Config.SurfaceWidthCompression
		local height = canvasSize.Y * Menu.scale * worldPerPixel
		local localX = (centerX - viewport.X * 0.5) * worldPerPixel
		local localY = -(centerY - viewport.Y * 0.5) * worldPerPixel

		part.Size = Vector3.new(width, height, 0.04)
		part.CFrame = camera.CFrame
			* CFrame.new(localX, localY, -depth)
			* CFrame.Angles(math.rad(pitch), math.rad(yaw), 0)
	end

	local KEY_INPUT_CANVAS_BOUNDS = {
		left = 50,
		top = 60,
		width = 335,
		height = 54,
	}

	local function surfaceCanvasPointToViewport(part, canvasSize, x, y, camera)
		local localPoint = Vector3.new(
			(x / canvasSize.X - 0.5) * part.Size.X,
			(0.5 - y / canvasSize.Y) * part.Size.Y,
			part.Size.Z * 0.5
		)
		local worldPoint = part.CFrame:PointToWorldSpace(localPoint)
		local viewportPoint, onScreen = camera:WorldToViewportPoint(worldPoint)
		return Vector2.new(viewportPoint.X, viewportPoint.Y), onScreen and viewportPoint.Z > 0
	end

	local function triangleSign(point, first, second)
		return (point.X - second.X) * (first.Y - second.Y) - (first.X - second.X) * (point.Y - second.Y)
	end

	local function pointInTriangle(point, first, second, third)
		local firstSign = triangleSign(point, first, second)
		local secondSign = triangleSign(point, second, third)
		local thirdSign = triangleSign(point, third, first)
		local hasNegative = firstSign < 0 or secondSign < 0 or thirdSign < 0
		local hasPositive = firstSign > 0 or secondSign > 0 or thirdSign > 0
		return not (hasNegative and hasPositive)
	end

	local function pointInsideSurfaceRoot(point, part, canvasSize, left, top, width, height, camera)
		local topLeft, topLeftVisible = surfaceCanvasPointToViewport(part, canvasSize, left, top, camera)
		local topRight, topRightVisible = surfaceCanvasPointToViewport(part, canvasSize, left + width, top, camera)
		local bottomLeft, bottomLeftVisible = surfaceCanvasPointToViewport(part, canvasSize, left, top + height, camera)
		local bottomRight, bottomRightVisible = surfaceCanvasPointToViewport(part, canvasSize, left + width, top + height, camera)
		if not (topLeftVisible or topRightVisible or bottomLeftVisible or bottomRightVisible) then
			return false
		end
		return pointInTriangle(point, topLeft, topRight, bottomRight)
			or pointInTriangle(point, topLeft, bottomRight, bottomLeft)
	end

	local function pointInsideMenu(point)
		if not Menu.open then
			return false
		end
		local camera = workspaceService.CurrentCamera
		if not camera then
			return false
		end
		return pointInsideSurfaceRoot(point, leftPart, LEFT_CANVAS_SIZE, 50, 12, 335, 452, camera)
			or pointInsideSurfaceRoot(point, rightPart, RIGHT_CANVAS_SIZE, 14, 8, 323, 313, camera)
	end

	-- No separate orientation square is needed: a drag that begins directly on
	-- either menu plane rotates both planes. A small movement threshold preserves
	-- normal taps on Home, Announcements, Updates, Key System, and the footer.
	connect(userInputService.InputBegan, function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		local keyInput = Menu.references.KeySystemInput
		if keyInput and keyInput.Visible then
			return
		end
		local position = Vector2.new(input.Position.X, input.Position.Y)
		if not pointInsideMenu(position) then
			return
		end
		Menu.orientationDragging = true
		Menu.orientationDragInput = input
		Menu.orientationDragStart = position
		Menu.orientationDragOriginYaw = Menu.targetOrientationYaw
		Menu.orientationDragOriginPitch = Menu.targetOrientationPitch
		Menu.orientationDragDistance = 0
	end)

	connect(userInputService.InputChanged, function(input)
		if not Menu.orientationDragging then
			return
		end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if Menu.orientationDragInput.UserInputType == Enum.UserInputType.Touch and input ~= Menu.orientationDragInput then
			return
		end
		local position = Vector2.new(input.Position.X, input.Position.Y)
		local totalDelta = position - Menu.orientationDragStart
		Menu.orientationDragDistance = totalDelta.Magnitude
		if Menu.orientationDragDistance >= 4 then
			Menu:SetOrientation(
				Menu.orientationDragOriginYaw + totalDelta.X * 0.22,
				Menu.orientationDragOriginPitch - totalDelta.Y * 0.18
			)
		end
	end)

	connect(userInputService.InputEnded, function(input)
		if not Menu.orientationDragging then
			return
		end
		local activeInput = Menu.orientationDragInput
		local matchingMouse = activeInput.UserInputType == Enum.UserInputType.MouseButton1
			and input.UserInputType == Enum.UserInputType.MouseButton1
		local matchingTouch = activeInput.UserInputType == Enum.UserInputType.Touch and input == activeInput
		if not matchingMouse and not matchingTouch then
			return
		end
		Menu.orientationDragging = false
		Menu.orientationDragInput = nil
	end)

	local function updateKeyInputOverlay(camera)
		local keyInput = Menu.references.KeySystemInput
		if not keyInput or not keyInput.Visible then
			return
		end

		local bounds = KEY_INPUT_CANVAS_BOUNDS
		local topLeft, topLeftVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left, bounds.top, camera)
		local topRight, topRightVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left + bounds.width, bounds.top, camera)
		local bottomLeft, bottomLeftVisible = surfaceCanvasPointToViewport(leftPart, LEFT_CANVAS_SIZE, bounds.left, bounds.top + bounds.height, camera)
		local bottomRight, bottomRightVisible = surfaceCanvasPointToViewport(
			leftPart,
			LEFT_CANVAS_SIZE,
			bounds.left + bounds.width,
			bounds.top + bounds.height,
			camera
		)
		if not (topLeftVisible or topRightVisible or bottomLeftVisible or bottomRightVisible) then
			return
		end

		local centerX = (topLeft.X + topRight.X + bottomLeft.X + bottomRight.X) * 0.25
		local centerY = (topLeft.Y + topRight.Y + bottomLeft.Y + bottomRight.Y) * 0.25
		local projectedWidth = ((topRight - topLeft).Magnitude + (bottomRight - bottomLeft).Magnitude) * 0.5
		local projectedHeight = ((bottomLeft - topLeft).Magnitude + (bottomRight - topRight).Magnitude) * 0.5
		local edgeCenterLeft = (topLeft + bottomLeft) * 0.5
		local edgeCenterRight = (topRight + bottomRight) * 0.5
		local edge = edgeCenterRight - edgeCenterLeft
		local rotation = math.deg(math.atan2(edge.Y, edge.X))
		-- Back-face canvas X can project in reverse; normalize so mobile text never
		-- flips upside down while still following the bar's visible slope.
		if rotation > 90 then
			rotation = rotation - 180
		elseif rotation < -90 then
			rotation = rotation + 180
		end

		keyInput.Position = UDim2.fromOffset(centerX, centerY)
		keyInput.Size = UDim2.fromOffset(math.max(120, projectedWidth - 26), math.max(34, projectedHeight - 4))
		keyInput.Rotation = rotation
	end

	local function updateWorldPresentation(deltaTime)
		if Menu.destroyed then
			return
		end
		local camera = workspaceService.CurrentCamera
		if not camera then
			return
		end
		local viewport = camera.ViewportSize
		if viewport.X <= 0 or viewport.Y <= 0 then
			return
		end
		Menu.scale = math.clamp(math.min(viewport.X / Config.ReferenceWidth, viewport.Y / Config.ReferenceHeight), 0.62, 1.35)

		Menu.openAmount = expApproach(Menu.openAmount, Menu.targetOpenAmount, 10.5, deltaTime)
		Menu.orientationYaw = expApproach(Menu.orientationYaw, Menu.targetOrientationYaw, 9, deltaTime)
		Menu.orientationPitch = expApproach(Menu.orientationPitch, Menu.targetOrientationPitch, 9, deltaTime)

		if math.abs(Menu.openAmount - Menu.targetOpenAmount) < 0.001 then
			Menu.openAmount = Menu.targetOpenAmount
		end
		local reveal = 1 - ((1 - Menu.openAmount) ^ 3)
		local leftClosedX = -LEFT_CANVAS_SIZE.X * Menu.scale * 0.58
		local leftOpenX = 210 * Menu.scale
		local rightClosedX = viewport.X + RIGHT_CANVAS_SIZE.X * Menu.scale * 0.58
		local rightOpenX = viewport.X - 193 * Menu.scale
		local leftCenterX = leftClosedX + (leftOpenX - leftClosedX) * reveal
		local rightCenterX = rightClosedX + (rightOpenX - rightClosedX) * reveal
		-- Align the visible root tops even though each SurfaceGui has a different
		-- canvas height and internal top padding.
		local sharedRootTop = viewport.Y * 0.245
		local leftCenterY = sharedRootTop + (LEFT_CANVAS_SIZE.Y * 0.5 - 12) * Menu.scale
		local rightCenterY = sharedRootTop + (RIGHT_CANVAS_SIZE.Y * 0.5 - 8) * Menu.scale

		placeSurface(
			leftPart,
			LEFT_CANVAS_SIZE,
			leftCenterX,
			leftCenterY,
			Config.SurfaceInwardYaw + Menu.orientationYaw,
			Menu.orientationPitch,
			camera,
			viewport
		)
		placeSurface(
			rightPart,
			RIGHT_CANVAS_SIZE,
			rightCenterX,
			rightCenterY,
			-Config.SurfaceInwardYaw + Menu.orientationYaw,
			Menu.orientationPitch,
			camera,
			viewport
		)
		updateKeyInputOverlay(camera)

		local shouldRender = Menu.openAmount > 0.005 or Menu.targetOpenAmount > 0
		leftSurface.Enabled = shouldRender
		rightSurface.Enabled = shouldRender
	end

	connect(runService.RenderStepped, updateWorldPresentation)

		if self.destroyed then
			return
		end

		shouldOpen = shouldOpen == true
		self.open = shouldOpen
		self.targetOpenAmount = shouldOpen and 1 or 0

		if shouldOpen then
			leftSurface.Enabled = true
			rightSurface.Enabled = true
		else
			self.orientationDragging = false
			self.orientationDragInput = nil
		end
		if not shouldOpen and self.references.KeySystemInput and self.references.KeySystemInput.Visible then

			self.references.KeySystemInput.Visible = false
			self.references.KeySystemButton.Visible = true
			self.references.KeySystemLabel.Visible = true
		end

		if instant then
			self.openAmount = self.targetOpenAmount
			updateWorldPresentation(0)
		end
		setMenuEnvironment(shouldOpen, instant == true)
	end

	local function sanitizeDescription(value)
		if type(value) ~= "string" or value == "" then
			return Config.DefaultDescription
		end
		local sanitized = string.gsub(value, "[%c]+", " ")
		sanitized = string.gsub(sanitized, "%s+", " ")

		-- Truncate by UTF-8 character boundary so an emoji or accented title can
		-- never leave invalid text in the Roblox label.
		local characterCount = utf8.len(sanitized)
		if characterCount and characterCount > 175 then
			local byteAfterLimit = utf8.offset(sanitized, 173)
			if byteAfterLimit then
				sanitized = string.sub(sanitized, 1, byteAfterLimit - 1) .. "..."
			end
		end
		return sanitized
	end

	local function formatRemaining(seconds)
		seconds = math.max(0, math.floor(seconds + 0.5))
		if seconds <= 0 then
			return "Expired"
		end
		if seconds >= Config.StandardDurationSeconds then
			return "12 Hours - Remaining"
		end
		local hours = math.floor(seconds / 3600)
		local minutes = math.floor((seconds % 3600) / 60)
		local remainingSeconds = seconds % 60
		return string.format("%02d:%02d:%02d - Remaining", hours, minutes, remainingSeconds)
	end

	local function formatPlayed(seconds)
		seconds = math.max(0, math.floor(seconds))
		local days = math.floor(seconds / 86400)
		local hours = math.floor((seconds % 86400) / 3600)
		local minutes = math.floor((seconds % 3600) / 60)
		return string.format("%dd %dh %dm", days, hours, minutes)
	end

	local function updateTimers()
		local now = getServerTime()
		Menu.references.CountdownValue.Text = (Menu.keyAccepted and Menu.standardExpiresAt) and formatRemaining(Menu.standardExpiresAt - os.time()) or "Pending..."
		Menu.references.TimePlayedValue.Text = formatPlayed(now - startedAt)
	end

	connect(runService.Heartbeat, function(deltaTime)
		timerAccumulator = timerAccumulator + deltaTime
		if timerAccumulator >= 1 then
			timerAccumulator = timerAccumulator % 1
			updateTimers()
		end
	end)
	updateTimers()

	-- RightShift always toggles the complete presentation, even while the key box is focused.
	connect(userInputService.InputBegan, function(input, gameProcessed)
		if input.KeyCode == Enum.KeyCode.RightShift then
			Menu:Toggle()
			return
		end
		if gameProcessed or userInputService:GetFocusedTextBox() then
			return
		end

		if input.KeyCode == Enum.KeyCode.Z then
			if not Menu.open then
				Menu:SetOpen(true)
			end
			Menu:OpenHome()
		elseif input.KeyCode == Enum.KeyCode.F then
			Menu:OpenAnnouncements()
		elseif input.KeyCode == Enum.KeyCode.X then
			Menu:PulseHotkey("X")
			Menu:CopyKeyLink()
			local callback = Menu.callbacks.KeyLink or Menu.callbacks.X
			if type(callback) == "function" then
				task.spawn(callback, Menu)
			end
		elseif input.KeyCode == Enum.KeyCode.C then
			Menu:OpenDiscord()
		end
	end)

	local function bindViewport()
		local camera = workspaceService.CurrentCamera
		if camera then
			connect(camera:GetPropertyChangedSignal("ViewportSize"), function()
				Menu:UpdateScale()
			end)
		end
	end

	connect(workspaceService:GetPropertyChangedSignal("CurrentCamera"), function()
		Menu:UpdateScale()
	end)

	function Menu:Destroy()
		if self.destroyed then
			return
		end

		if self.references.KeySystemInput and self.references.KeySystemInput:IsFocused() then
			self.references.KeySystemInput:ReleaseFocus()
		end
		if self.coreGuiCaptured then
			setRobloxChromeHidden(false)
		end
		self.destroyed = true

		for _, connection in ipairs(self.connections) do
			if connection and connection.Connected then
				connection:Disconnect()
			end
		end
		table.clear(self.connections)

		for _, referenceName in ipairs({ "ScreenGui", "LeftSurface", "RightSurface", "SurfaceFolder", "BlurEffect" }) do
			local object = self.references[referenceName]
			if object and object.Parent then
				object:Destroy()
			end
		end

		if sharedEnvironment.ZNTHMainMenu == self then
			sharedEnvironment.ZNTHMainMenu = nil
		end
	end

	bindViewport()
	Menu:UpdateScale()
	bindCurrentGame()
	bindLocalPlayer()
	sharedEnvironment.ZNTHMainMenu = Menu

	-- Start just off-screen, then reproduce the two-sided entrance from the reference.
	Menu:SetOpen(false, true)
	task.defer(function()
		if not Menu.destroyed then
			Menu:SetOpen(true)
		end
	end)

	return Menu
end

	local function normalize(text)
		text = string.lower(tostring(text or ""))
		text = string.gsub(text, "^%s+", "")
		text = string.gsub(text, "%s+$", "")
		return text
	end

	local function normalizeLetters(text)
		return string.gsub(normalize(text), "[^a-z]", "")
	end

	local function escapeRichText(text)
		text = string.gsub(text, "&", "&amp;")
		text = string.gsub(text, "<", "&lt;")
		text = string.gsub(text, ">", "&gt;")
		text = string.gsub(text, '"', "&quot;")
		return text
	end

	local function extractCurrentFragment(text, cursorPosition)
		text = string.lower(tostring(text or ""))
		cursorPosition = tonumber(cursorPosition)
		if cursorPosition and cursorPosition > 0 then
			text = string.sub(text, 1, math.min(#text, cursorPosition - 1))
		end
		return string.match(text, "([a-z]+)$") or ""
	end

	local WORD_BASE64_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
	local WORD_BASE64_LOOKUP = {}
	for index = 1, #WORD_BASE64_ALPHABET do
		WORD_BASE64_LOOKUP[string.sub(WORD_BASE64_ALPHABET, index, index)] = index - 1
	end

	local function decodeWordAsset(data)
		local output = table.create(math.floor(#data * 0.75))
		local outputLength, accumulator, groupLength = 0, 0, 0
		local function appendByte(value)
			outputLength = outputLength + 1
			output[outputLength] = string.char(value)
		end
		for index = 1, #data do
			local value = WORD_BASE64_LOOKUP[string.sub(data, index, index)]
			if value then
				accumulator = accumulator * 64 + value
				groupLength = groupLength + 1
				if groupLength == 4 then
					appendByte(math.floor(accumulator / 65536) % 256)
					appendByte(math.floor(accumulator / 256) % 256)
					appendByte(accumulator % 256)
					accumulator, groupLength = 0, 0
				end
			end
		end
		if groupLength == 2 then
			appendByte(math.floor(accumulator / 16) % 256)
		elseif groupLength == 3 then
			appendByte(math.floor(accumulator / 1024) % 256)
			appendByte(math.floor(accumulator / 4) % 256)
		end
		return table.concat(output, "", 1, outputLength)
	end

	local function normalizeWordAsset(value)
		local text = tostring(value or "")
		if text == "" or text == "0" then return "" end
		if string.find(text, "://", 1, true) then return text end
		if string.match(text, "^%d+$") then return "rbxassetid://" .. text end
		return text
	end

	local function resolveWordAsset(key, fileName)
		local configured = normalizeWordAsset(Config.Assets[key])
		if configured ~= "" then return configured end
		if not writeFile or not getCustomAsset then return "" end
		local path = "Zenith/" .. fileName
		local wrote = pcall(function()
			if makeFolder then pcall(makeFolder, "Zenith") end
			writeFile(path, decodeWordAsset(EMBEDDED_WORD_ASSETS[key]))
		end)
		if not wrote then return "" end
		local ok, asset = pcall(getCustomAsset, path)
		return ok and type(asset) == "string" and asset or ""
	end

	local function parseDictionary(content)
		local output = {}
		local seen = {}
		local sorted = true
		local previousWord = nil
		local processed = 0
		for rawWord in string.gmatch(tostring(content or ""), "[^\r\n]+") do
			processed = processed + 1
			if processed % 5000 == 0 then
				if WordHelper.destroyed then return {} end
				task.wait()
			end
			local word = normalize(rawWord)
			if #word >= 2 and #word <= 32 and string.match(word, "^[a-z]+$") and not seen[word] then
				seen[word] = true
				if previousWord and word < previousWord then
					sorted = false
				end
				previousWord = word
				table.insert(output, word)
			end
		end
		if not sorted then
			table.sort(output)
		end
		return output
	end

	local function buildBuckets(words)
		local firstLetterBuckets = {}
		local prefixBuckets = {}
		for index, word in ipairs(words) do
			if index % 5000 == 0 then
				if WordHelper.destroyed then return {}, {} end
				task.wait()
			end
			local first = string.sub(word, 1, 1)
			local firstBucket = firstLetterBuckets[first]
			if not firstBucket then
				firstBucket = {}
				firstLetterBuckets[first] = firstBucket
			end
			table.insert(firstBucket, word)

			if #word >= 2 then
				local prefix = string.sub(word, 1, 2)
				local prefixBucket = prefixBuckets[prefix]
				if not prefixBucket then
					prefixBucket = {}
					prefixBuckets[prefix] = prefixBucket
				end
				table.insert(prefixBucket, word)
			end
		end
		return firstLetterBuckets, prefixBuckets
	end

	local function installCleanWords(words)
		WordHelper.words = words
		WordHelper.firstLetterBuckets, WordHelper.prefixBuckets = buildBuckets(words)
		local wordSet = {}
		for index, word in ipairs(words) do
			if index % 5000 == 0 then
				if WordHelper.destroyed then return end
				task.wait()
			end
			wordSet[word] = true
		end
		WordHelper.wordSet = wordSet
		local added = false
		for word in pairs(WordHelper.pendingUsedWords) do
			if wordSet[word] and not WordHelper.usedWords[word] then
				WordHelper.usedWords[word] = true
				WordHelper.usedCount = WordHelper.usedCount + 1
				WordHelper.pendingUsedWords[word] = nil
				added = true
			end
		end
		if added then updateUsedChip() end
	end

	-- Highlights the typed part wherever it appears (start for "Empieza con",

	local function formatWord(word, query)
		local startIndex = (query ~= "" and string.find(word, query, 1, true)) or 1
		local before = escapeRichText(string.sub(word, 1, startIndex - 1))
		local match = escapeRichText(string.sub(word, startIndex, startIndex + #query - 1))
		local after = escapeRichText(string.sub(word, startIndex + #query))
		return '<font color="#B5B9BF">' .. before .. '</font><font color="#F2F3F5"><b>' .. match
			.. '</b></font><font color="#B5B9BF">' .. after .. "</font>"
	end

	local function decodeHtml(text)
		text = string.gsub(text, "<.->", "")
		text = string.gsub(text, "&nbsp;", " ")
		text = string.gsub(text, "&quot;", '"')
		text = string.gsub(text, "&#39;", "'")
		text = string.gsub(text, "&lt;", "<")
		text = string.gsub(text, "&gt;", ">")
		text = string.gsub(text, "&amp;", "&")
		text = string.gsub(text, "%s+", " ")
		text = string.gsub(text, "^%s+", "")
		text = string.gsub(text, "[%s:]+$", "")
		return text
	end

	local function fetchDefinition(word)
		local page = httpGet(Config.CambridgeUrl .. word)
		if page then
			local raw = string.match(page, 'class="def ddef_d db">(.-)</div>')
			if raw then
				local definition = decodeHtml(raw)
				if definition ~= "" then
					return definition, "Cambridge"
				end
			end
		end
		local json = httpGet(Config.BackupDefinitionUrl .. word)
		if json then
			local ok, data = pcall(function()
				return game:GetService("HttpService"):JSONDecode(json)
			end)
			local entry = ok and type(data) == "table" and data[1]
			local meaning = entry and entry.meanings and entry.meanings[1]
			local first = meaning and meaning.definitions and meaning.definitions[1]
			if first and type(first.definition) == "string" then
				return first.definition, "Free Dictionary"
			end
		end
		return nil, nil
	end

	-- If the word came from a real TextBox (chat or a game box), finish the word there.
	local function completeExternalTextBox(word)
		local textBox = WordHelper.lastExternalTextBox
		if WordHelper.detectionSource ~= "external" or not textBox or not textBox.Parent then
			return false
		end
		local ok = pcall(function()
			local text = textBox.Text
			local fragment = string.match(text, "([%a]+)$") or ""
			if string.lower(fragment) ~= WordHelper.currentQuery then
				error("fragment changed")
			end
			local completed = string.sub(text, 1, #text - #fragment) .. word
			textBox.Text = completed
			textBox.CursorPosition = #completed + 1
		end)
		return ok
	end

	function WordHelper:SelectWord(word)
		if self.destroyed then
			return false
		end
		self.selectedWord = word
		styleRows()

		local copied = false
		if clipboardWriter then
			copied = pcall(clipboardWriter, word)
		end
		local completed = completeExternalTextBox(word)
		if completed then
			setStatus("Completado: " .. word .. (copied and "  (copiado)" or ""), Config.Secondary)
		else
			setStatus(copied and ("Copiado: " .. word) or ("Seleccionado: " .. word), copied and Config.Secondary or Config.Muted)
		end
		showDefinition(word)

		local callback = self.callbacks.WordSelected
		if type(callback) == "function" then
			task.spawn(callback, self, word, copied)
		end
		return copied
	end

	local function refreshWords()
		if WordHelper.destroyed then
			return
		end
		clearRows()

		local query = WordHelper.currentQuery
		if query == "" then
			emptyLabel.Visible = true
			emptyLabel.Text = "Escribe una letra en el chat o juego"
			wordList.CanvasPosition = Vector2.new(0, 0)
			setStatus("Esperando texto...", Config.Muted)
			WordHelper.bestWord = nil
			updatePreview()
			return
		end

		local containsMode = WordHelper.searchMode == "contains"
		local used = WordHelper.usedWords
		local matches = {}
		if containsMode then
			-- "Contiene": word-bomb style games give a syllable that can be anywhere.
			for _, word in ipairs(WordHelper.words) do
				if not used[word] and string.find(word, query, 1, true) then
					table.insert(matches, word)
				end
			end
		else
			local source
			if #query == 1 then
				source = WordHelper.firstLetterBuckets[query] or {}
			else
				source = WordHelper.prefixBuckets[string.sub(query, 1, 2)] or {}
			end
			for _, word in ipairs(source) do
				if not used[word] and string.sub(word, 1, #query) == query then
					table.insert(matches, word)
				end
			end
		end
		local totalMatches = #matches

		-- Sort: Comunes = most frequent first; Largas = longest first; Cortas = shortest first.
		local rank = WordHelper.frequencyRank
		local sortMode = WordHelper.sortMode
		table.sort(matches, function(a, b)
			if sortMode == "long" and #a ~= #b then
				return #a > #b
			elseif sortMode == "short" and #a ~= #b then
				return #a < #b
			end
			local rankA, rankB = rank[a] or math.huge, rank[b] or math.huge
			if rankA ~= rankB then
				return rankA < rankB
			end
			if #a ~= #b then
				return #a < #b
			end
			return a < b
		end)

		-- Preview word: the shortest common word that is clearly longer than the
		-- prefix ("s" -> short word, "sim" -> "simple"), best frequency on ties.
		local best, bestLength, bestRank = nil, math.huge, math.huge
		for _, word in ipairs(matches) do
			if #word >= #query + 2 then
				local wordRank = rank[word] or math.huge
				local common = wordRank ~= math.huge
				local bestCommon = bestRank ~= math.huge
				if (common and not bestCommon)
					or (common == bestCommon and (#word < bestLength or (#word == bestLength and wordRank < bestRank))) then
					best, bestLength, bestRank = word, #word, wordRank
				end
			end
		end
		WordHelper.bestWord = best or matches[1]
		updatePreview()

		local resultLimit = isTouchDevice and Config.MobileMaximumResults or Config.MaximumResults
		WordHelper.resultLimit = resultLimit
		local shown = math.min(totalMatches, resultLimit)
		for index = 1, shown do
			createWordRow(matches[index], query, index)
		end

		emptyLabel.Visible = shown == 0
		emptyLabel.Text = "No se encontraron palabras"
		wordList.CanvasPosition = Vector2.new(0, 0)
		if shown == 0 then
			setStatus('Sin resultados para "' .. query .. '"', Config.Muted)
		elseif totalMatches > shown then
			setStatus(string.format("%d de %d resultados", shown, totalMatches), Config.Muted)
		else
			setStatus(string.format("%d resultado%s", shown, shown == 1 and "" or "s"), Config.Muted)
		end
	end

	local function scheduleRefresh()
		WordHelper.refreshVersion = WordHelper.refreshVersion + 1
		local version = WordHelper.refreshVersion
		task.delay(isTouchDevice and 0.075 or 0.045, function()
			if not WordHelper.destroyed and version == WordHelper.refreshVersion then
				refreshWords()
			end
		end)
	end

	local synchronizingSearch = false
	local function applyDetectedQuery(query, source, reflectInSearch)
		if WordHelper.destroyed then
			return
		end
		query = normalizeLetters(query)
		local queryChanged = query ~= WordHelper.currentQuery
		WordHelper.currentQuery = query
		WordHelper.detectionSource = source or "automatic"
		local searchFocused = false
		pcall(function() searchFocused = searchBox:IsFocused() end)
		if reflectInSearch and not searchFocused and searchBox.Text ~= query then
			synchronizingSearch = true
			searchBox.Text = query
			synchronizingSearch = false
		end
		if queryChanged and WordHelper.selectedWord then
			WordHelper.definitionVersion = WordHelper.definitionVersion + 1
			WordHelper.selectedWord = nil
			definitionWord.Text = "Definición"
			definitionText.Text = "Toca una palabra para ver su significado."
			definitionText.TextColor3 = Config.Muted
		end
		if query == "" then
			detectionLabel.Text = "Detección automática activa"
		else
			local SOURCE_LABELS = {
				manual = "manual",
				external = "chat / caja de texto",
				passive = "teclado",
				remote = "teclado del juego",
				visual = "interfaz del juego",
				turn = "tu turno",
			}
			local sourceLabel = SOURCE_LABELS[source] or "detectado automáticamente"
			detectionLabel.Text = "Texto: " .. query .. "  |  " .. sourceLabel
		end
		if queryChanged then scheduleRefresh() end
	end

	local function queryForTypedBuffer(typed)
		typed = normalizeLetters(typed)
		local prompt = WordHelper.turnPrompt
		if WordHelper.localTurn == true and prompt ~= "" then
			-- The confirmed game pre-fills the required letters and transmits only
			-- what the local player adds. Build "s" + "o" as "so" exactly once.
			return string.sub(prompt .. typed, 1, 32)
		end
		return typed
	end

	function WordHelper:SetWords(words)
		if self.destroyed or type(words) ~= "table" then
			return
		end
		local clean = {}
		local seen = {}
		for _, value in ipairs(words) do
			local word = normalize(value)
			if #word >= 2 and #word <= 32 and string.match(word, "^[a-z]+$") and not seen[word] then
				seen[word] = true
				table.insert(clean, word)
			end
		end
		table.sort(clean)
		installCleanWords(clean)
		scheduleRefresh()
	end

	local function bindFocusedTextBox(textBox)
		if textBox == searchBox then
			disconnectFocusedTextBox()
			return
		end
		if textBox == WordHelper.activeExternalTextBox then
			return
		end
		disconnectFocusedTextBox()
		if not textBox then
			return
		end
		local isTextBox = false
		pcall(function()
			isTextBox = textBox:IsA("TextBox")
		end)
		if not isTextBox then
			return
		end

		WordHelper.activeExternalTextBox = textBox
		WordHelper.lastExternalTextBox = textBox
		local ok, textConnection = pcall(function()
			return textBox:GetPropertyChangedSignal("Text"):Connect(function()
				syncFromFocusedTextBox(textBox)
			end)
		end)
		if ok and textConnection then
			table.insert(WordHelper.focusConnections, textConnection)
		end
		local cursorOk, cursorConnection = pcall(function()
			return textBox:GetPropertyChangedSignal("CursorPosition"):Connect(function()
				syncFromFocusedTextBox(textBox)
			end)
		end)
		if cursorOk and cursorConnection then
			table.insert(WordHelper.focusConnections, cursorConnection)
		end
		local focusOk, focusConnection = pcall(function()
			return textBox.FocusLost:Connect(function(enterPressed)
				if enterPressed and markUsed then
					local okText, text = pcall(function() return textBox.Text end)
					if okText then markUsed(extractCurrentFragment(text, #tostring(text) + 1)) end
				end
			end)
		end)
		if focusOk and focusConnection then
			table.insert(WordHelper.focusConnections, focusConnection)
		end
		syncFromFocusedTextBox(textBox)
	end

	function WordHelper:Destroy()
		if self.destroyed then
			return
		end
		if searchBox:IsFocused() then
			searchBox:ReleaseFocus()
		end
		self.destroyed = true
		if self.remoteHook and self.remoteHook.handler == self.ProcessRemoteCall then
			self.remoteHook.handler = nil
		end
		disconnectFocusedTextBox()
		disconnectRows()
		for _, connection in ipairs(self.connections) do
			if connection and connection.Connected then
				connection:Disconnect()
			end
		end
		table.clear(self.connections)
		if screenGui and screenGui.Parent then
			screenGui:Destroy()
		end
		local replacement = sharedEnvironment.ZenithChess
		if type(replacement) ~= "table" or replacement.destroyed then replacement = nil end
		if sharedEnvironment.ZenithVocabulary == self then sharedEnvironment.ZenithVocabulary = nil end
		if sharedEnvironment.ZenithAssistant == self then sharedEnvironment.ZenithAssistant = replacement end
		if sharedEnvironment.ZenithWordHelper == self then sharedEnvironment.ZenithWordHelper = replacement end
		if type(shared) == "table" then
			local sharedReplacement = shared.ZenithChess
			if type(sharedReplacement) ~= "table" or sharedReplacement.destroyed then sharedReplacement = replacement end
			if shared.ZenithVocabulary == self then shared.ZenithVocabulary = nil end
			if shared.ZenithAssistant == self then shared.ZenithAssistant = sharedReplacement end
			if shared.ZenithWordHelper == self then shared.ZenithWordHelper = sharedReplacement end
		end
	end

	local function targetScale()
		local viewport = currentViewport()
		local preferredScale = isTouchDevice and Config.MobileUIScale or Config.UIScale
		return math.clamp(math.min(
			(viewport.X - 12) / Config.WindowWidth,
			(viewport.Y - 12) / Config.WindowHeight,
			preferredScale
		), Config.MinimumUIScale, preferredScale)
	end

	local function clampWindowPosition(x, y)
		local viewport = currentViewport()
		local scale = windowScale.Scale
		local width = Config.WindowWidth * scale
		local headerHeight = Config.HeaderHeight * scale
		-- Deliberately allow most of the menu outside any edge. Keep only a small
		-- draggable part of the header reachable so it can always be recovered.
		local visibleX = math.min(width, math.max(48, 56 * scale))
		local visibleHeaderY = math.min(headerHeight, math.max(22, 24 * scale))
		local minimumX = -width + visibleX
		local maximumX = viewport.X - visibleX
		local minimumY = -headerHeight + visibleHeaderY
		local maximumY = viewport.Y - visibleHeaderY
		return math.clamp(x, minimumX, maximumX), math.clamp(y, minimumY, maximumY)
	end

	updateResponsivePlacement = function()
		if WordHelper.destroyed then
			return
		end
		local viewport = currentViewport()
		local scale = targetScale()
		windowScale.Scale = scale
		if not WordHelper.hasUserPosition then
			local x = viewport.X - Config.WindowWidth * scale - 16
			local y = (viewport.Y - Config.WindowHeight * scale) * 0.5
			x, y = clampWindowPosition(x, y)
			window.Position = UDim2.fromOffset(x, y)
		else
			local x, y = clampWindowPosition(window.Position.X.Offset, window.Position.Y.Offset)
			window.Position = UDim2.fromOffset(x, y)
		end
	end

	-- Dragging stays entirely in ScreenGui offset space. Mixing AbsolutePosition

		local topLeft, size = guiObject.AbsolutePosition, guiObject.AbsoluteSize
		return point.X >= topLeft.X and point.X <= topLeft.X + size.X
			and point.Y >= topLeft.Y and point.Y <= topLeft.Y + size.Y
	end

	-- Shared buffer for input that never reaches a real TextBox.
	-- action: a single letter, "back" (delete one) or "end" (space/enter = new word).
	-- A submitted dictionary word is remembered so it is hidden next time
	-- (most word games forbid repeating a word).
	markUsed = function(word)
		word = normalizeLetters(word)
		if #word < 2 or WordHelper.usedWords[word] then return end
		if not (WordHelper.wordSet and WordHelper.wordSet[word]) then
			-- Keep early submissions until the full dictionary replaces the fallback.
			WordHelper.pendingUsedWords[word] = true
			return
		end
		WordHelper.pendingUsedWords[word] = nil
		WordHelper.usedWords[word] = true
		WordHelper.usedCount = WordHelper.usedCount + 1
		updateUsedChip()
	end

	local function passiveType(action, resetSeconds)
		-- In the confirmed two-player game, consumed physical keys and world-space
		-- keyboard buttons exist for both players. Never treat either as input unless
		-- MatchStatus or a local outgoing typing call has proved that this is our turn.
		if currentPlaceId == WORD_BOMB_PLACE_ID and WordHelper.localTurn ~= true then return end
		local now = os.clock()
		if now - WordHelper.lastPassiveInput > resetSeconds then
			WordHelper.passiveBuffer = ""
		end
		WordHelper.lastPassiveInput = now
		if action == "back" then
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer, 1, math.max(0, #WordHelper.passiveBuffer - 1))
		elseif action == "end" then
			if currentPlaceId == WORD_BOMB_PLACE_ID then
				-- The server's confirm event is authoritative. Invalid words leave the
				-- same turn active, so do not hide a word before confirmation.
				WordHelper.awaitingSubmission = true
				WordHelper.passiveBuffer = ""
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			else
				markUsed(WordHelper.passiveBuffer)
				WordHelper.passiveBuffer = ""
				applyDetectedQuery("", "passive", true)
			end
			return
		else
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer .. action, -32)
		end
		applyDetectedQuery(queryForTypedBuffer(WordHelper.passiveBuffer), "passive", true)
	end

	-- The confirmed trace emits MatchStatus as:
	--   type(prompt, typingRemote, startedAt, Player, turnConfig)
	--   confirm(Player, word, ...) / wrongAnswer(Player, reason, ...)
	-- The Player instance is authoritative and stops opponent words from ever
	-- becoming Zenith's query.
	local function processMatchStatus(...)
		if currentPlaceId ~= WORD_BOMB_PLACE_ID then return end
		local arguments = table.pack(...)
		local eventType = type(arguments[1]) == "string" and string.lower(arguments[1]) or ""
		local owner = nil
		for index = 2, arguments.n do
			local value = arguments[index]
			if value == localPlayer then
				owner = value
				break
			end
			pcall(function()
				if not owner and typeof(value) == "Instance" and value:IsA("Player") then owner = value end
			end)
		end
		if not owner then
			for index = 2, arguments.n do
				local value = arguments[index]
				if type(value) == "table" then
					local ownerName = value.Player or value.PlayerName or value.Name
					if (type(ownerName) == "string" and (ownerName == localPlayer.Name or ownerName == localPlayer.DisplayName))
						or (value.UserId ~= nil and localPlayer.UserId ~= nil and value.UserId == localPlayer.UserId) then
						owner = localPlayer
						break
					end
				end
			end
		end

		local function waitForLocalTurn()
			WordHelper.localTurn = false
			WordHelper.turnPrompt = ""
			WordHelper.passiveBuffer = ""
			WordHelper.awaitingSubmission = false
			applyDetectedQuery("", "turn", true)
			detectionLabel.Text = "Esperando tu turno"
		end

		if eventType == "type" then
			WordHelper.lastTurnEventAt = os.clock()
			local prompt = normalizeLetters(arguments[2])
			local ownTurn = owner == localPlayer
			WordHelper.turnOwner = owner
			WordHelper.turnSource = "match"
			WordHelper.awaitingSubmission = false
			WordHelper.searchMode = "prefix"
			modeChip.Text = "Empieza con"
			if ownTurn then
				WordHelper.localTurn = true
				local recentVisual = os.clock() - WordHelper.visiblePromptAt <= 0.75 and WordHelper.visiblePrompt or ""
				WordHelper.turnPrompt = prompt ~= "" and prompt or recentVisual
				WordHelper.passiveBuffer = ""
				WordHelper.lastPassiveInput = os.clock()
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			else
				waitForLocalTurn()
			end
		elseif eventType == "confirm" and owner == localPlayer then
			WordHelper.lastTurnEventAt = os.clock()
			if type(arguments[3]) == "string" then markUsed(arguments[3]) end
			WordHelper.turnSource = "match"
			waitForLocalTurn()
		elseif eventType == "wronganswer" and owner == localPlayer then
			WordHelper.lastTurnEventAt = os.clock()
			WordHelper.awaitingSubmission = false
			local reason = type(arguments[3]) == "string" and string.lower(arguments[3]) or ""
			if reason == "healthsubtracted" then
				WordHelper.turnSource = "match"
				waitForLocalTurn()
			else
				-- WordDoesNotExist (and equivalent retryable errors) does not pass
				-- the turn. Preserve the required letters and offer another word.
				WordHelper.localTurn = true
				WordHelper.turnOwner = localPlayer
				WordHelper.turnSource = "match"
				WordHelper.passiveBuffer = ""
				if WordHelper.turnPrompt == "" and os.clock() - WordHelper.visiblePromptAt <= 0.75 then
					WordHelper.turnPrompt = WordHelper.visiblePrompt
				end
				applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			end
		end
	end

	WordHelper.ProcessMatchStatus = processMatchStatus
	local boundMatchStatus = setmetatable({}, { __mode = "k" })
	local function bindMatchStatus(remote)
		if currentPlaceId ~= WORD_BOMB_PLACE_ID or not remote or boundMatchStatus[remote] then return end
		local valid = false
		pcall(function() valid = remote:IsA("RemoteEvent") and remote.Name == "MatchStatus" end)
		if not valid then return end
		boundMatchStatus[remote] = true
		pcall(function() connect(remote.OnClientEvent, processMatchStatus) end)
	end
	if currentPlaceId == WORD_BOMB_PLACE_ID then
		bindMatchStatus(replicatedStorage:FindFirstChild("MatchStatus", true))
		pcall(function() connect(replicatedStorage.DescendantAdded, bindMatchStatus) end)
	end

	local function bridgeBlockedSearchInput(input)
		local letter = LETTER_BY_KEYCODE[input.KeyCode]
		local isBackspace = input.KeyCode == Enum.KeyCode.Backspace
		if not letter and not isBackspace then return false end
		local inputAt = os.clock()
		WordHelper.manualCaptureUntil = inputAt + 2.5
		local before = searchBox.Text
		local cursor = tonumber(searchBox.CursorPosition) or (#before + 1)
		if cursor <= 0 then cursor = #before + 1 end
		local nativeAlreadyHandled = WordHelper.lastNativeSearchEditAt > 0
			and inputAt - WordHelper.lastNativeSearchEditAt <= 0.04
		-- Give Roblox's normal TextBox editor a frame first. Only synthesize when no
		-- native edit occurred either immediately before or after InputBegan.
		task.delay(0.035, function()
			if WordHelper.destroyed or nativeAlreadyHandled or searchBox.Text ~= before
				or (not searchBox:IsFocused() and os.clock() > WordHelper.manualCaptureUntil) then
				return
			end
			local updated, nextCursor = before, cursor
			if letter then
				updated = string.sub(before, 1, cursor - 1) .. letter .. string.sub(before, cursor)
				nextCursor = cursor + 1
			elseif isBackspace and cursor > 1 then
				updated = string.sub(before, 1, cursor - 2) .. string.sub(before, cursor)
				nextCursor = cursor - 1
			end
			updated = string.sub(normalizeLetters(updated), 1, 32)
			synchronizingSearch = true
			searchBox.Text = updated
			synchronizingSearch = false
			WordHelper.passiveBuffer = updated
			applyDetectedQuery(updated, "manual", false)
			pcall(function() searchBox.CursorPosition = math.min(#updated + 1, nextCursor) end)
		end)
		return true
	end

	connect(userInputService.TextBoxFocusReleased, function(textBox)
		if textBox ~= searchBox then return end
		if WordHelper.destroyed or WordHelper.minimized or not WordHelper.visible then
			WordHelper.manualCaptureUntil = 0
			return
		end
		WordHelper.manualCaptureUntil = os.clock() + 1.25
	end)

	local function exactRemoteHandledNear(action, inputAt)
		return currentPlaceId == WORD_BOMB_PLACE_ID
			and WordHelper.lastRemoteAction == action
			and math.abs(WordHelper.lastRemoteActionAt - inputAt) <= 0.55
	end

	connect(userInputService.InputBegan, function(input, gameProcessed)
		if input.KeyCode == Enum.KeyCode.RightShift then
			WordHelper:Toggle()
			return
		end

		local focusedTextBox = userInputService:GetFocusedTextBox()
		if focusedTextBox then
			if focusedTextBox == searchBox then
				bridgeBlockedSearchInput(input)
			else
				bindFocusedTextBox(focusedTextBox)
			end
			return
		end
		-- Some custom-keyboard controllers forcibly release the TextBox before this
		-- observer runs. Keep a short, explicit manual-search lease after the user
		-- focused Zenith so that the same physical key still edits Zenith, not the
		-- game-word detector.
		if WordHelper.manualCaptureUntil > 0 and os.clock() <= WordHelper.manualCaptureUntil then
			pcall(function() searchBox:CaptureFocus() end)
			bridgeBlockedSearchInput(input)
			return
		end

		local letter = LETTER_BY_KEYCODE[input.KeyCode]
		local action = letter
			or (input.KeyCode == Enum.KeyCode.Backspace and "back")
			or ((input.KeyCode == Enum.KeyCode.Space
				or input.KeyCode == Enum.KeyCode.Return
				or input.KeyCode == Enum.KeyCode.KeypadEnter) and "end")
			or nil
		if not action then
			return
		end
		if currentPlaceId == WORD_BOMB_PLACE_ID and WordHelper.localTurn ~= true then
			return
		end

		-- Record activity even when the game's ContextAction does not mark the key
		-- processed. This lets the read-only remote adapter validate the exact key,
		-- but raw movement/chat keys are never added unless the game consumed them.
		local pressedAt = os.clock()
		WordHelper.lastTypingActivity = pressedAt
		WordHelper.lastTypedAction = action
		local knownOwnTurn = WordHelper.localTurn == true
			and (currentPlaceId == WORD_BOMB_PLACE_ID or currentPlaceId == LAST_LETTER_PLACE_ID)
		if not gameProcessed and not knownOwnTurn then return end
		-- With the confirmed hook active, the outgoing call is the sole source of
		-- truth. A second physical fallback would append the same letter twice.
		if WordHelper.exactRemoteHookActive then return end

		-- Wait briefly: an exact game remote or visible-word update wins, preventing
		-- one physical/on-screen key from being counted twice. The exact remote may
		-- occur just before InputBegan on touch controllers, so compare both ways.
		local fallbackDelay = currentPlaceId == WORD_BOMB_PLACE_ID and 0.5 or 0.25
		task.delay(fallbackDelay, function()
			local remoteWon = exactRemoteHandledNear(action, pressedAt)
				or (currentPlaceId ~= WORD_BOMB_PLACE_ID and WordHelper.lastRemoteTyping >= pressedAt)
			if not WordHelper.destroyed and not remoteWon
				and WordHelper.lastVisualUpdate < pressedAt then
				passiveType(action, Config.PassiveResetSeconds)
			end
		end)
	end)

	-- Custom on-screen keyboards: games that draw their own A-Z buttons instead of
	-- using a TextBox. Each key press is read from the button label when clicked.
	local BACK_WORDS = { "back", "del", "borrar", "⌫", "←", "<-" }
	local ENTER_WORDS = { "enter", "submit", "send", "enviar", "ok", "go", "✓", "✔", "space", "espacio" }

	local function buttonKeyLabel(button)
		local text = ""
		pcall(function()
			if button:IsA("TextButton") then text = button.Text end
		end)
		if not string.find(text, "%S") then
			pcall(function()
				local label = button:FindFirstChildWhichIsA("TextLabel", true)
				text = label and label.Text or ""
			end)
		end
		if not string.find(text, "%S") then
			pcall(function()
				for _, attributeName in ipairs({ "Key", "Letter", "Value" }) do
					local value = button:GetAttribute(attributeName)
					if type(value) == "string" and value ~= "" then
						text = value
						break
					end
				end
			end)
		end
		-- Several mobile keyboards use ImageButtons named Q, Delete, or Done.
		if not string.find(text, "%S") then
			pcall(function() text = button.Name end)
		end
		text = string.gsub(text, "<.->", "")
		text = string.lower(string.gsub(text, "^%s*(.-)%s*$", "%1"))
		if string.match(text, "^[a-z]$") then
			return text
		end
		for _, word in ipairs(BACK_WORDS) do
			if text == word or (#word > 2 and string.find(text, word, 1, true)) then
				return "back"
			end
		end
		for _, word in ipairs(ENTER_WORDS) do
			if text == word then
				return "end"
			end
		end
		return nil
	end

	local function bindKeyboardButton(object)
		if WordHelper.keyboardButtons[object] or object:IsDescendantOf(screenGui) then
			return
		end
		local isButton = false
		pcall(function()
			isButton = object:IsA("GuiButton")
		end)
		if not isButton then
			return
		end
		WordHelper.keyboardButtons[object] = true
		connect(object.Activated, function()
			local action = buttonKeyLabel(object)
			if not action then return end
			if currentPlaceId == WORD_BOMB_PLACE_ID and WordHelper.localTurn ~= true then return end
			local pressedAt = os.clock()
			WordHelper.lastTypingActivity = pressedAt
			WordHelper.lastTypedAction = action
			-- Exact remote/visual text wins, preventing duplicate letters even when
			-- Activated is delivered just after the game's outgoing remote.
			local fallbackDelay = currentPlaceId == WORD_BOMB_PLACE_ID and 0.5 or 0.25
			task.delay(fallbackDelay, function()
				local remoteWon = exactRemoteHandledNear(action, pressedAt)
					or (currentPlaceId ~= WORD_BOMB_PLACE_ID and WordHelper.lastRemoteTyping >= pressedAt)
				if not WordHelper.destroyed and not remoteWon
					and WordHelper.lastVisualUpdate < pressedAt then
					passiveType(action, 6)
				end
			end)
		end)
	end

	-- Scan screen GUIs and in-world keyboards (SurfaceGuis on tables, etc.).
	local function scanForKeyboards(root)
		task.spawn(function()
			local ok, descendants = pcall(function()
				return root:GetDescendants()
			end)
			if not ok then
				return
			end
			for index, object in ipairs(descendants) do
				if WordHelper.destroyed then
					return
				end
				bindKeyboardButton(object)
				if index % 400 == 0 then
					task.wait()
				end
			end
		end)
		pcall(function()
			connect(root.DescendantAdded, bindKeyboardButton)
		end)
	end

	-- Place 129866685202296 renders the live entry under InGame.Frame.CurrentWord.
	-- Reading those visible labels is more reliable than guessing the contents of
	-- its shared Packet buffer and remains completely read-only.
	local function guiObjectVisible(object, boundary)
		local current = object
		while current and current ~= boundary do
			local hidden = false
			pcall(function()
				if current:IsA("GuiObject") and not current.Visible then hidden = true end
				if current:IsA("LayerCollector") and not current.Enabled then hidden = true end
			end)
			if hidden then return false end
			current = current.Parent
		end
		return true
	end

	local function visibleText(object)
		local text = ""
		local ok = pcall(function()
			if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
				text = object.Text
			end
		end)
		if not ok then return "" end
		text = string.gsub(tostring(text), "<.->", "")
		text = string.lower(string.gsub(text, "^%s*(.-)%s*$", "%1"))
		return string.match(text, "^[a-z]+$") and text or ""
	end

	local function readLargePromptLetters()
		if currentPlaceId ~= WORD_BOMB_PLACE_ID then return nil end
		local camera = workspaceService.CurrentCamera
		local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
		local records = {}
		local ok, descendants = pcall(function() return playerGui:GetDescendants() end)
		if not ok then return nil end
		for _, object in ipairs(descendants) do
			local isOwnGui = false
			pcall(function() isOwnGui = object:IsDescendantOf(screenGui) end)
			if not isOwnGui and guiObjectVisible(object, playerGui) then
				local text = visibleText(object)
				if #text == 1 then
					local position, size, textSize = Vector2.new(0, 0), Vector2.new(0, 0), 0
					pcall(function()
						position, size, textSize = object.AbsolutePosition, object.AbsoluteSize, object.TextSize
					end)
					local centerX, centerY = position.X + size.X * 0.5, position.Y + size.Y * 0.5
					if centerX >= viewport.X * 0.14 and centerX <= viewport.X * 0.86
						and centerY >= viewport.Y * 0.05 and centerY <= viewport.Y * 0.65
						and (size.Y >= 28 or textSize >= 20) then
						table.insert(records, {
							text = text, x = centerX, y = centerY,
							height = math.max(size.Y, textSize), area = math.max(1, size.X * size.Y),
						})
					end
				end
			end
		end
		local best, bestScore = nil, 0
		for _, seed in ipairs(records) do
			local row, score = {}, 0
			for _, record in ipairs(records) do
				if math.abs(record.y - seed.y) <= math.max(22, seed.height * 0.45) then
					table.insert(row, record)
					score = score + record.area
				end
			end
			if #row >= 1 and #row <= 4 and score > bestScore then best, bestScore = row, score end
		end
		if not best then return nil end
		table.sort(best, function(a, b) return a.x < b.x end)
		local letters = table.create(#best)
		for index, record in ipairs(best) do letters[index] = record.text end
		return table.concat(letters)
	end

	local function pollWordBombPrompt()
		if currentPlaceId ~= WORD_BOMB_PLACE_ID then return end
		local prompt = readLargePromptLetters()
		if not prompt or prompt == "" then return end
		WordHelper.visiblePrompt = prompt
		WordHelper.visiblePromptAt = os.clock()
		-- A visible prompt alone is not proof of ownership: every client can see the
		-- opponent's letters. It may only supplement an already verified local turn.
		if WordHelper.localTurn == true and WordHelper.turnPrompt == "" then
			WordHelper.turnPrompt = prompt
			WordHelper.searchMode = "prefix"
			modeChip.Text = "Empieza con"
			applyDetectedQuery(prompt, "turn", true)
		end
	end

	local function readLastLetterVisibleWord()
		if currentPlaceId ~= LAST_LETTER_PLACE_ID then return nil end
		local inGame = playerGui:FindFirstChild("InGame")
		local root = inGame and inGame:FindFirstChild("CurrentWord", true)
		if not root then root = playerGui:FindFirstChild("CurrentWord", true) end
		if not root or not guiObjectVisible(root, playerGui) then return nil end

		local letters, wholeWords = {}, {}
		local candidates = { root }
		local ok, descendants = pcall(function() return root:GetDescendants() end)
		if ok then
			for _, object in ipairs(descendants) do table.insert(candidates, object) end
		end
		for _, object in ipairs(candidates) do
			if guiObjectVisible(object, root) then
				local text = visibleText(object)
				if #text == 1 then
					local layoutOrder, position = 0, Vector2.new(0, 0)
					pcall(function()
						layoutOrder = object.LayoutOrder
						position = object.AbsolutePosition
					end)
					table.insert(letters, { text = text, order = layoutOrder, position = position })
				elseif #text >= 2 and #text <= 32 then
					local name = string.lower(tostring(object.Name or ""))
					local score = (string.find(name, "word", 1, true) and 4 or 0)
						+ (string.find(name, "text", 1, true) and 2 or 0)
						+ (object == root and 8 or 0)
					table.insert(wholeWords, { text = text, score = score })
				end
			end
		end
		if #letters > 0 then
			table.sort(letters, function(a, b)
				if a.order ~= b.order then return a.order < b.order end
				if math.abs(a.position.Y - b.position.Y) > 3 then return a.position.Y < b.position.Y end
				return a.position.X < b.position.X
			end)
			local output = table.create(#letters)
			for index, record in ipairs(letters) do output[index] = record.text end
			return table.concat(output)
		end
		table.sort(wholeWords, function(a, b)
			if a.score ~= b.score then return a.score > b.score end
			return #a.text > #b.text
		end)
		return wholeWords[1] and wholeWords[1].text or ""
	end

	local function pollVisibleGameWord()
		local word = readLastLetterVisibleWord()
		if word == nil then
			if currentPlaceId == LAST_LETTER_PLACE_ID and WordHelper.lastVisualWord ~= ""
				and os.clock() - WordHelper.lastVisualUpdate > 0.5 then
				WordHelper.lastVisualWord = ""
				WordHelper.passiveBuffer = ""
				applyDetectedQuery("", "visual", true)
			end
			return
		end
		if word == WordHelper.lastVisualWord then return end
		local now, previous = os.clock(), WordHelper.lastVisualWord
		WordHelper.lastVisualWord = word
		WordHelper.lastVisualUpdate = now
		if previous ~= "" and WordHelper.lastTypedAction == "end"
			and now - WordHelper.lastTypingActivity <= 1 then
			markUsed(previous)
		end
		WordHelper.passiveBuffer = word
		WordHelper.lastPassiveInput = now
		applyDetectedQuery(word, "visual", true)
	end

	-- ===== Remote typing detection (executor only: needs hookmetamethod) =====
	-- Formats observed in real custom-keyboard games:
	--  * Per-letter remote: FireServer(id, true, "a") adds "a", FireServer(id, false, "a")
	--    deletes it, FireServer(id, true, " ", n) submits the word.
	--  * Full-text events: BindableEvent "TypingStopped":Fire("anything"), and the
	--    server echo OnClientEvent("syncFilteredWord", "eam").
	--  * Packet / ByteNet style libraries: everything is packed in a buffer; strings
	--    inside are length-prefixed, so letters are read straight from the bytes.
	local function bufferStrings(value)
		local output, seen = {}, {}
		local ok, raw = pcall(buffer.tostring, value)
		if not ok or type(raw) ~= "string" then return output end
		local size = #raw
		local function addCandidate(startIndex, length)
			if length < 1 or length > 32 or startIndex < 1 or startIndex + length - 1 > size then return end
			local text = string.sub(raw, startIndex, startIndex + length - 1)
			if string.match(text, "^[%a ]+$") and not seen[text] then
				seen[text] = true
				table.insert(output, text)
			end
		end
		-- Packet libraries seen in the supplied traces use U8 for String and U16
		-- for StringLong. Scan both little- and big-endian U16 without assuming the
		-- packet's leading numeric ID.
		for index = 1, size do
			local first = string.byte(raw, index)
			addCandidate(index + 1, first)
			if index < size then
				local second = string.byte(raw, index + 1)
				addCandidate(index + 2, first + second * 256)
				addCandidate(index + 2, first * 256 + second)
			end
		end
		return output
	end

	local function collectRemoteValues(arguments)
		local strings, hasTrue, hasFalse, fromBuffer = {}, false, false, false
		local function visit(value, depth)
			local kind = typeof(value)
			if kind == "string" then
				table.insert(strings, value)
			elseif kind == "boolean" then
				if value then
					hasTrue = true
				else
					hasFalse = true
				end
			elseif kind == "buffer" then
				fromBuffer = true
				for _, text in ipairs(bufferStrings(value)) do
					table.insert(strings, text)
				end
			elseif kind == "table" and depth < 2 then
				for _, nested in pairs(value) do
					visit(nested, depth + 1)
				end
			end
		end
		for index = 1, arguments.n do
			visit(arguments[index], 0)
		end
		return strings, hasTrue, hasFalse, fromBuffer
	end

	local TYPING_NAME_PARTS = { "typ", "word", "text", "answer", "guess", "input", "letter", "spell" }
	local function nameLooksLikeTyping(instance)
		local ok, name = pcall(function()
			return string.lower(instance.Name)
		end)
		if not ok then
			return false
		end
		for _, part in ipairs(TYPING_NAME_PARTS) do
			if string.find(name, part, 1, true) then
				return true
			end
		end
		return false
	end

	local function remoteSetText(text)
		local fragment = string.match(string.lower(text), "([a-z]+)$") or ""
		local now = os.clock()
		WordHelper.passiveBuffer = fragment
		WordHelper.lastPassiveInput = now
		WordHelper.lastRemoteTyping = now
		applyDetectedQuery(fragment, "remote", true)
	end

	local function trustRemote(remote)
		local now = os.clock()
		WordHelper.trustedRemotes[remote] = now
		WordHelper.lastRemoteTyping = now
		if WordHelper.listenedRemotes[remote] then
			return
		end
		WordHelper.listenedRemotes[remote] = true
		pcall(function()
			setStatus("Teclado del juego detectado: " .. remote.Name, Config.Secondary)
		end)
		-- Server echo of the current word on the same remote ("syncFilteredWord", "eam").
		-- Only used right after you typed, so an opponent's word is not picked up.
		pcall(function()
			if not remote:IsA("RemoteEvent") then
				return
			end
			connect(remote.OnClientEvent, function(...)
				-- In the confirmed two-player game this echo also contains the enemy's
				-- word. The local MatchStatus prompt and local outgoing call are the only
				-- accepted sources there.
				if currentPlaceId == WORD_BOMB_PLACE_ID then return end
				if os.clock() - WordHelper.lastTypingActivity > Config.TypingActivitySeconds then return end
				local arguments = table.pack(...)
				for index = 1, arguments.n - 1 do
					local key, value = arguments[index], arguments[index + 1]
					if type(key) == "string" and string.find(string.lower(key), "word", 1, true)
						and type(value) == "string" and string.match(string.lower(value), "^[a-z ]*$") then
						remoteSetText(value)
						return
					end
				end
			end)
		end)
	end

	local letterCandidates = setmetatable({}, { __mode = "k" })

	local function instancePath(instance)
		local ok, path = pcall(function() return instance:GetFullName() end)
		return ok and type(path) == "string" and path or ""
	end

	local function handleKnownWordBombCall(remote, method, arguments)
		if currentPlaceId ~= WORD_BOMB_PLACE_ID or method ~= "FireServer" or arguments.n < 3 then
			return false
		end
		local sessionId, adding, value = arguments[1], arguments[2], arguments[3]
		local path = instancePath(remote)
		if type(sessionId) ~= "string" or not string.match(sessionId, "^[%x%-]+$")
			or type(adding) ~= "boolean" or type(value) ~= "string"
			or not string.find(path, "MatchmakingFolders", 1, true) then
			return false
		end
		local text = string.lower(value)
		if not (string.match(text, "^[a-z]$") or text == " ") then return false end

		local now = os.clock()
		local signature = sessionId .. "|" .. tostring(adding) .. "|" .. text
		if WordHelper.lastRemoteSignature == signature and now - WordHelper.lastRemoteSignatureAt <= 0.03 then
			return true
		end
		WordHelper.lastRemoteSignature = signature
		WordHelper.lastRemoteSignatureAt = now
		local remoteAction = text == " " and "end" or (adding and text or "back")
		trustRemote(remote)
		WordHelper.lastRemoteAction = remoteAction
		WordHelper.lastRemoteActionAt = now
		WordHelper.localTurn = true -- a local outgoing typing call is authoritative
		WordHelper.turnOwner = localPlayer
		WordHelper.turnSource = "remote"
		WordHelper.lastLocalRemoteCall = now
		WordHelper.lastRemoteTyping = now
		if WordHelper.turnPrompt == "" then
			local visiblePrompt = readLargePromptLetters()
			if visiblePrompt and visiblePrompt ~= "" then
				WordHelper.visiblePrompt = visiblePrompt
				WordHelper.visiblePromptAt = now
			elseif now - WordHelper.visiblePromptAt <= 0.75 then
				visiblePrompt = WordHelper.visiblePrompt
			end
			if visiblePrompt and visiblePrompt ~= "" then WordHelper.turnPrompt = visiblePrompt end
		end
		WordHelper.searchMode = "prefix"
		modeChip.Text = "Empieza con"
		if text == " " then
			WordHelper.lastSubmitAt = now
			WordHelper.awaitingSubmission = true
			WordHelper.passiveBuffer = ""
			applyDetectedQuery(WordHelper.turnPrompt, "turn", true)
			return true
		end
		WordHelper.awaitingSubmission = false
		if now - WordHelper.lastPassiveInput > 8 then WordHelper.passiveBuffer = "" end
		if adding then
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer .. text, -32)
		else
			WordHelper.passiveBuffer = string.sub(WordHelper.passiveBuffer, 1, math.max(0, #WordHelper.passiveBuffer - 1))
		end
		WordHelper.lastPassiveInput = now
		applyDetectedQuery(queryForTypedBuffer(WordHelper.passiveBuffer), "remote", true)
		return true
	end

	local function onRemoteCall(remote, method, arguments)
		if WordHelper.destroyed then return end
		if handleKnownWordBombCall(remote, method, arguments) then return end
		-- This place has a confirmed protocol. Its generic remotes and incoming
		-- echoes contain both players, so accepting any other channel can follow the
		-- opponent. Unknown calls are deliberately ignored here.
		if currentPlaceId == WORD_BOMB_PLACE_ID then return end
		local strings, hasTrue, hasFalse, fromBuffer = collectRemoteValues(arguments)
		if #strings == 0 then
			return
		end
		local now = os.clock()
		local trustedAt = WordHelper.trustedRemotes[remote]
		local trusted = trustedAt ~= nil and now - trustedAt < 180
		local recentTyping = now - WordHelper.lastTypingActivity <= Config.TypingActivitySeconds
		local lastAction = WordHelper.lastTypedAction
		local path = instancePath(remote)
		local packetRemote = currentPlaceId == LAST_LETTER_PLACE_ID
			and method == "FireServer"
			and path == "ReplicatedStorage.Modules.Packet.RemoteEvent"

		-- This game's Packet.RemoteEvent carries every system, not only typing. Trust
		-- each packet only when its decoded text exactly matches the key just pressed;
		-- never mark the shared packet channel as globally trusted.
		if packetRemote then
			for _, raw in ipairs(strings) do
				local text = string.lower(raw)
				if recentTyping and string.match(text, "^[a-z]$") and lastAction == text then
					WordHelper.lastRemoteTyping = now
					passiveType(text, 6)
					return
				elseif recentTyping and #text >= 2 and #text <= 32 and string.match(text, "^[a-z]+$")
					and lastAction ~= "back" and lastAction ~= "end" and string.sub(text, -1) == lastAction then
					WordHelper.lastRemoteTyping = now
					remoteSetText(text)
					return
				end
			end
			return
		end

		for _, raw in ipairs(strings) do
			local text = string.lower(raw)
			if (text == " " or text == "\n" or text == "\r") and trusted then
				-- Submit: remember the word as used and start a fresh one.
				trustRemote(remote)
				markUsed(WordHelper.passiveBuffer)
				WordHelper.lastSubmitAt = now
				WordHelper.passiveBuffer = ""
				applyDetectedQuery("", "remote", true)
				return
			elseif string.match(text, "^[a-z]$") and method ~= "Fire" then
				-- Per-letter remote. Require a typing-like name, the observed boolean
				-- protocol, or a packed buffer in addition to a recent matching key.
				local flagged = (hasTrue or hasFalse) and not fromBuffer
				local typed = recentTyping and (lastAction == text or lastAction == "back")
					and (flagged or fromBuffer or nameLooksLikeTyping(remote))
				local earlierLetters = nil
				if flagged and not trusted and not typed then
					local candidate = letterCandidates[remote]
					if candidate and now - candidate.time < 4 then
						candidate.count = candidate.count + 1
					else
						candidate = { count = 1, letters = "" }
						letterCandidates[remote] = candidate
					end
					candidate.time = now
					typed = candidate.count >= 2
					if typed then
						-- Keep the letter(s) seen before the remote was confirmed.
						earlierLetters = candidate.letters
					elseif hasTrue then
						candidate.letters = candidate.letters .. text
					end
				end
				if trusted or typed then
					trustRemote(remote)
					if now - WordHelper.lastPassiveInput > 8 then
						WordHelper.passiveBuffer = ""
					end
					if earlierLetters then
						WordHelper.passiveBuffer = earlierLetters
					end
					local current = WordHelper.passiveBuffer
					if hasFalse and not hasTrue then
						-- (id, false, letter): that letter was deleted.
						current = string.sub(current, 1, math.max(0, #current - 1))
					else
						current = string.sub(current .. text, -32)
					end
					WordHelper.passiveBuffer = current
					WordHelper.lastPassiveInput = now
					applyDetectedQuery(current, "remote", true)
					return
				end
			elseif #text >= 2 and #text <= 40 and string.match(text, "^[a-z ]+$") then
				-- Full current text (e.g. TypingStopped("anything") or a buffer string).
				if method == "Fire" and WordHelper.lastSubmitAt and now - WordHelper.lastSubmitAt < 2 then
					-- Echo of the word that was just submitted.
					markUsed(text)
					return
				end
				local fragment = string.match(text, "([a-z]+)$") or ""
				local typed = recentTyping and lastAction ~= nil
					and (lastAction == "back" or lastAction == "end" or string.sub(fragment, -1) == lastAction)
				local typingNamed = nameLooksLikeTyping(remote)
				if trusted or (recentTyping and typingNamed and (typed or method == "Fire")) then
					trustRemote(remote)
					remoteSetText(text)
					return
				end
			end
		end
	end

	-- Exposed for deterministic adapters/tests; it only observes arguments and never
	-- invokes a remote or changes game controls.
	WordHelper.ProcessRemoteCall = onRemoteCall

	local remoteHook = sharedEnvironment.ZenithWordRemoteHookV3
	if type(remoteHook) ~= "table" then
		local hookMetamethod = zenithFileFunction("hookmetamethod")
		local getNamecallMethod = zenithFileFunction("getnamecallmethod")
		local newCClosure = zenithFileFunction("newcclosure") or function(callback)
			return callback
		end
		if hookMetamethod and getNamecallMethod then
			-- Installed once per game session; re-executions just swap the handler.
			local state = {}
			local installed = pcall(function()
				local original
				original = hookMetamethod(game, "__namecall", newCClosure(function(self, ...)
					local handler = state.handler
					if handler then
						local method = getNamecallMethod()
						if method == "FireServer" or method == "InvokeServer" or method == "Fire" then
							local arguments = table.pack(...)
							task.defer(pcall, handler, self, method, arguments)
						end
					end
					return original(self, ...)
				end))
			end)
			if installed then
				remoteHook = state
				sharedEnvironment.ZenithWordRemoteHookV3 = state
			end
		end
	end
	if type(remoteHook) == "table" then
		remoteHook.handler = onRemoteCall
		WordHelper.remoteHook = remoteHook
	end
	WordHelper.exactRemoteHookActive = currentPlaceId == WORD_BOMB_PLACE_ID and type(remoteHook) == "table"

	-- Avoid walking every Workspace descendant or binding every key button in the
	-- confirmed game when its exact outgoing remote is available. This was both a
	-- major mobile cost and a second source for the same touch letter.
	if not WordHelper.exactRemoteHookActive then
		scanForKeyboards(playerGui)
		if currentPlaceId ~= WORD_BOMB_PLACE_ID then scanForKeyboards(workspaceService) end
	end

	local focusPollAccumulator, visualPollAccumulator = 0, 0
	connect(runService.Heartbeat, function(deltaTime)
		if currentPlaceId == LAST_LETTER_PLACE_ID then
			visualPollAccumulator = visualPollAccumulator + deltaTime
			if visualPollAccumulator >= Config.VisualPollSeconds then
				visualPollAccumulator = 0
				pollVisibleGameWord()
			end
		end
		focusPollAccumulator = focusPollAccumulator + deltaTime
		if focusPollAccumulator < Config.FocusPollSeconds then return end
		focusPollAccumulator = 0
		local focusedTextBox = userInputService:GetFocusedTextBox()
		if focusedTextBox and focusedTextBox ~= searchBox then
			bindFocusedTextBox(focusedTextBox)
			-- Polling as well as listening makes detection survive protected/custom
			-- TextBoxes whose property-changed signal cannot be connected to.
			syncFromFocusedTextBox(focusedTextBox)
		elseif not focusedTextBox and WordHelper.activeExternalTextBox then
			disconnectFocusedTextBox()
		end
	end)

	local function bindViewport()
		local camera = workspaceService.CurrentCamera
		if camera then
			connect(camera:GetPropertyChangedSignal("ViewportSize"), updateResponsivePlacement)
		end
		updateResponsivePlacement()
	end
	connect(workspaceService:GetPropertyChangedSignal("CurrentCamera"), bindViewport)
	bindViewport()

	sharedEnvironment.ZenithVocabulary = WordHelper
	sharedEnvironment.ZenithAssistant = WordHelper
	sharedEnvironment.ZenithWordHelper = WordHelper -- compatibility alias
	if type(shared) == "table" then
		shared.ZenithVocabulary = WordHelper
		shared.ZenithAssistant = WordHelper
		shared.ZenithWordHelper = WordHelper
	end

	WordHelper:SetWords(parseDictionary(FALLBACK_WORDS))

	-- Upgrade to the full cached/downloaded English dictionary in the background.
	-- Read a cached file, otherwise download it from the first working URL and cache it.
	local function loadCachedOrDownload(fileName, urls)
		if readFile then
			local ok, value = pcall(function()
				if isFile and not isFile(fileName) then
					return nil
				end
				return readFile(fileName)
			end)
			if ok and type(value) == "string" and #value > 1000 then
				return value
			end
		end
		for _, url in ipairs(urls) do
			local value = httpGet(url)
			if value and #value > 1000 then
				if writeFile then
					pcall(function()
						if makeFolder then
							pcall(makeFolder, "Zenith")
						end
						writeFile(fileName, value)
					end)
				end
				return value
			end
		end
		return nil
	end

	task.spawn(function()
		setStatus("Cargando diccionario...", Config.Muted)

		local frequency = loadCachedOrDownload(Config.FrequencyFile, { Config.FrequencyUrl })
		if frequency and not WordHelper.destroyed then
			local rank = {}
			local position, scanned = 0, 0
			for word in string.gmatch(frequency, "[^\r\n]+") do
				scanned = scanned + 1
				if scanned % 5000 == 0 then
					if WordHelper.destroyed then return end
					task.wait()
				end
				word = normalize(word)
				if string.match(word, "^[a-z]+$") and not rank[word] then
					position = position + 1
					rank[word] = position
				end
			end
			WordHelper.frequencyRank = rank
		end

		local content = loadCachedOrDownload(Config.DictionaryFile, Config.DictionaryUrls)
		if WordHelper.destroyed then
			return
		end
		if content then
			local words = parseDictionary(content)
			if WordHelper.destroyed then return end
			if #words > 0 then
				installCleanWords(words)
				if WordHelper.destroyed then return end
				WordHelper.dictionaryLabel = "ENABLE dictionary"
			end
		end
		scheduleRefresh()
		task.wait(0.1)
		if not WordHelper.destroyed and WordHelper.currentQuery == "" then
			setStatus(content and string.format("Diccionario listo: %d palabras", #WordHelper.words)
				or "Diccionario básico (sin conexión)", Config.Muted)
		end
	end)

	-- Keep the existing twelve-hour entitlement without adding a timer/stat panel.
	standardExpiresAt = tonumber(standardExpiresAt) or (os.time() + ZenithKeyLog.DurationSeconds)
	task.spawn(function()
		while not WordHelper.destroyed do
			if standardExpiresAt <= os.time() then
				WordHelper:Destroy()
				launchKeySystem(function(newExpiresAt)
					launchWordHelper(newExpiresAt)
				end)
				return
			end
			task.wait(1)
		end
	end)

	-- Small, non-blocking entrance; there is deliberately no BlurEffect.
	local finalScale = targetScale()
	windowScale.Scale = finalScale * 0.9
	updateResponsivePlacement()
	windowScale.Scale = finalScale * 0.9
	tween(windowScale, OPEN_TWEEN, { Scale = finalScale })

	return WordHelper
end

-- Bootstrap: skip the Key System while the logged key is still valid.
local function startZenith()
	local expiresAt = ZenithKeyLog.ReadExpiry()
	if expiresAt and expiresAt > os.time() then
		launchWordHelper(expiresAt)
	else
		launchKeySystem(function(newExpiresAt)
			launchWordHelper(newExpiresAt)
		end)
	end
end

startZenith()

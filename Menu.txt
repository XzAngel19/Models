---@module Menu.Utility
-- Instance helpers shared by every component.
local Utility = {}

---Create an Instance, apply properties, then parent it.
---@param className string
---@param properties table
---@param parent Instance?
---@return Instance
function Utility.create(className, properties, parent)
	local instance = Instance.new(className)
	for propertyName, propertyValue in pairs(properties) do
		instance[propertyName] = propertyValue
	end
	if parent then
		instance.Parent = parent
	end
	return instance
end

---Round the corners of an instance.
---@param parent Instance
---@param radius number
function Utility.corner(parent, radius)
	Utility.create("UICorner", {
		CornerRadius = UDim.new(0, radius),
	}, parent)
end

---Draw a border stroke around an instance.
---@param parent Instance
---@param color Color3
---@param thickness number
function Utility.stroke(parent, color, thickness)
	Utility.create("UIStroke", {
		Color = color,
		Thickness = thickness,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

---Resolve a safe parent for the ScreenGui using an executor fallback chain.
---@return Instance
function Utility.getSafeParent()
	-- Prefer the executor's hidden UI container.
	if gethui then
		local success, hiddenGui = pcall(gethui)
		if success and hiddenGui then
			return hiddenGui
		end
	end
	-- Fall back to CoreGui.
	local coreSuccess, coreGui = pcall(function()
		return game:GetService("CoreGui")
	end)
	if coreSuccess and coreGui then
		return coreGui
	end
	-- Fall back to PlayerGui, then Players as a last resort.
	local localPlayer = playersService.LocalPlayer
	if localPlayer then
		local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
		if playerGui then
			return playerGui
		end
	end
	return playersService
end

---Visibility scale so the window still fits smaller viewports.
---@return number
function Utility.getWindowScale()
	local camera = workspace.CurrentCamera
	local viewport = camera and camera.ViewportSize or Vector2.new(1920, 1080)
	if viewport.Y >= WINDOW_HEIGHT + 40 then
		return 1
	end
	return math.max(0.55, (viewport.Y - 40) / WINDOW_HEIGHT)
end

---@module Menu.Icons
-- Outline icons drawn from bars so no external assets are required.
local Icons = {}

---Draw one bar between two points of a 26x26 icon canvas.
---@param parent Instance
---@param x1 number
---@param y1 number
---@param x2 number
---@param y2 number
---@param thickness number
---@param color Color3
---@return Frame
function Icons.segment(parent, x1, y1, x2, y2, thickness, color)
	local deltaX = x2 - x1
	local deltaY = y2 - y1
	local length = math.sqrt(deltaX * deltaX + deltaY * deltaY)
	local bar = Utility.create("Frame", {
		Name = "Segment",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset((x1 + x2) / 2, (y1 + y2) / 2),
		Size = UDim2.fromOffset(length, thickness),
		Rotation = math.deg(math.atan2(deltaY, deltaX)),
		BackgroundColor3 = color,
		BorderSizePixel = 0,
	}, parent)
	Utility.corner(bar, thickness / 2)
	return bar
end

---Draw a small filled dot.
---@param parent Instance
---@param x number
---@param y number
---@param size number
---@param color Color3
---@return Frame
function Icons.dot(parent, x, y, size, color)
	local dot = Utility.create("Frame", {
		Name = "Dot",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(x, y),
		Size = UDim2.fromOffset(size, size),
		BackgroundColor3 = color,
		BorderSizePixel = 0,
	}, parent)
	Utility.corner(dot, size / 2)
	return dot
end

---Write the cropped bitmaps to the executor workspace and resolve them.
---@return table?
local function loadIconAssets()
	if not (writefile and base64decode and getcustomasset and isfile and isfolder and makefolder) then
		return nil
	end
	local success, result = pcall(function()
		if not isfolder(ICON_FOLDER) then
			makefolder(ICON_FOLDER)
		end
		local ids = {}
		for iconName, encoded in pairs(ICON_ASSETS) do
			local path = ICON_FOLDER .. "/" .. iconName .. ".png"
			if not isfile(path) then
				writefile(path, base64decode(encoded))
			end
			ids[iconName] = getcustomasset(path)
		end
		return ids
	end)
	if success and type(result) == "table" then
		return result
	end
	return nil
end

local ICON_IDS = loadIconAssets()

---Build an icon by name into a parent and return its drawable parts.
---@param name string
---@param parent Instance
---@param color Color3
---@return table
function Icons.build(name, parent, color)
	local rect = ICON_RECTS[name]
	-- Prefer the cropped reference bitmap when the executor can host it.
	if ICON_IDS and ICON_IDS[name] and rect then
		local image = Utility.create("ImageLabel", {
			Name = name .. "Icon",
			Position = UDim2.fromOffset(rect[1], rect[2]),
			Size = UDim2.fromOffset(rect[3], rect[4]),
			BackgroundTransparency = 1,
			Image = ICON_IDS[name],
			ImageColor3 = color,
		}, parent)
		return { images = { image }, bars = {} }
	end
	-- Vector fallback keeps the menu usable without file access.
	local canvas = Utility.create("Frame", {
		Name = name .. "Icon",
		Position = UDim2.fromOffset(14, 11),
		Size = UDim2.fromOffset(ICON_SIZE, ICON_SIZE),
		BackgroundTransparency = 1,
	}, parent)
	local data = ICON_DATA[name]
	if not data then
		return { images = {}, bars = {} }
	end
	for _, segmentData in ipairs(data.stroke) do
		Icons.segment(
			canvas,
			segmentData[1],
			segmentData[2],
			segmentData[3],
			segmentData[4],
			ICON_STROKE,
			color
		)
	end
	for _, dotData in ipairs(data.dots or {}) do
		Icons.dot(canvas, dotData[1], dotData[2], dotData[3], color)
	end
	local parts = {}
	for _, child in ipairs(canvas:GetDescendants()) do
		if child:IsA("Frame") then
			table.insert(parts, child)
		end
	end
	return { images = {}, bars = parts }
end

---Build the main window shell.
---@param screenGui ScreenGui
---@return Frame, Frame
function Window.new(screenGui)
	-- Window shell.
	local windowFrame = Utility.create("Frame", {
		Name = "Window",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.5, 0),
		Size = UDim2.fromOffset(WINDOW_WIDTH, WINDOW_HEIGHT),
		BackgroundColor3 = Theme.windowBackground,
		BorderSizePixel = 0,
		ClipsDescendants = true,
	}, screenGui)
	Utility.corner(windowFrame, 10)
	Utility.stroke(windowFrame, Theme.windowStroke, 1)

	-- Thin accent strip along the right edge.
	Utility.create("Frame", {
		Name = "AccentStrip",
		Position = UDim2.fromOffset(WINDOW_WIDTH - 3, 78),
		Size = UDim2.fromOffset(3, WINDOW_HEIGHT - 81),
		BackgroundColor3 = Theme.accentStrip,
		BorderSizePixel = 0,
	}, windowFrame)

	-- Title bar doubles as the drag handle.
	local titleBar = Utility.create("Frame", {
		Name = "TitleBar",
		Size = UDim2.new(1, 0, 0, 64),
		BackgroundTransparency = 1,
		Active = true,
	}, windowFrame)

	Utility.create("TextLabel", {
		Name = "Title",
		Position = UDim2.fromOffset(35, 0),
		Size = UDim2.new(0, 700, 1, 0),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium,
		TextSize = 23,
		TextColor3 = Theme.text,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Center,
		Text = TITLE_TEXT,
	}, titleBar)

	-- Close button drawn from two crossing bars.
	local closeButton = Utility.create("TextButton", {
		Name = "Close",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(WINDOW_WIDTH - 37, 31),
		Size = UDim2.fromOffset(44, 44),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
	}, titleBar)
	local closeCanvas = Utility.create("Frame", {
		Name = "Canvas",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.5, 0),
		Size = UDim2.fromOffset(19, 19),
		BackgroundTransparency = 1,
	}, closeButton)
	Icons.segment(closeCanvas, 0.5, 0.5, 18.5, 18.5, 2.3, Theme.text)
	Icons.segment(closeCanvas, 18.5, 0.5, 0.5, 18.5, 2.3, Theme.text)

	-- Drag state.
	local isDragging = false
	local dragOffset = Vector2.zero

	---@param input InputObject
	local function isPrimaryPointer(input)
		return input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch
	end

	titleBar.InputBegan:Connect(function(input)
		if isPrimaryPointer(input) then
			isDragging = true
			local mouse = userInputService:GetMouseLocation()
			local origin = windowFrame.AbsolutePosition
			dragOffset = Vector2.new(mouse.X - origin.X, mouse.Y - origin.Y)
		end
	end)

	userInputService.InputChanged:Connect(function(input)
		if not isDragging then
			return
		end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		local mouse = userInputService:GetMouseLocation()
		-- dragOffset holds the grab point inside the window, so the target is the new top-left.
		local target = Vector2.new(mouse.X - dragOffset.X, mouse.Y - dragOffset.Y)
		local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1920, 1080)
		local windowSize = windowFrame.AbsoluteSize
		-- Keep the whole window on screen, then convert back to centre-anchored position.
		local clampedX = math.clamp(target.X, 0, math.max(0, viewport.X - windowSize.X))
		local clampedY = math.clamp(target.Y, 0, math.max(0, viewport.Y - windowSize.Y))
		windowFrame.Position = UDim2.fromOffset(clampedX + windowSize.X / 2, clampedY + windowSize.Y / 2)
	end)

	userInputService.InputEnded:Connect(function(input)
		if isPrimaryPointer(input) then
			isDragging = false
		end
	end)

	-- Close simply hides the menu.
	closeButton.MouseButton1Click:Connect(function()
		screenGui.Enabled = false
	end)

	return windowFrame, titleBar
end

		local label = Utility.create("TextLabel", {
			Name = "Label",

			Text = tab.label,
		}, button)

		button.MouseButton1Click:Connect(function()
			onSelect(tab.id)
		end)

		buttons[tab.id] = {
			button = button,
			label = label,
			icons = icons,
		}
	end

---Repaint the sidebar for the active tab.
---@param buttons table
---@param selectedId string
function Navigation.refresh(buttons, selectedId)
	for tabId, entry in pairs(buttons) do
		local isSelected = tabId == selectedId
		local glyphColor = isSelected and Theme.tabSelectedGlyph or Theme.text
		entry.button.BackgroundColor3 = isSelected and Theme.tabSelected or Theme.controlBackground
		entry.label.TextColor3 = glyphColor
		entry.label.TextTransparency = 0
		for _, image in ipairs(entry.icons.images) do
			image.ImageColor3 = glyphColor
		end
		for _, bar in ipairs(entry.icons.bars) do
			bar.BackgroundColor3 = glyphColor
			bar.BackgroundTransparency = 0
		end
	end
end

---Create a slider row (library support; unused by the Main tab).
---@param parent Instance
---@param config table
---@return table
function Controls.Slider(parent, config)
	local row = Utility.create("Frame", {
		Name = config.title,
		Size = UDim2.fromOffset(ROW_WIDTH, 76),
		BackgroundColor3 = Theme.controlBackground,
		BorderSizePixel = 0,
	}, parent)
	Utility.corner(row, 10)
	Utility.stroke(row, Theme.controlStroke, 1)

	Utility.create("TextLabel", {
		Name = "Title",
		Position = UDim2.fromOffset(ROW_PADDING, 12),
		Size = UDim2.new(1, -100, 0, 24),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium,
		TextSize = 20,
		TextColor3 = Theme.text,
		TextXAlignment = Enum.TextXAlignment.Left,
		Text = config.title,
	}, row)

	local valueLabel = Utility.create("TextLabel", {
		Name = "Value",
		AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, -ROW_PADDING, 12),
		Size = UDim2.fromOffset(80, 24),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium,
		TextSize = 18,
		TextColor3 = Theme.textMuted,
		TextXAlignment = Enum.TextXAlignment.Right,
		Text = tostring(config.default or config.min or 0),
	}, row)

	local track = Utility.create("Frame", {
		Name = "Track",
		Position = UDim2.fromOffset(ROW_PADDING, 52),
		Size = UDim2.new(1, -ROW_PADDING * 2, 0, 8),
		BackgroundColor3 = Theme.toggleWell,
		BorderSizePixel = 0,
		Active = true,
	}, row)
	Utility.corner(track, 4)

	local fill = Utility.create("Frame", {
		Name = "Fill",
		Size = UDim2.fromScale(0.5, 1),
		BackgroundColor3 = Theme.toggleOn,
		BorderSizePixel = 0,
	}, track)
	Utility.corner(fill, 4)

	local minValue = config.min or 0
	local maxValue = config.max or 100
	local currentValue = config.default or minValue

	local function refresh(ratio)
		ratio = math.clamp(ratio, 0, 1)
		currentValue = minValue + (maxValue - minValue) * ratio
		fill.Size = UDim2.fromScale(ratio, 1)
		valueLabel.Text = tostring(math.floor(currentValue + 0.5))
	end
	refresh((currentValue - minValue) / (maxValue - minValue))

	local isSliding = false
	local function applyInput(input)
		local ratio = (input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X
		refresh(ratio)
		if config.callback then
			config.callback(currentValue)
		end
	end

	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			isSliding = true
			applyInput(input)
		end
	end)
	userInputService.InputChanged:Connect(function(input)
		if not isSliding then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			applyInput(input)
		end
	end)
	userInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			isSliding = false
		end
	end)

	return {
		frame = row,
		get = function()
			return currentValue
		end,
		set = function(newValue)
			refresh((newValue - minValue) / (maxValue - minValue))
		end,
	}
end

}

---@module Menu.App
-- Assembles the window, sidebar and Main page content.
local App = {}

---Build the whole menu.
function App.build()
	local safeParent = Utility.getSafeParent()

	-- Remove a previous instance on re-execution.
	local existing = safeParent:FindFirstChild(GUI_NAME)
	if existing then
		existing:Destroy()
	end

	-- ScreenGui root.
	local screenGui = Utility.create("ScreenGui", {
		Name = GUI_NAME,
		ResetOnSpawn = false,
		IgnoreGuiInset = true,
		DisplayOrder = 999,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	}, safeParent)

	-- Window shell.
	local windowFrame = Window.new(screenGui)

	-- Fit-to-viewport scaling.
	local scale = Utility.getWindowScale()
	if scale < 1 then
		Utility.create("UIScale", { Scale = scale }, windowFrame)
	end

	-- Sidebar container.
	local sidebar = Utility.create("Frame", {
		Name = "Sidebar",
		Position = UDim2.fromOffset(25, 85),
		Size = UDim2.fromOffset(311, WINDOW_HEIGHT - 85 - 24),
		BackgroundTransparency = 1,
	}, windowFrame)

	-- Content container.
	local content = Utility.create("Frame", {
		Name = "Content",
		Position = UDim2.fromOffset(385, 60),
		Size = UDim2.fromOffset(ROW_WIDTH, WINDOW_HEIGHT - 60 - 12),
		BackgroundTransparency = 1,
	}, windowFrame)

	-- One scrolling page per tab.
	local pages = {}
	for _, tab in ipairs(Tabs) do
		local page = Utility.create("ScrollingFrame", {
			Name = tab.id .. "Page",
			Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
			ScrollBarThickness = 3,
			ScrollBarImageColor3 = Theme.toggleWell,
			CanvasSize = UDim2.new(0, 0, 0, 0),
			ScrollingDirection = Enum.ScrollingDirection.Y,
			Visible = tab.id == "Main",
		}, content)
		pages[tab.id] = page
	end

	-- Sidebar buttons and tab switching.
	local sidebarButtons

	local function selectTab(tabId)
		Navigation.refresh(sidebarButtons, tabId)
		for pageId, page in pairs(pages) do
			page.Visible = pageId == tabId
		end
	end

	sidebarButtons = Navigation.new(sidebar, Tabs, selectTab)
	selectTab("Main")

	-- Layout cursor used to stack sections and rows with exact spacing.
	local cursor = 17
	local nextGap = 0
	local function sectionHeader(title)
		if cursor > 17 then
			cursor = cursor + 18
		end
		Section.header(pages.Main, title, cursor)
		cursor = cursor + 30
		nextGap = 20
	end
	local function placeRow(control, height)
		cursor = cursor + nextGap
		control.frame.Position = UDim2.fromOffset(0, cursor)
		cursor = cursor + height
		nextGap = 12
	end

	-- Skill section.
	sectionHeader("Skill")
	placeRow(Controls.Toggle(pages.Main, {
		title = "Auto Ultimate",
		subtitle = "This is Meaning Auto Use Ultimate",
		default = state.autoUltimate,
		callback = function(enabled)
			state.autoUltimate = enabled
			Features.autoUltimate(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)
	placeRow(Controls.Toggle(pages.Main, {
		title = "Auto Dash",
		subtitle = "This is Meaning Auto Use Dash",
		default = state.autoDash,
		callback = function(enabled)
			state.autoDash = enabled
			Features.autoDash(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)

	-- Aim section.
	sectionHeader("Aim")
	placeRow(Controls.Dropdown(pages.Main, {
		title = "Choose Aim Part",
		options = AIM_PARTS,
		default = state.aimPart,
		callback = function(partName)
			state.aimPart = partName
			Features.aimPart(partName)
		end,
	}), 56)
	placeRow(Controls.Toggle(pages.Main, {
		title = "Aimlock",
		subtitle = "Aimlock = Lock Camera Player",
		default = state.aimlock,
		callback = function(enabled)
			state.aimlock = enabled
			Features.aimlock(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)

	-- Body section.
	sectionHeader("Body")
	placeRow(Controls.Toggle(pages.Main, {
		title = "Anti-Knockback",
		subtitle = "BETA",
		default = state.antiKnockback,
		callback = function(enabled)
			state.antiKnockback = enabled
			Features.antiKnockback(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)
	placeRow(Controls.Toggle(pages.Main, {
		title = "Anti-Stun",
		subtitle = "BETA",
		default = state.antiStun,
		callback = function(enabled)
			state.antiStun = enabled
			Features.antiStun(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)
	placeRow(Controls.Toggle(pages.Main, {
		title = "Anti-Busy",
		subtitle = "BETA",
		default = state.antiBusy,
		callback = function(enabled)
			state.antiBusy = enabled
			Features.antiBusy(enabled)
		end,
	}), TOGGLE_ROW_HEIGHT)

	-- Canvas must cover the stacked content.
	pages.Main.CanvasSize = UDim2.fromOffset(0, cursor + 8)

	-- Home, Farming, Killer and Misc pages stay empty stubs.
	return screenGui
end

---Called when initialization errors.
---@param error string
local function onInitializeError(error)
	warn("Failed to initialize Speed Hub X.")
	warn(error)
	warn(debug.traceback())
end

-- Safely build the GUI and report failures without killing the client thread.
xpcall(App.build, onInitializeError)
